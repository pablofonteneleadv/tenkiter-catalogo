/** Sempre devolve 11 dígitos quando dá: ddd + 9 + nnnnnnnn (ex.: 85991047615).
 * Número antigo sem o nono dígito (10 dígitos) ganha o "9" de volta. Sem DDD (8-9 dígitos)
 * não tem como adivinhar o DDD, então fica como está. */
function normalizarWhats_(w) {
  var d = String(w || '').replace(/\D/g, '');
  if (d.length >= 12 && d.slice(0, 2) === '55') d = d.slice(2);
  if (d.length === 10) d = d.slice(0, 2) + '9' + d.slice(2);
  return d;
}

function acNorm_(s) {
  return String(s == null ? '' : s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').trim();
}
function acWhats_(w) { return normalizarWhats_(w); }
/** Texto vindo de fora nunca vira fórmula na planilha: se começa com = + - @ ganha um apóstrofo (o Sheets guarda como texto puro). */
function acTexto_(v) {
  var t = String(v == null ? '' : v);
  return /^[=+\-@\t\r]/.test(t) ? "'" + t : t;
}
function acBool_(v) {
  if (v === true || v === 1) return true;
  if (v === false || v === 0) return false;
  var s = String(v == null ? '' : v).trim().toLowerCase();
  if (s === 'true' || s === 'sim' || s === '1') return true;
  if (s === 'false' || s === 'nao' || s === 'não' || s === '0') return false;
  return null; // em branco / indefinido -> herda
}

function acAba_(nome, cabecalho, esconder) {
  var ss = acSS_();
  var aba = ss.getSheetByName(nome);
  if (!aba) {
    aba = ss.insertSheet(nome);
    if (cabecalho && cabecalho.length) aba.appendRow(cabecalho);
    if (esconder && aba.hideSheet) aba.hideSheet();
  }
  return aba;
}
function acConfigGet_(chave) {
  var aba = acAba_('_Config', ['Chave', 'Valor'], true);
  var dados = aba.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) if (String(dados[i][0]) === chave) return String(dados[i][1] == null ? '' : dados[i][1]);
  return '';
}
function acConfigSet_(chave, valor) {
  var aba = acAba_('_Config', ['Chave', 'Valor'], true);
  var dados = aba.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) {
    if (String(dados[i][0]) === chave) { aba.getRange(i + 1, 2).setValue(valor); return; }
  }
  aba.appendRow([chave, valor]);
}

/* ---------- senha, hash e token ---------- */
function acBytesHex_(bytes) {
  return bytes.map(function (b) { var v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}
function acSha256Hex_(texto) {
  return acBytesHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, texto, Utilities.Charset.UTF_8));
}
function acHashSenha_(senha) {
  var sal = Utilities.getUuid().replace(/-/g, '').slice(0, 16);
  return 'h1$' + sal + '$' + acSha256Hex_(sal + ':' + senha);
}
function acGuardarSenha_(senha) { return acSenhasEmHash_() ? acHashSenha_(senha) : String(senha); }
function acIguais_(a, b) {
  a = String(a); b = String(b);
  if (a.length !== b.length) return false;
  var r = 0;
  for (var i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}
function acSenhaSimples_(guardada, informado) {
  guardada = String(guardada == null ? '' : guardada);
  informado = String(informado == null ? '' : informado);
  if (!informado) return false;
  if (guardada.indexOf('h1$') === 0) {
    var p = guardada.split('$');
    if (p.length !== 3) return false;
    return acIguais_(acSha256Hex_(p[1] + ':' + informado), p[2]);
  }
  return acIguais_(guardada, informado);
}
function acHmac_(whats, guardada) {
  return acBytesHex_(Utilities.computeHmacSha256Signature(whats + '|' + String(guardada), acSegredo_()));
}
/** Token de sessão (autodescritivo): tk2.<whatsapp>.<assinatura>. Só vale enquanto a senha guardada não mudar. */
function acToken_(whats, guardada) { return 'tk2.' + whats + '.' + acHmac_(whats, guardada); }
function acEhToken_(cred) { var s = String(cred || ''); return s.indexOf('tk1.') === 0 || s.indexOf('tk2.') === 0; }
function acCredencialConfere_(linha, cred) {
  cred = String(cred == null ? '' : cred);
  var whats = acWhats_(linha.whatsapp);
  if (cred.indexOf('tk2.') === 0) {
    var p = cred.split('.');
    return p.length === 3 && p[1] === whats && acIguais_(acHmac_(whats, linha.senha), p[2]);
  }
  if (cred.indexOf('tk1.') === 0) return acIguais_('tk1.' + acHmac_(whats, linha.senha), cred);
  return acSenhaSimples_(linha.senha, cred);
}

/* ---------- leitura das pessoas ---------- */
function acCabecalhos_(sh) {
  var n = sh.getLastColumn();
  if (n < 1) return [];
  return sh.getRange(1, 1, 1, n).getValues()[0].map(function (h) { return String(h).trim().toLowerCase(); });
}
function acObj_(headers, valores, numLinha) {
  var o = { __row: numLinha };
  headers.forEach(function (h, i) { if (h) o[h] = valores[i]; });
  return o;
}
/** Só as linhas de UM WhatsApp (lê a coluna de WhatsApp inteira e depois só as linhas certas -- rápido mesmo com muitos clientes). */
function acLinhasDoWhats_(whatsapp) {
  var alvo = acWhats_(whatsapp);
  if (!alvo) return [];
  var sh = acPlanilha_();
  var ultima = sh.getLastRow();
  if (ultima < 2) return [];
  var headers = acCabecalhos_(sh);
  var iw = headers.indexOf('whatsapp');
  if (iw === -1) return [];
  var col = sh.getRange(2, iw + 1, ultima - 1, 1).getValues();
  var out = [];
  for (var i = 0; i < col.length; i++) {
    if (acWhats_(col[i][0]) === alvo) {
      var n = i + 2;
      out.push(acObj_(headers, sh.getRange(n, 1, 1, headers.length).getValues()[0], n));
    }
  }
  return out;
}
function acTodasLinhas_() {
  var sh = acPlanilha_();
  var dados = sh.getDataRange().getValues();
  if (dados.length < 2) return [];
  var headers = dados[0].map(function (h) { return String(h).trim().toLowerCase(); });
  var out = [];
  for (var i = 1; i < dados.length; i++) {
    if (dados[i].join('').trim() === '') continue;
    out.push(acObj_(headers, dados[i], i + 1));
  }
  return out;
}

function acPerfilPadraoObj_(p) {

  acChaves_().forEach(function (k) { perms[k] = p.perms.indexOf(k) !== -1; });
  return { nome: p.nome, desc: p.desc, perms: perms };
}
/** Lê a aba "Perfis" (editável por você); se não existir, usa os perfis de fábrica. */
function acPerfis_() {
  var mapa = {}, ordem = [];
  var aba = acSS_().getSheetByName('Perfis');
  if (aba && aba.getLastRow() >= 2) {
    var dados = aba.getDataRange().getValues();
    var h = dados[0].map(function (x) { return String(x).trim().toLowerCase(); });
    for (var i = 1; i < dados.length; i++) {
      var nome = String(dados[i][h.indexOf('perfil')] || '').trim();
      if (!nome) continue;
      var perms = {};
      acChaves_().forEach(function (k) {
        var idx = h.indexOf('perm_' + k);
        var b = idx === -1 ? null : acBool_(dados[i][idx]);
        perms[k] = b === true;
      });
      mapa[acNorm_(nome)] = { nome: nome, desc: String(dados[i][h.indexOf('descricao')] || ''), perms: perms };
      ordem.push(acNorm_(nome));
    }
  }
  if (!ordem.length) {
    AC_PERFIS_PADRAO.forEach(function (p) { var o = acPerfilPadraoObj_(p); mapa[acNorm_(p.nome)] = o; ordem.push(acNorm_(p.nome)); });
  }
  return { mapa: mapa, ordem: ordem };
}
/** Modelo de fábrica de um perfil (usado se o nome não existir mais na aba Perfis: ninguém fica sem acesso por causa de um renomeio). */
function acPerfilDeFabrica_(nome) {
  var p = AC_PERFIS_PADRAO.filter(function (x) { return acNorm_(x.nome) === acNorm_(nome); })[0];
  return p ? acPerfilPadraoObj_(p) : null;
}
function acPerfilLegado_(nivel) {
  var n = parseInt(nivel, 10) || 1;
  if (n >= 10) return 'Admin total';
  if (n >= 3) return 'Funcionário'; // 4 a 9 (não usados pelo painel) contam como Funcionário
  if (n === 2) return 'Estagiário';
  return 'Aluno'; // quem se cadastrou no portal é Aluno; "Novato" é escolha manual da gestão
}
function acNivelDe_(perms) {
  if (perms.gerir_acessos) return 10;
  if (perms.manual_3) return 3;
  if (perms.manual_2) return 2;
  if (perms.portal || perms.manual_1) return 1;
  return 0;
}
/** Permissões efetivas de UMA linha: caixinha marcada/desmarcada vale; em branco herda do perfil (ou do Nivel antigo). */
function acPermsDaLinha_(linha, perfis) {
  perfis = perfis || acPerfis_();
  var nomePerfil = String(linha.perfil || '').trim() || acPerfilLegado_(linha.nivel);
  var base = (perfis.mapa[acNorm_(nomePerfil)] || perfis.mapa[acNorm_(acPerfilLegado_(linha.nivel))] || acPerfilDeFabrica_(nomePerfil) || acPerfilDeFabrica_(acPerfilLegado_(linha.nivel)) || { perms: {} }).perms;
  var out = {};
  acChaves_().forEach(function (k) {
    var b = acBool_(linha['perm_' + k]);
    out[k] = b === null ? base[k] === true : b;
  });
  return out;
}
function acPermsLista_(perms) { return acChaves_().filter(function (k) { return perms[k]; }); }

/* ---------- verificação de credenciais ---------- */
/**
 * Confere whatsapp + (senha OU token de sessão). Se o mesmo WhatsApp aparecer em
 * mais de uma linha, une as permissões e fica ativo se qualquer linha estiver ativa.
 * Devolve { ok, usuario, sessao } ou { ok:false, erro, inativa?, bloqueado? }.
 */
function acVerificar_(whatsapp, cred) {
  cred = String(cred == null ? '' : cred);
  if (cred.indexOf('tk2.') === 0) {
    var dono = acWhats_(cred.split('.')[1] || '');
    // um token só fala pelo seu dono: se o pedido cita OUTRO WhatsApp, é tentativa de se passar por alguém
    if (acWhats_(whatsapp) && dono && acWhats_(whatsapp) !== dono) return { ok: false, sessaoInvalida: true, erro: 'Sessão inválida. Entre de novo.' };
    whatsapp = dono || whatsapp;
  }
  var alvo = acWhats_(whatsapp);
  if (!alvo || !cred) return { ok: false, erro: 'WhatsApp ou senha incorretos.' };
  var ehToken = acEhToken_(cred);
  if (!ehToken && acBloqueado_(alvo)) {
    return { ok: false, bloqueado: true, erro: 'Muitas tentativas erradas. Aguarde 15 minutos ou fale com a gestão.' };
  }
  var linhas = acLinhasDoWhats_(alvo);
  var conferem = linhas.filter(function (l) { return acCredencialConfere_(l, cred); });
  var bateu = conferem[0];
  if (!bateu) {
    if (!ehToken) acFalha_(alvo);
    return { ok: false, erro: ehToken ? 'Sessão expirada. Entre de novo.' : 'WhatsApp ou senha incorretos.', sessaoInvalida: ehToken };
  }
  acLimpaFalhas_(alvo);
  var perfis = acPerfis_();
  var ativa = false, nome = '', perms = {}, perfil = '', pessoal = {};
  // só contam as linhas que a credencial abriu E que estão ativas (uma linha antiga suspensa não empresta poder à outra)
  var ativas = conferem.filter(function (l) { return String(l.status || 'Ativo').trim().toLowerCase() !== 'inativo'; });
  ativa = ativas.length > 0;
  ativas.forEach(function (l) {
    nome = String(l.nome || '').trim() || nome;
    var p = acPermsDaLinha_(l, perfis);
    acChaves_().forEach(function (k) { if (p[k]) perms[k] = true; });
    perfil = String(l.perfil || '').trim() || perfil || acPerfilLegado_(l.nivel);
    ['datanascimento', 'instagram', 'cep', 'endereco', 'bairro', 'cidade', 'estado', 'altura', 'peso', 'fotoperfilurl', 'fotocorpourl'].forEach(function (c) {
      var v = l[c];
      if (v !== undefined && v !== '' && v !== null) pessoal[c] = Object.prototype.toString.call(v) === '[object Date]' ? v.toISOString() : String(v);
    });
  });
  if (!ativa) return { ok: false, inativa: true, erro: 'Essa conta está desativada. Fale com a gestão.' };
  var usuario = {
    whatsapp: alvo, name: nome || alvo, status: 'ativo', perfil: perfil,
    level: acNivelDe_(perms), permissoes: acPermsLista_(perms),
    dataNascimento: pessoal.datanascimento || '', instagram: pessoal.instagram || '', cep: pessoal.cep || '',
    endereco: pessoal.endereco || '', bairro: pessoal.bairro || '', cidade: pessoal.cidade || '', estado: pessoal.estado || '',
    altura: pessoal.altura || '', peso: pessoal.peso || '',
    fotoPerfilUrl: pessoal.fotoperfilurl || '', fotoCorpoUrl: pessoal.fotocorpourl || ''
  };
  return { ok: true, usuario: usuario, sessao: acToken_(alvo, bateu.senha) };
}
/** Exige login válido E uma permissão. Devolve { ok, usuario } ou { ok:false, erro, semPermissao? }. */
function acExigir_(whatsapp, cred, permissao) {
  var v = acVerificar_(whatsapp, cred);
  if (!v.ok) return v;
  if (permissao && v.usuario.permissoes.indexOf(permissao) === -1) {
    return { ok: false, semPermissao: true, erro: 'Seu acesso não permite esta ação. Fale com a gestão.' };
  }
  return v;
}

/* ---------- escrita ---------- */
function acGarantirColunas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var faltam = [];
  if (headers.indexOf('perfil') === -1) faltam.push({ chave: 'perfil', nome: 'Perfil' });
  acChaves_().forEach(function (k) { if (headers.indexOf('perm_' + k) === -1) faltam.push({ chave: 'perm_' + k, nome: 'perm_' + k }); });
  if (headers.indexOf('nivel') === -1) faltam.unshift({ chave: 'nivel', nome: 'Nivel' });
  if (headers.indexOf('status') === -1) faltam.unshift({ chave: 'status', nome: 'Status' });
  if (headers.indexOf('cadastro') === -1) faltam.unshift({ chave: 'cadastro', nome: 'Cadastro' });
  if (faltam.length) {
    var col = headers.length + 1;
    faltam.forEach(function (f) { sh.getRange(1, col).setValue(f.nome); col++; });
  }
  return sh;
}
/** Marca as caixinhas visuais nas colunas de permissão (só estética; o valor continua TRUE/FALSE). */
function acCaixinhas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var ultima = sh.getLastRow();
  if (ultima < 2) return;
  var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
  acChaves_().forEach(function (k) {
    var i = headers.indexOf('perm_' + k);
    // validação de caixinha NÃO altera o que já está marcado (insertCheckboxes() desmarcaria tudo)
    if (i !== -1) { try { sh.getRange(2, i + 1, ultima - 1, 1).setDataValidation(regra); } catch (e) {} }
  });
}
/** Caixinhas visuais só na linha nova (quando alguém se cadastra). */
function acCaixinhaLinha_(sh, headers, numLinha) {
  try {
    var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
    acChaves_().forEach(function (k) {
      var i = headers.indexOf('perm_' + k);
      if (i !== -1) sh.getRange(numLinha, i + 1).setDataValidation(regra);
    });
  } catch (e) {}
}
/** Grava só as células que mudaram (uma linha), sem reinterpretar o resto dela. */
function acRegravarLinha_(sh, headers, numLinha, mudancas) {
  var atuais = sh.getRange(numLinha, 1, 1, headers.length).getValues()[0];
  Object.keys(mudancas).forEach(function (chave) {
    var i = headers.indexOf(chave);
    if (i === -1) return;
    var novo = mudancas[chave];
    if (String(atuais[i]) === String(novo)) return; // já está assim: não mexe
    var celula = sh.getRange(numLinha, i + 1);
    // texto puro nas colunas de texto: preserva zeros à esquerda (senha "0123") e nunca vira fórmula
    if (typeof novo === 'string' && chave !== 'nivel' && chave.indexOf('perm_') !== 0) celula.setNumberFormat('@');
    celula.setValue(novo);
  });
}
function acMudancasPerfil_(perfilNome, perms) {
  var m = { perfil: perfilNome, nivel: acNivelDe_(perms) };
  acChaves_().forEach(function (k) { m['perm_' + k] = perms[k] === true; });
  return m;
}

  var aba = acAba_('Acessos_Log', ['DataHora', 'Quem', 'QuemWhatsApp', 'Alvo', 'Acao', 'Detalhe']);
  aba.appendRow([new Date(), quemNome || '', quemWhats || '', alvo || '', acao || '', detalhe || '']);
}
/** Cria uma pessoa nova com o perfil dado (todas as caixinhas explícitas). */
function acRegistrar_(nome, whatsapp, senha, perfilNome, extras) {
  var w = acWhats_(whatsapp);
  nome = String(nome || '').trim();
  senha = String(senha || '');
  if (!nome || !w || !senha) return { ok: false, erro: 'Nome, WhatsApp e senha são obrigatórios.' };
  if (senha.length < 4) return { ok: false, erro: 'A senha precisa ter pelo menos 4 caracteres.' };
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    if (acLinhasDoWhats_(w).length) return { ok: false, erro: 'Já existe uma conta com esse WhatsApp.' };
    var sh = acGarantirColunas_();
    var headers = acCabecalhos_(sh);
    var perfis = acPerfis_();
    var perfil = perfis.mapa[acNorm_(perfilNome)];
    if (!perfil) {
      // perfil apagado/renomeado na aba Perfis: usa o modelo de fábrica (nunca "o primeiro da lista", que poderia ser Admin)
      var padrao = AC_PERFIS_PADRAO.filter(function (p) { return acNorm_(p.nome) === acNorm_(perfilNome); })[0];
      if (!padrao) return { ok: false, erro: 'Perfil não encontrado.' };
      perfil = acPerfilPadraoObj_(padrao);
    }
    var m = acMudancasPerfil_(perfil.nome, perfil.perms);
    m.nome = nome; m.whatsapp = w; m.senha = acGuardarSenha_(senha); m.status = 'Ativo';
    m.cadastro = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss");
    extras = extras || {};
    Object.keys(extras).forEach(function (k) { if (extras[k]) m[String(k).toLowerCase()] = extras[k]; });
    var linha = headers.map(function (h) { return m[h] !== undefined ? m[h] : ''; });
    var novaLinha = sh.getLastRow() + 1;
    // texto puro nas colunas de texto (não vira fórmula, não perde zero à esquerda)
    headers.forEach(function (h, i) {
      if (h && h !== 'nivel' && h.indexOf('perm_') !== 0 && linha[i] !== '') sh.getRange(novaLinha, i + 1).setNumberFormat('@');
    });
    sh.getRange(novaLinha, 1, 1, headers.length).setValues([linha]);
    acCaixinhaLinha_(sh, headers, novaLinha);
  } finally {
    lock.releaseLock();
  }
  return { ok: true };
}

/**
 * Lê a aba "Manuais" inteira e devolve uma lista pronta pro site consumir,
 * já ordenada pela coluna Ordem. O CacheJSON é o conteúdo (partes/capítulos/
 * blocos) da última vez que alguém clicou em "Sincronizar agora".
 */
function lerManuais_() {
  var sheet = getManuaisSheet_();
  var dados = sheet.getDataRange().getValues();
  dados.shift(); // cabeçalho
  return dados
    .filter(function (r) { return String(r[0]).trim() !== ''; })
    .map(function (r) {
      var parts = [];
      try { parts = JSON.parse(r[6] || '[]'); } catch (e) { parts = []; }
      return {
        id: String(r[0]),
        titulo: r[1] || '',
        blurb: r[2] || '',
        nivelMinimo: parseInt(r[3], 10) || 1,
        docId: r[4] || '',
        docUrl: r[4] ? ('https://docs.google.com/document/d/' + r[4] + '/edit') : '',
        ordem: Number(r[5]) || 0,
        parts: parts,
        atualizadoEm: r[7] instanceof Date ? r[7].toISOString() : String(r[7] || '')
      };
    })
    .sort(function (a, b) { return a.ordem - b.ordem; });
}

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'funcionarios';

  if (action === 'config_quiz') {
    var sheet = getConfigQuizSheet_();
    var rows = sheet.getDataRange().getValues();
    var headers = rows.shift().map(function (h) { return String(h).trim().toLowerCase(); });
    var configs = rows
      .filter(function (r) { return r.join("").trim() !== ""; })
      .map(function (r) {
        var obj = {};
        headers.forEach(function (h, i) { obj[h] = r[i]; });
        return obj;
      });
    return jsonOut_({ ok: true, configs: configs });
  }

  if (action === 'manuais_lista') {
    return jsonOut_({ ok: true, manuais: lerManuais_() });
  }

  if (action === 'versao') {
    return jsonOut_({ ok: true, versao: '4.1', acessos: true, curriculos: true, listaFechada: listaFechada_() });
  }

  // padrão: lista de funcionários (login e níveis) -- formato ANTIGO, só existe
  // enquanto a lista não for fechada (fecharListaPublica). Depois disso o
  // site novo faz login pelo servidor e nada sensível sai por aqui.
  if (listaFechada_()) return jsonOut_({ ok: true, funcionarios: [], listaFechada: true });
  return jsonOut_({ ok: true, funcionarios: listaLegadaSegura_() });
}

  var dataNascimento = String(data.dataNascimento || '').trim();
  var instagram = String(data.instagram || '').trim();

  if (!nome || !whatsapp || !senha) {
    return jsonOut_({ ok: false, erro: 'Nome, WhatsApp e senha são obrigatórios.' });
  }
  // Cadastro aberto ao público sempre entra com o perfil mais básico (Aluno, ou
  // Cliente se vier do catálogo). Quem sobe de perfil é o Admin total, no painel.
  var perfilInicial = data.origem === 'catalogo' ? 'Cliente' : 'Aluno';
  var cacheReg = CacheService.getScriptCache();
  var feitosReg = parseInt(cacheReg.get('reg_hora') || '0', 10);
  if (feitosReg >= 120) return jsonOut_({ ok: false, erro: 'Muitos cadastros agora. Tente de novo em alguns minutos.' });
  cacheReg.put('reg_hora', String(feitosReg + 1), 3600);
  var r = acRegistrar_(nome, whatsapp, senha, perfilInicial, {});
  if (!r.ok && perfilInicial === 'Aluno') {
    // Cliente do catálogo que agora se cadastra no portal: com a MESMA senha, sobe de Cliente para Aluno (nunca mais que isso)
    var ex = acVerificar_(whatsapp, senha);
    if (ex.ok && ex.usuario && acNorm_(ex.usuario.perfil) === 'cliente') {
      var up = acessosDefinirPerfil_(whatsapp, 'Aluno', { name: ex.usuario.name, whatsapp: ex.usuario.whatsapp });
      if (up && up.ok) r = { ok: true };
    }
  }
  if (!r.ok) return jsonOut_({ ok: false, erro: r.erro });
  if (dataNascimento) gravarCampoFuncionario_(whatsapp, 'datanascimento', 'DataNascimento', dataNascimento);
  if (instagram) gravarCampoFuncionario_(whatsapp, 'instagram', 'Instagram', instagram);
  // site novo já entra logado; site antigo ignora esses campos extras
  var v = acVerificar_(whatsapp, senha);
  if (v.ok) return jsonOut_({ ok: true, usuario: v.usuario, sessao: v.sessao });
  return jsonOut_({ ok: true });
}

 * criando a coluna no fim da planilha na hora se ela ainda não existir --
 * é assim que os campos novos do perfil (data de nascimento, Instagram,
 * endereço, altura, peso, fotos...) vão aparecendo na planilha sozinhos,
 * sem precisar editar a estrutura dela na mão.

function gravarCampoFuncionario_(whatsapp, colunaChave, colunaBonita, valor) {

  var sheet = getFuncionariosSheet_();

/**
 * Recebe uma foto (perfil ou corpo inteiro) em base64 -- já redimensionada
 * no navegador antes de enviar, pra não pesar -- salva como arquivo no
 * Drive e grava o link na planilha.
 */
function enviarFotoPerfil_(data) {
  var whatsapp = normalizarWhats_(data.whatsapp);
  var vu = verificarUsuario_(whatsapp, data.senha);
  if (!vu.ok) {
    return jsonOut_({ ok: false, erro: vu.erro || 'WhatsApp ou senha incorretos.' });
  }
  var tipo = data.tipo === 'corpo' ? 'corpo' : 'perfil';
  var base64 = String(data.fotoBase64 || '');
  var m = base64.match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,([\s\S]+)$/);
  if (!m) return jsonOut_({ ok: false, erro: 'Formato de imagem inválido.' });
  var bytes;
  try {
    bytes = Utilities.base64Decode(m[2]);
  } catch (err) {
    return jsonOut_({ ok: false, erro: 'Não foi possível ler a imagem enviada.' });
  }
  if (bytes.length > 6 * 1024 * 1024) {
    return jsonOut_({ ok: false, erro: 'Imagem grande demais (máximo 6MB).' });
  }
  var blob = Utilities.newBlob(bytes, m[1], whatsapp + '-' + tipo);
  var pasta = getPastaFotosPerfil_();
  var arquivo = pasta.createFile(blob);
  arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var url = 'https://lh3.googleusercontent.com/d/' + arquivo.getId();
  if (tipo === 'corpo') {
    gravarCampoFuncionario_(whatsapp, 'fotocorpourl', 'FotoCorpoURL', url);
  } else {
    gravarCampoFuncionario_(whatsapp, 'fotoperfilurl', 'FotoPerfilURL', url);
  }
  return jsonOut_({ ok: true, url: url });
}

/**
 * Lê um Google Doc inteiro e transforma em {parts:[{title, chapters:[{title,
 * blocks}]}]} -- o mesmo formato que o site já usava. Convenções:
 *  - Título 1 (Heading 1)  -> nova PARTE do manual
 *  - Título 2 ou 3         -> novo CAPÍTULO dentro da parte atual
 *  - Subtítulo (Subtitle)  -> vira uma citação em destaque
 *  - Texto começando com "> "  -> também vira citação (alternativa ao Subtítulo)
 *  - Lista com marcadores   -> vira tópicos (mantém o nível de indentação)
 *  - Texto todo em negrito  -> vira parágrafo em destaque
 *  - Imagem dentro do texto -> extraída pro Drive e inserida como bloco de imagem,
 *    já no tamanho (largura/altura) em que foi deixada no Doc
 *  - TUDO que vem antes do primeiro Título 1 (título do documento, subtítulo,
 *    sumário...) é ignorado: é só apresentação do Doc, não vira parte nem
 *    capítulo no site. (Se o Doc não tiver nenhum Título 1, nada é ignorado.)
 */
function analisarDocManual_(docId) {
  var doc = DocumentApp.openById(docId);
  var body = doc.getBody();
  var pastaImagens = null; // só cria/abre a pasta se realmente aparecer uma imagem
  var parts = [];
  var parteAtual = null;
  var capituloAtual = null;

  function garantirParte() {
    if (!parteAtual) {
      parteAtual = { title: 'Geral', chapters: [] };
      parts.push(parteAtual);
    }
    return parteAtual;
  }
  function garantirCapitulo() {
    garantirParte();
    if (!capituloAtual) {
      capituloAtual = { title: 'Introdução', blocks: [] };
      parteAtual.chapters.push(capituloAtual);
    }
    return capituloAtual;
  }

  var total = body.getNumChildren();

  // Tudo antes do primeiro Título 1 (título, subtítulo, sumário) é só
  // apresentação do Doc: não vira parte nem capítulo no site.
  var inicio = 0;
  for (var k = 0; k < total; k++) {
    var elK = body.getChild(k);
    if (elK.getType() === DocumentApp.ElementType.PARAGRAPH &&
        elK.asParagraph().getHeading() === DocumentApp.ParagraphHeading.HEADING1) {
      inicio = k;
      break;
    }
  }

  for (var i = inicio; i < total; i++) {
    var el = body.getChild(i);
    var tipo = el.getType();

    if (tipo === DocumentApp.ElementType.PARAGRAPH) {
      var par = el.asParagraph();
      var heading = par.getHeading();
      var texto = par.getText();

      // imagens inline dentro deste parágrafo
      var imagensDoParagrafo = [];
      var nFilhos = par.getNumChildren();
      for (var j = 0; j < nFilhos; j++) {
        var filho = par.getChild(j);
        if (filho.getType() === DocumentApp.ElementType.INLINE_IMAGE) {
          if (!pastaImagens) pastaImagens = getPastaImagens_();
          imagensDoParagrafo.push(extrairImagemDrive_(filho.asInlineImage(), pastaImagens));
        }
      }

      if (heading === DocumentApp.ParagraphHeading.HEADING1) {
        parteAtual = { title: texto || 'Parte', chapters: [] };
        parts.push(parteAtual);
        capituloAtual = null;
        continue;
      }
      if (heading === DocumentApp.ParagraphHeading.HEADING2 || heading === DocumentApp.ParagraphHeading.HEADING3) {
        garantirParte();
        capituloAtual = { title: texto || 'Capítulo', blocks: [] };
        parteAtual.chapters.push(capituloAtual);
        continue;
      }
      if (heading === DocumentApp.ParagraphHeading.TITLE) {
        continue; // título do documento -- o título do manual já vem da configuração no site
      }

      if (texto.trim() === '' && !imagensDoParagrafo.length) continue; // parágrafo realmente vazio -- nem cria capítulo "Geral" à toa

      var cap = garantirCapitulo();
      imagensDoParagrafo.forEach(function (img) { cap.blocks.push(img); });

      if (texto.trim() === '') continue; // só tinha imagem, já foi adicionada acima

      if (heading === DocumentApp.ParagraphHeading.SUBTITLE) {
        cap.blocks.push({ type: 'quote', text: texto });
        continue;
      }
      var mQuote = texto.match(/^>\s?(.*)$/);
      if (mQuote) {
        cap.blocks.push({ type: 'quote', text: mQuote[1] });
        continue;
      }
      // isBold() sem argumentos: true = parágrafo todo em negrito, false = nada,
      // null = misturado (nesse caso trata como não-negrito, pra não travar a sincronização).
      var negrito = par.editAsText().isBold();
      cap.blocks.push({ type: 'p', text: texto, bold: negrito === true });

    } else if (tipo === DocumentApp.ElementType.LIST_ITEM) {
      var item = el.asListItem();
      var txt2 = item.getText();
      if (txt2.trim() === '') continue;
      garantirCapitulo().blocks.push({ type: 'bul', level: item.getNestingLevel() || 0, text: txt2 });

    } else if (tipo === DocumentApp.ElementType.INLINE_IMAGE) {
      // imagem solta fora de um parágrafo normal (raro, mas cobre o caso)
      if (!pastaImagens) pastaImagens = getPastaImagens_();
      garantirCapitulo().blocks.push(extrairImagemDrive_(el.asInlineImage(), pastaImagens));
    }
  }

  return parts;
}

/**
 * Escreve um conteúdo {parts:[...]} dentro de um Google Doc recém-criado --
 * usado só na migração dos 3 manuais antigos, pra já nascerem preenchidos
 * com o texto que já existia no site.
 */
function escreverDocDeParts_(doc, parts) {
  var body = doc.getBody();
  body.clear();
  (parts || []).forEach(function (part) {
    body.appendParagraph(part.title || 'Parte').setHeading(DocumentApp.ParagraphHeading.HEADING1);
    (part.chapters || []).forEach(function (ch) {
      body.appendParagraph(ch.title || 'Capítulo').setHeading(DocumentApp.ParagraphHeading.HEADING2);
      (ch.blocks || []).forEach(function (b) {
        if (b.type === 'bul') {
          var li = body.appendListItem(b.text || '');
          li.setNestingLevel(b.level || 0);
          li.setGlyphType(DocumentApp.GlyphType.BULLET);
        } else if (b.type === 'quote') {
          body.appendParagraph(b.text || '').setHeading(DocumentApp.ParagraphHeading.SUBTITLE);
        } else if (b.type === 'img') {
          // imagens antigas (se algum dia existiram em base64) não são recriadas
          // automaticamente no Doc -- a partir de agora, imagem se adiciona direto no Doc.
        } else {
          var p = body.appendParagraph(b.text || '');
          p.setHeading(DocumentApp.ParagraphHeading.NORMAL);
          if (b.bold) p.editAsText().setBold(true);
        }
      });
    });
  });
}

/**
 * Cria o Google Doc (vazio ou já preenchido) e registra a linha na aba
 * "Manuais". Devolve o id gerado, o id do Doc e o link de edição.
 */
function criarManualRegistro_(manualId, titulo, blurb, nivelMinimo, parts) {
  var doc = DocumentApp.create(titulo || 'Novo manual');
  var docId = doc.getId();
  if (parts && parts.length) {
    escreverDocDeParts_(doc, parts);
  } else {
    doc.getBody().appendParagraph('Capítulo 1').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    doc.getBody().appendParagraph('Escreva aqui o conteúdo do capítulo...');
  }
  doc.saveAndClose();

  var sheet = getManuaisSheet_();
  var dados = sheet.getDataRange().getValues();
  var maxOrdem = 0;
  for (var i = 1; i < dados.length; i++) { maxOrdem = Math.max(maxOrdem, Number(dados[i][5]) || 0); }

  sheet.appendRow([
    manualId, titulo || '', blurb || '', parseInt(nivelMinimo, 10) || 1,
    docId, maxOrdem + 1, JSON.stringify(parts || []), new Date()
  ]);
  return { manualId: manualId, docId: docId, docUrl: 'https://docs.google.com/document/d/' + docId + '/edit' };
}

/**
 * Copia as planilhas de Funcionarios e Resultados para a pasta de backups
 * do Drive e mantém só as 8 cópias mais recentes de cada. Para automatizar:
 * Gatilhos (relógio, menu à esquerda) -> Adicionar gatilho -> fazerBackupAgora
 * -> Baseado em tempo -> Contador semanal.
 */
function fazerBackupAgora() {
  var pasta = getPastaBackups_();
  var carimbo = Utilities.formatDate(new Date(), 'America/Fortaleza', 'yyyy-MM-dd HH-mm');
  var alvos = [
    { nome: 'Funcionarios+Manuais', id: FUNCIONARIOS_SHEET_ID },
    { nome: 'Resultados', id: RESULTADOS_SHEET_ID }
  ];
  alvos.forEach(function (a) {
    DriveApp.getFileById(a.id).makeCopy('Backup ' + a.nome + ' ' + carimbo, pasta);
    var copias = [];
    var it = pasta.getFiles();
    while (it.hasNext()) {
      var f = it.next();
      if (f.getName().indexOf('Backup ' + a.nome + ' ') === 0) copias.push(f);
    }
    copias.sort(function (x, y) { return y.getDateCreated() - x.getDateCreated(); });
    copias.slice(8).forEach(function (f) { f.setTrashed(true); });
  });
  Logger.log('Backup feito em ' + carimbo + '.');
}

/** Rode 1x (Executar): corrige números antigos sem o "9" (10 dígitos) nas planilhas de Pessoas e de
 * Currículos, deixando todo mundo em ddd+9+nnnnnnnn (11 dígitos). Se dois números viram o mesmo depois
 * de corrigido (duplicata), AVISA no log e não mexe -- você decide à mão o que fazer com essas. */
function normalizarWhatsExistentes_() {
  function corrigirAba(sh, colunaNome) {
    var headers = acCabecalhos_(sh);
    var iCol = headers.indexOf(colunaNome.toLowerCase());
    if (iCol === -1) { Logger.log('Coluna "' + colunaNome + '" não encontrada em ' + sh.getName() + '.'); return; }
    var ultima = sh.getLastRow();
    if (ultima < 2) return;
    var valores = sh.getRange(2, iCol + 1, ultima - 1, 1).getValues();
    var vistos = {}; valores.forEach(function (v) { var d = String(v[0] || '').replace(/\D/g, ''); if (d) vistos[d] = (vistos[d] || 0) + 1; });
    var feitos = 0, duplicadas = [];
    for (var i = 0; i < valores.length; i++) {
      var atual = String(valores[i][0] || '');
      var novo = normalizarWhats_(atual);
      if (!novo || novo === atual.replace(/\D/g, '')) continue;
      if (vistos[novo]) { duplicadas.push(atual + ' -> ' + novo); continue; } // já existe alguém com o número corrigido: não mexe, avisa
      sh.getRange(i + 2, iCol + 1).setNumberFormat('@').setValue(novo);
      feitos++;
    }
    Logger.log(sh.getName() + ' (' + colunaNome + '): corrigidos ' + feitos + '.' + (duplicadas.length ? ' Duplicatas (não mexidas, resolva à mão): ' + duplicadas.join(', ') : ''));
  }
  corrigirAba(acPlanilha_(), 'Whatsapp');
  corrigirAba(cvAbaNova_(), 'WhatsApp');
}

/** Preenche caixinhas/perfil em branco com o que a pessoa já tinha (perfil ou Nivel antigo). Devolve quantas linhas mexeu. */
function reconciliarLinhas_(maximo) {
  var sh = acGarantirColunas_();
  var headers = acCabecalhos_(sh);
  var perfis = acPerfis_();
  var mexeu = 0;
  var linhas = acTodasLinhas_();
  for (var i = 0; i < linhas.length && mexeu < (maximo || 200); i++) {

    var precisa = !String(l.perfil || '').trim();
    if (!precisa) acChaves_().forEach(function (k) { if (acBool_(l['perm_' + k]) === null) precisa = true; });
    if (!precisa) continue;
    var nomePerfil = String(l.perfil || '').trim() || acPerfilLegado_(l.nivel);

/** Normaliza uma data de cadastro (Date, ISO com hora, ou dd/mm/aaaa) para "aaaa-mm-dd". */
function acCadastroISO_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v)) {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  var s = String(v || '').trim();
  var m = /^(\d{4})-(\d{2})-(\d{2})/.exec(s);
  if (m) return m[1] + '-' + m[2] + '-' + m[3];
  m = /^(\d{1,2})\/(\d{1,2})\/(\d{4})$/.exec(s);
  if (m) return m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2);
  return '';
}

function cvNorm_(s) {
  return String(s == null ? '' : s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').trim();
}
function cvHash_(s) {
  s = String(s); var h = 5381;
  for (var i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
  return h >>> 0;
}
function cvRng_(seed) {
  var a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) | 0;
    var t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function cvEmbaralhar_(arr, rng) {
  var a = arr.slice();
  for (var i = a.length - 1; i > 0; i--) { var j = Math.floor(rng() * (i + 1)); var t = a[i]; a[i] = a[j]; a[j] = t; }
  return a;
}

}

/** Idade em anos completos (nasc = 'aaaa-mm-dd'). Devolve null se a data não existe ou é absurda. */
function cvIdade_(nasc, hoje) {
  var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(nasc || ''));
  if (!m) return null;
  var a = +m[1], mes = +m[2], d = +m[3];
  var dt = new Date(Date.UTC(a, mes - 1, d));
  if (dt.getUTCFullYear() !== a || dt.getUTCMonth() !== mes - 1 || dt.getUTCDate() !== d) return null;
  hoje = hoje || new Date();
  var idade = hoje.getFullYear() - a;
  var fez = (hoje.getMonth() + 1 > mes) || (hoje.getMonth() + 1 === mes && hoje.getDate() >= d);
  if (!fez) idade--;
  if (idade < 0 || idade > 100) return null;
  return idade;
}

/** Interpreta a matriz da aba antiga (linha 1 = cabeçalho). Lê Telefone, Telefone 2 e Telefone 3. */
function cvLegadoParse_(v) {
  if (!v || v.length < 2) return [];
  var h = v[0].map(cvNorm_);
  function idx(t, sem) { for (var i = 0; i < h.length; i++) if (h[i].indexOf(t) !== -1 && (!sem || h[i].indexOf(sem) === -1)) return i; return -1; }
  function idxExato(t) { for (var i = 0; i < h.length; i++) if (h[i] === t) return i; return -1; }
  var I = {
    ts: idx('carimbo'), email: idx('e-mail'), insta: idx('instagram'), nasc: idx('data de nascimento'), moradia: idx('priorizamos'),
    idade: idx('idade atual'), salario: idx('salario'), sobre: idx('me fala sobre'), util: idx('seria util'), exp: idx('experiencias trabalhistas'),
    estudos: idx('seus estudos'), disp: idx('disponibilidade'), nome: idx('nome completo'),
    tel: idxExato('telefone') !== -1 ? idxExato('telefone') : idx('telefone'),
    tel2: idxExato('telefone 2'), tel3: idxExato('telefone 3'),
    chamar: idx('chamar?'), encarregada: idx('sub encarregada'), obs: idx('observac')
  };
  var out = [];
  for (var i = 1; i < v.length; i++) {
    var r = v[i];
    if (r.join('').trim() === '') continue;
    var o = { linha: i + 1 };
    Object.keys(I).forEach(function (k) {
      var x = I[k] === -1 ? '' : r[I[k]];
      o[k] = (k === 'nasc') ? x : (Object.prototype.toString.call(x) === '[object Date]' ? x.toISOString() : String(x == null ? '' : x).trim());
    });
    o.email = cvEmail_(o.email);
    out.push(o);
  }
  return out;
}

/** Até 4 perguntas de múltipla escolha (4 alternativas) sobre o que a pessoa preencheu antes.

function cvGerarPerguntas_(reg, todos, email, hoje) {

  c.Ritmo = d.ritmo;
  if (d.ritmo === 'Sim') { c.Horario = CV_OP.horario[2]; c.ForaHorario = CV_OP.fora[1]; }
  else {
    if (!cvEm_(d.horario, CV_OP.horario)) return erro('Escolha seu horário disponível.');
    if (!cvEm_(d.fora, CV_OP.fora)) return erro('Diga se você trabalha fora do horário comercial.');
    c.Horario = d.horario; c.ForaHorario = d.fora;
  }
  c.Impedimento = cvTxt_(d.impedimento, 300);
  if (!cvEm_(d.jaTrabalhou, CV_OP.simnao)) return erro('Responda se você já trabalhou.');
  c.JaTrabalhou = d.jaTrabalhou;
  var vazios = ['UltEmpresa', 'UltFuncao', 'UltMeses', 'UltMotivo', 'TiposEmpresa', 'Tarefas', 'Liderou', 'LiderQuantos', 'Referencia', 'RefContato'];
  vazios.forEach(function (k) { c[k] = ''; });
  if (d.jaTrabalhou === 'Sim') {
    c.UltEmpresa = cvTxt_(d.ultEmpresa, 120); c.UltFuncao = cvTxt_(d.ultFuncao, 120); c.UltMotivo = cvTxt_(d.ultMotivo, 200);
    var meses = parseInt(d.ultMeses, 10);
    if (!c.UltEmpresa || !c.UltFuncao || !c.UltMotivo || isNaN(meses) || meses < 0 || meses > 600) return erro('Preencha empresa, função, tempo e motivo da saída do último trabalho.');
    c.UltMeses = String(meses);
    if (!cvTodosEm_(d.tiposEmpresa, CV_OP.tipos)) return erro('Escolha os tipos de empresa em que você já trabalhou.');
    c.TiposEmpresa = d.tiposEmpresa.join(' | ');
    if (!cvTodosEm_(d.tarefas, CV_OP.tarefas)) return erro('Escolha as tarefas que você já fez.');
    c.Tarefas = d.tarefas.join(' | ');
    if (!cvEm_(d.liderou, CV_OP.simnao)) return erro('Responda se você já liderou uma equipe.');
    c.Liderou = d.liderou;
    if (d.liderou === 'Sim') { c.LiderQuantos = cvTxt_(d.liderQuantos, 150); if (!c.LiderQuantos) return erro('Diga quantas pessoas e por quanto tempo.'); }
    if (!cvEm_(d.referencia, CV_OP.referencia)) return erro('Responda sobre a referência.');
    c.Referencia = d.referencia;
    if (d.referencia === CV_OP.referencia[0]) { c.RefContato = cvTxt_(d.refContato, 150); if (!c.RefContato) return erro('Informe nome e telefone da referência.'); }
  }
  if (!cvEm_(d.escolaridade, CV_OP.escolaridade)) return erro('Escolha sua escolaridade.');
  c.Escolaridade = d.escolaridade;
  c.Cursos = cvTxt_(d.cursos, 300);
  [['troca', 'Q_Troca'], ['troco', 'Q_Troco'], ['meta', 'Q_Meta'], ['tres', 'Q_TresClientes']].forEach(function (p) { c[p[1]] = d[p[0]]; });
  if (!cvEm_(d.troca, CV_OP.troca) || !cvEm_(d.troco, CV_OP.troco) || !cvEm_(d.meta, CV_OP.meta) || !cvEm_(d.tres, CV_OP.tres)) return erro('Responda as perguntas de situação.');
  if (!cvTodosEm_(d.motiva, CV_OP.motiva) || d.motiva.length > 2) return erro('Escolha até 2 coisas que mais te motivam.');
  c.Motiva = d.motiva.join(' | ');
  c.Situacao = cvTxtLongo_(d.situacao, 1200);
  if (c.Situacao.length < 10) return erro('Conte uma situação em que você resolveu um problema.');
  c.PorQue = cvTxtLongo_(d.porQue, 800);
  if (c.PorQue.length < 5) return erro('Conte por que você quer trabalhar na TENKiTER.');
  if (!cvEm_(d.origem, CV_OP.origem)) return erro('Diga como conheceu a vaga.');
  c.Origem = d.origem;
  c.ConheceAlguem = cvTxt_(d.conheceAlguem, 200);
  c.Obs = cvTxtLongo_(d.obs, 600);
  return { ok: true, campos: c };
}

function cvSS_() { return abrirSS_(CURRICULOS_SHEET_ID); }

function cvAbaNova_() {
  var ss = cvSS_();
  var aba = ss.getSheetByName(CV_ABA_NOVA);
  if (!aba) {
    aba = ss.insertSheet(CV_ABA_NOVA);
    aba.getRange(1, 1, 1, CV_HEADERS.length).setValues([CV_HEADERS]);
    aba.setFrozenRows(1);
    aba.getRange(1, 1, Math.max(aba.getMaxRows(), 1000), CV_HEADERS.length).setNumberFormat('@'); // tudo texto: não vira data/fórmula
    return aba;
  }
  var atuais = aba.getRange(1, 1, 1, Math.max(aba.getLastColumn(), 1)).getValues()[0].map(String);
  var faltam = CV_HEADERS.filter(function (h) { return atuais.indexOf(h) === -1; });
  if (faltam.length) aba.getRange(1, atuais.length + 1, 1, faltam.length).setValues([faltam]);
  return aba;
}
function cvLegadoAba_() {
  var ss = cvSS_();
  return ss.getSheetByName(CV_ABA_LEGADO) || ss.getSheets()[0];
}
function cvLegadoRegs_() { return cvLegadoParse_(cvLegadoAba_().getDataRange().getValues()); }

function cvLerNovos_() {
  var aba = cvAbaNova_();
  var dados = aba.getDataRange().getValues();
  var headers = dados[0].map(String);
  var rows = [];
  for (var i = 1; i < dados.length; i++) {
    if (dados[i].join('').trim() === '') continue;
    var o = { __row: i + 1 };
    headers.forEach(function (h, j) { o[h] = dados[i][j] == null ? '' : String(dados[i][j]); });
    rows.push(o);
  }
  return { aba: aba, headers: headers, rows: rows };
}
function cvAcharNovo_(rows, campo, valor) {
  var ach = null;
  for (var i = 0; i < rows.length; i++) if (rows[i][campo] === valor) ach = rows[i];
  return ach;
}
function cvGravarLinha_(aba, headers, numLinha, campos) {
  var atual = numLinha ? aba.getRange(numLinha, 1, 1, headers.length).getValues()[0] : headers.map(function () { return ''; });
  var novo = headers.map(function (h, i) { return campos[h] !== undefined ? campos[h] : (atual[i] == null ? '' : atual[i]); });
  if (numLinha) aba.getRange(numLinha, 1, 1, headers.length).setValues([novo]);
  else {
    var alvo = aba.getLastRow() + 1;
    aba.getRange(alvo, 1, 1, headers.length).setNumberFormat('@');
    aba.getRange(alvo, 1, 1, headers.length).setValues([novo]);
  }
}
function cvComTrava_(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { return fn(); } finally { lock.releaseLock(); }
}
function cvAgora_() { return new Date().toISOString(); }

function cvIniciarAcao_(data) {
  var whats = normalizarWhats_(data.whatsapp);
  if (whats.length < 10 || whats.length > 11) return jsonOut_({ ok: false, erro: 'Digite o WhatsApp com DDD.' });
  // Já tem conta de verdade no site (funcionário, aluno etc.)? Então o lugar certo pra entrar é o
  // portal (manual.html) com a senha de sempre -- lá ela já vê se falta currículo e manda terminar.
  // Não faz sentido pedir senha de novo aqui, numa telinha separada só pro currículo.
  if (acLinhasDoWhats_(whats).length) return jsonOut_({ ok: true, modo: 'conta' });
  var cache = CacheService.getScriptCache();
  var feitos = parseInt(cache.get('cv_ini_hora') || '0', 10);
  if (feitos >= 600) return jsonOut_({ ok: false, erro: 'Muitos acessos agora. Tente de novo em alguns minutos.' });
  cache.put('cv_ini_hora', String(feitos + 1), 3600);

  var novos = cvLerNovos_().rows;
  if (cvAcharNovo_(novos, 'WhatsApp', whats)) return jsonOut_({ ok: true, modo: 'retomar' });

  var regs = cvLegadoRegs_();
  var reg = cvLegadoPorWhats_(regs, whats);
  if (!reg) return jsonOut_({ ok: true, modo: 'novo' });
  var tent = parseInt(cache.get('cvv_' + cvHash_(whats)) || '0', 10);
  if (tent >= CV_MAX_VERIFICACOES) return jsonOut_({ ok: true, modo: 'novo', esgotado: true });
  var perguntas = cvGerarPerguntasWhats_(reg, regs, whats);
  if (perguntas.length < 3) return jsonOut_({ ok: true, modo: 'novo' }); // dados antigos insuficientes: segue como candidata nova
  var id = Utilities.getUuid();
  var corretas = {};
  perguntas.forEach(function (p) { corretas[p.id] = p.correta; });
  cache.put('cvd_' + id, JSON.stringify({ whatsapp: whats, linha: reg.linha, corretas: corretas }), 900);
  return jsonOut_({ ok: true, modo: 'verificar', desafioId: id, perguntas: perguntas.map(function (p) { return { id: p.id, texto: p.texto, opcoes: p.opcoes }; }) });
}

function cvVerificarAcao_(data) {
  var cache = CacheService.getScriptCache();
  var bruto = cache.get('cvd_' + String(data.desafioId || ''));
  if (!bruto) return jsonOut_({ ok: false, erro: 'A verificação expirou. Digite o WhatsApp de novo.' });
  var d = JSON.parse(bruto);
  var chave = 'cvv_' + cvHash_(d.whatsapp);
  var n = parseInt(cache.get(chave) || '0', 10);
  if (n >= CV_MAX_VERIFICACOES) return jsonOut_({ ok: true, verificado: false, esgotado: true });
  cache.put(chave, String(n + 1), 21600);
  var resp = data.respostas || {};
  var ids = Object.keys(resp);
  var certo = ids.length === 3 && ids.every(function (k) { return d.corretas[k] !== undefined && String(resp[k]) === d.corretas[k]; });
  if (!certo) {
    var rest = CV_MAX_VERIFICACOES - (n + 1);
    return jsonOut_({ ok: true, verificado: false, restantes: rest, esgotado: rest <= 0 });
  }
  cache.remove('cvd_' + data.desafioId);
  var reg = cvLegadoRegs_().filter(function (r) { return r.linha === d.linha; })[0];
  if (!reg || !cvRegBateWhats_(reg, cvW8_(d.whatsapp))) return jsonOut_({ ok: false, erro: 'Não foi possível carregar seus dados. Preencha como candidata nova.' });
  var token = Utilities.getUuid();
  cache.put('cvt_' + token, JSON.stringify({ whatsapp: d.whatsapp, linha: reg.linha }), 7200);
  var ig = cvInstaNorm_(reg.insta);
  return jsonOut_({
    ok: true, verificado: true, cvToken: token,
    prefill: {
      nome: cvTxt_(reg.nome, 120), email: cvEmail_(reg.email), whatsapp: d.whatsapp, instagram: ig ? '@' + ig : '',
      moradia: cvMoradiaLegado_(reg.moradia), expAnterior: cvTxtLongo_(reg.exp, 1200), estudosAnterior: cvTxtLongo_(reg.estudos, 600)
    }
  });
}

function cvCadastroAcao_(data) {
  var email = cvEmail_(data.email);
  var nome = cvTxt_(data.nome, 120);
  var whats = normalizarWhats_(data.whatsapp);
  var senha = String(data.sessao || data.senha || ''); // logada no site: vale a sessão; senão, a senha digitada
  if (!cvEmailValido_(email)) return jsonOut_({ ok: false, erro: 'Digite um e-mail válido.' });
  if (nome.split(' ').length < 2) return jsonOut_({ ok: false, erro: 'Digite seu nome completo.' });
  if (whats.length < 10 || whats.length > 11) return jsonOut_({ ok: false, erro: 'Digite o WhatsApp com DDD.' });
  if (senha.length < 4) return jsonOut_({ ok: false, erro: 'A senha precisa ter pelo menos 4 caracteres.' });
  if (!cvEm_(data.moradia, CV_OP.moradia)) return jsonOut_({ ok: false, erro: 'Escolha onde você mora.' });
  if (!cvEm_(data.chegada, CV_OP.chegada)) return jsonOut_({ ok: false, erro: 'Diga como chegaria ao trabalho.' });
  if (data.lgpd !== true) return jsonOut_({ ok: false, erro: 'Sem a autorização não conseguimos guardar seus dados.' });
  var idade = cvIdade_(data.dataNascimento);
  if (idade === null) return jsonOut_({ ok: false, erro: 'Confira a data de nascimento.' });

  var cache = CacheService.getScriptCache();
  var feitos = parseInt(cache.get('reg_hora') || '0', 10);
  if (feitos >= 120) return jsonOut_({ ok: false, erro: 'Muitos cadastros agora. Tente de novo em alguns minutos.' });

  // conta (Aluno já inclui a conta de cliente). Se já existe, só entra com a MESMA senha.
  var conta = acVerificar_(whats, senha);
  if (!conta.ok && data.sessao) return jsonOut_({ ok: false, erro: 'Sua sessão expirou. Entre de novo no site e tente outra vez.' }); // nunca cria conta usando um token como senha
  if (!conta.ok) {
    if (conta.bloqueado) return jsonOut_({ ok: false, erro: conta.erro });
    if (acLinhasDoWhats_(whats).length) return jsonOut_({ ok: false, erro: 'Esse WhatsApp já tem cadastro no nosso site com outra senha. Use a mesma senha do site ou fale com a loja.' });
    cache.put('reg_hora', String(feitos + 1), 3600);
    var r = acRegistrar_(nome, whats, senha, 'Aluno', {});
    if (!r.ok) return jsonOut_({ ok: false, erro: r.erro });
    conta = acVerificar_(whats, senha);
    if (!conta.ok) return jsonOut_({ ok: false, erro: 'Cadastro criado, mas não foi possível entrar. Fale com a loja.' });
  } else if (acNorm_(conta.usuario.perfil) === 'cliente') {
    acessosDefinirPerfil_(whats, 'Aluno', { name: conta.usuario.name, whatsapp: conta.usuario.whatsapp });
  }
  gravarCampoFuncionario_(whats, 'datanascimento', 'DataNascimento', data.dataNascimento);

  var menor = idade < 18;
  var agora = cvAgora_();
  var legadoLinha = '', legadoChamar = '';
  if (data.cvToken) {
    var t = cache.get('cvt_' + String(data.cvToken));
    if (t) {
      var tk = JSON.parse(t);
      if (cvW8_(tk.whatsapp) === cvW8_(whats)) {
        legadoLinha = String(tk.linha);
        var rg = cvLegadoRegs_().filter(function (x) { return x.linha === tk.linha; })[0];
        if (rg) legadoChamar = cvTxtLongo_(rg.chamar, 500);
      }
    }
  }
  var resultado = cvComTrava_(function () {
    var L = cvLerNovos_();
    var ex = cvAcharNovo_(L.rows, 'WhatsApp', whats); // a chave agora é o WhatsApp; o e-mail é só um campo editável
    var campos = {
      Atualizado: agora, Etapa: menor ? 'Menor de 18' : 'Instagram', Email: email, Nome: nome, WhatsApp: whats,
      Moradia: data.moradia, Chegada: data.chegada, LGPD_Aceite: agora, LGPD_Texto: CV_LGPD_TEXTO,
      Nascimento: data.dataNascimento, Idade: String(idade), Menor18: menor ? 'SIM' : 'NÃO', Retomar18Em: menor ? cvDia18_(data.dataNascimento) : ''
    };
    if (!ex) { campos.CvId = Utilities.getUuid(); campos.Criado = agora; campos.Status = 'Novo'; campos.LegadoLinha = legadoLinha; campos.LegadoChamar = legadoChamar; }
    else if (ex.Etapa === 'Enviado' && !menor) campos.Etapa = 'Enviado';
    cvGravarLinha_(L.aba, L.headers, ex ? ex.__row : 0, campos);
    return { ok: true };
  });
  if (resultado.erro) return jsonOut_({ ok: false, erro: resultado.erro });
  return jsonOut_({ ok: true, menor: menor, nome: nome, whatsapp: whats, sessao: conta.sessao, fotoUrl: conta.usuario.fotoPerfilUrl || '', etapaAtual: menor ? 'Menor de 18' : 'Instagram' });
}

/** Data de nascimento da CONTA (aba de funcionários) em 'aaaa-mm-dd'; aceita 'aaaa-mm-dd', ISO com hora e 'dd/mm/aaaa'. '' se não reconhecer. */
function cvNascConta_(t) {
  t = String(t || '').trim();
  var m = /^(\d{4})-(\d{2})-(\d{2})/.exec(t);
  if (m) return m[1] + '-' + m[2] + '-' + m[3];
  m = /^(\d{1,2})\/(\d{1,2})\/(\d{4})$/.exec(t);
  if (m) return m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2);
  return '';
}

/**
 * A data de nascimento da conta (corrigida no cadastro de funcionários, por exemplo) manda.
 * Se ela difere da do currículo, atualiza a linha (idade, Menor18, Retomar18Em, etapa) e devolve a linha já corrigida.
 */
function cvSincronizarNasc_(L, row, usuario) {
  var n = cvNascConta_(usuario && usuario.dataNascimento);
  if (!n || cvIdade_(n) === null || n === row.Nascimento) return row;
  var idade = cvIdade_(n), menor = idade < 18;
  var campos = { Atualizado: cvAgora_(), Nascimento: n, Idade: String(idade), Menor18: menor ? 'SIM' : 'NÃO', Retomar18Em: menor ? cvDia18_(n) : '' };
  if (menor) { if (row.Etapa !== 'Enviado') campos.Etapa = 'Menor de 18'; }
  else if (row.Etapa === 'Menor de 18') campos.Etapa = 'Instagram';
  cvComTrava_(function () { cvGravarLinha_(L.aba, L.headers, row.__row, campos); });
  Object.keys(campos).forEach(function (k) { row[k] = campos[k]; });
  return row;
}

function cvLinhaDaConta_(data) {
  var v = acVerificar_(data.whatsapp, data.sessao);
  if (!v.ok) return { erro: v.erro || 'Sessão expirada. Entre de novo.' };
  var L = cvLerNovos_();
  var row = cvAcharNovo_(L.rows, 'WhatsApp', v.usuario.whatsapp);
  if (!row) return { erro: 'Currículo não encontrado. Comece pelo e-mail.' };
  row = cvSincronizarNasc_(L, row, v.usuario);
  return { L: L, row: row, usuario: v.usuario };
}

  var agora = cvAgora_();

    return jsonOut_({ ok: true, semInstagram: true });

  var ig = cvInstaNorm_(data.instagram);

  gravarCampoFuncionario_(x.usuario.whatsapp, 'instagram', 'Instagram', '@' + ig);
  return jsonOut_({ ok: true });
}

function cvRetomarAcao_(data) {
  var L = cvLerNovos_();

  if (data.sessao) { // já está logada no site: entra sem pedir senha
    v = acVerificar_(data.whatsapp, data.sessao);
    if (!v.ok) return jsonOut_({ ok: false, erro: v.erro || 'Sessão expirada.', sessaoInvalida: true });

  } else {
    var whatsRet = normalizarWhats_(data.whatsapp);
    row = cvAcharNovo_(L.rows, 'WhatsApp', whatsRet);

/** Situação do currículo de quem está logado no portal (para o aviso da tela inicial). */

  var row = cvAcharNovo_(L.rows, 'WhatsApp', v.usuario.whatsapp);
  if (!row) return jsonOut_({ ok: true, estado: 'nenhum', faltando: [] });
  row = cvSincronizarNasc_(L, row, v.usuario);
  var idade = cvIdade_(row.Nascimento);
  if (idade === null ? row.Menor18 === 'SIM' : idade < 18) return jsonOut_({ ok: true, estado: 'menor', faltando: [] });
  var faltando = [];
  if (!row.Instagram) faltando.push('Instagram');
  if (!v.usuario.fotoPerfilUrl) faltando.push('Foto de perfil');
  if (row.Etapa !== 'Enviado') faltando.push('Responder as perguntas e enviar o currículo');
  else { var val = cvValidarFinal_(cvRowParaDados_(row)); if (!val.ok) faltando.push(val.erro); }
  return jsonOut_({ ok: true, estado: faltando.length ? 'incompleto' : 'completo', faltando: faltando });
}

function cvFinalizarAcao_(data) {
  var x = cvLinhaDaConta_(data);
  if (x.erro) return jsonOut_({ ok: false, erro: x.erro });
  if (x.row.Menor18 === 'SIM') return jsonOut_({ ok: false, erro: 'Currículos só a partir dos 18 anos.' });
  if (!x.row.Instagram) return jsonOut_({ ok: false, erro: 'Falta o Instagram. Volte e informe seu @.' });
  if (!x.usuario.fotoPerfilUrl) return jsonOut_({ ok: false, erro: 'Falta a foto de perfil. Volte e envie sua foto.' });
  var val = cvValidarFinal_(data.dados);
  if (!val.ok) return jsonOut_({ ok: false, erro: val.erro });
  var agora = cvAgora_();
  val.campos.Atualizado = agora; val.campos.Etapa = 'Enviado'; val.campos.EnviadoEm = agora;
  cvComTrava_(function () { cvGravarLinha_(x.L.aba, x.L.headers, x.row.__row, val.campos); });
  return jsonOut_({ ok: true });
}

function cvAdminAtualizarAcao_(data) {
  var v = exigirPerm_(data, 'dados_pessoais');
  if (!v.ok) return negarAcesso_(v);
  var status = String(data.status || ''), motivo = String(data.motivo || '');
  if (!cvEm_(status, CV_STATUS)) return jsonOut_({ ok: false, erro: 'Status inválido.' });
  if (motivo && !cvEm_(motivo, CV_MOTIVOS)) return jsonOut_({ ok: false, erro: 'Motivo inválido.' });
  return cvComTrava_(function () {
    var L = cvLerNovos_();
    var row = cvAcharNovo_(L.rows, 'CvId', String(data.cvId || ''));
    if (!row) return jsonOut_({ ok: false, erro: 'Currículo não encontrado.' });
    cvGravarLinha_(L.aba, L.headers, row.__row, { Status: status, Motivo: motivo, Nota: cvTxtLongo_(data.nota, 2000) });
    return jsonOut_({ ok: true });
  });
}
/** Respostas ANTIGAS (planilha do formulário). Fora de propósito: estado civil, filhos, família, doenças e foto. */
function cvAdminLegadoListarAcao_(data) {
  var v = exigirPerm_(data, 'dados_pessoais');
  if (!v.ok) return negarAcesso_(v);
  var atualizaramEmail = {}, atualizaramTel = {};
  cvLerNovos_().rows.forEach(function (r) {
    if (r.Etapa !== 'Enviado') return;
    if (r.Email) atualizaramEmail[r.Email] = true;
    var w8 = cvW8_(r.WhatsApp); if (w8) atualizaramTel[w8] = true;
  });
  var regs = cvLegadoRegs_().map(function (r) {
    var ano = cvAnoNasc_(r.nasc);
    var bateuTel = [r.tel, r.tel2, r.tel3].some(function (t) { var w8 = cvW8_(t); return w8 && atualizaramTel[w8]; });
    return {
      linha: r.linha, ts: r.ts, nome: r.nome, tel: r.tel, insta: r.insta, email: r.email, idade: r.idade || (ano ? String(new Date().getFullYear() - ano) : ''),
      moradia: r.moradia, salario: r.salario, encarregada: r.encarregada, exp: r.exp, estudos: r.estudos, disp: r.disp, sobre: r.sobre, util: r.util,
      obs: r.obs, chamar: r.chamar, atualizou: !!(atualizaramEmail[r.email] || bateuTel)
    };
  });
  return jsonOut_({ ok: true, antigos: regs });
}
/** Grava a sua anotação na coluna "chamar?" da planilha antiga (o mesmo lugar onde você já anotava). */
function cvAdminLegadoNotaAcao_(data) {
  var v = exigirPerm_(data, 'dados_pessoais');
  if (!v.ok) return negarAcesso_(v);
  var linha = parseInt(data.linha, 10);
  return cvComTrava_(function () {
    var aba = cvLegadoAba_();
    if (isNaN(linha) || linha < 2 || linha > aba.getLastRow()) return jsonOut_({ ok: false, erro: 'Linha inválida.' });
    var h = aba.getRange(1, 1, 1, aba.getLastColumn()).getValues()[0].map(cvNorm_);
    var col = h.indexOf('chamar?') + 1;
    if (!col) return jsonOut_({ ok: false, erro: 'Coluna "chamar?" não encontrada.' });
    aba.getRange(linha, col).setValue(cvTxtLongo_(data.texto, 1000));
    return jsonOut_({ ok: true });
  });
}

/** LGPD: apaga currículos novos parados há mais de CV_LGPD_ANOS anos. Agende (Acionadores) 1x por mês. */
function curriculosLimpezaLgpd() {
  var L = cvLerNovos_();
  var limite = new Date(); limite.setFullYear(limite.getFullYear() - CV_LGPD_ANOS);
  var apagar = L.rows.filter(function (r) { var d = new Date(r.Atualizado); return !isNaN(d) && d < limite; }).map(function (r) { return r.__row; }).sort(function (a, b) { return b - a; });
  apagar.forEach(function (n) { L.aba.deleteRow(n); });
  Logger.log('Apagados: ' + apagar.length);
}

/** LGPD: faz uma CÓPIA da planilha antiga e depois esvazia as colunas sensíveis (foto, doenças, filhos, estado civil, família). Rode só quando quiser. */
function limparColunasSensiveisPlanilhaAntiga() {
  DriveApp.getFileById(CURRICULOS_SHEET_ID).makeCopy('BACKUP antes de limpar dados sensíveis - ' + new Date().toISOString().slice(0, 10));
  var aba = cvLegadoAba_();
  var h = aba.getRange(1, 1, 1, aba.getLastColumn()).getValues()[0].map(cvNorm_);
  var alvos = ['composicao familiar', 'filhos', 'estado civil', 'doencas', 'foto'];
  var ult = aba.getLastRow();
  var feitos = [];
  h.forEach(function (t, i) {
    if (alvos.some(function (a) { return t.indexOf(a) === 0; }) && ult > 1) { aba.getRange(2, i + 1, ult - 1, 1).clearContent(); feitos.push(aba.getRange(1, i + 1).getValue()); }
  });
  Logger.log('Colunas esvaziadas: ' + feitos.join(' | '));
}
