function acNorm_(s) {
  return String(s == null ? '' : s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').trim();
}
function acWhats_(w) {
  var d = String(w || '').replace(/\D/g, '');
  if (d.length >= 12 && d.slice(0, 2) === '55') d = d.slice(2);
  return d;
}
/** Texto vindo de fora nunca vira fórmula na planilha: se começa com = + - @ ganha um apóstrofo (o Sheets guarda como texto puro). */
function acTexto_(v) {
  var t = String(v == null ? '' : v);
  return /^[=+\-@\t\r]/.test(t) ? "'" + t : t;
}
function acBool_(v) {
  if (v === true || v === 1) return true;
  if (v === false || v === 0) return false;
  var s = String(v == null ? '' : v).trim().toLowerCase();
  if (s === 'true' || s === 'sim' || s === '1') return true;
  if (s === 'false' || s === 'nao' || s === 'não' || s === '0') return false;
  return null; // em branco / indefinido -> herda
}

function acSS_() { return SpreadsheetApp.openById(PESSOAS_PLANILHA_ID); }

function acAba_(nome, cabecalho, esconder) {
  var ss = acSS_();
  var aba = ss.getSheetByName(nome);
  if (!aba) {
    aba = ss.insertSheet(nome);
    if (cabecalho && cabecalho.length) aba.appendRow(cabecalho);
    if (esconder && aba.hideSheet) aba.hideSheet();
  }
  return aba;
}
function acConfigGet_(chave) {
  var aba = acAba_('_Config', ['Chave', 'Valor'], true);
  var dados = aba.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) if (String(dados[i][0]) === chave) return String(dados[i][1] == null ? '' : dados[i][1]);
  return '';
}
function acConfigSet_(chave, valor) {
  var aba = acAba_('_Config', ['Chave', 'Valor'], true);
  var dados = aba.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) {
    if (String(dados[i][0]) === chave) { aba.getRange(i + 1, 2).setValue(valor); return; }
  }
  aba.appendRow([chave, valor]);
}

/* ---------- senha, hash e token ---------- */
function acBytesHex_(bytes) {
  return bytes.map(function (b) { var v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}
function acSha256Hex_(texto) {
  return acBytesHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, texto, Utilities.Charset.UTF_8));
}
function acHashSenha_(senha) {
  var sal = Utilities.getUuid().replace(/-/g, '').slice(0, 16);
  return 'h1$' + sal + '$' + acSha256Hex_(sal + ':' + senha);
}
function acGuardarSenha_(senha) { return acSenhasEmHash_() ? acHashSenha_(senha) : String(senha); }
function acIguais_(a, b) {
  a = String(a); b = String(b);
  if (a.length !== b.length) return false;
  var r = 0;
  for (var i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}
function acSenhaSimples_(guardada, informado) {
  guardada = String(guardada == null ? '' : guardada);
  informado = String(informado == null ? '' : informado);
  if (!informado) return false;
  if (guardada.indexOf('h1$') === 0) {
    var p = guardada.split('$');
    if (p.length !== 3) return false;
    return acIguais_(acSha256Hex_(p[1] + ':' + informado), p[2]);
  }
  return acIguais_(guardada, informado);
}
function acHmac_(whats, guardada) {
  return acBytesHex_(Utilities.computeHmacSha256Signature(whats + '|' + String(guardada), acSegredo_()));
}
/** Token de sessão (autodescritivo): tk2.<whatsapp>.<assinatura>. Só vale enquanto a senha guardada não mudar. */
function acToken_(whats, guardada) { return 'tk2.' + whats + '.' + acHmac_(whats, guardada); }
function acEhToken_(cred) { var s = String(cred || ''); return s.indexOf('tk1.') === 0 || s.indexOf('tk2.') === 0; }
function acCredencialConfere_(linha, cred) {
  cred = String(cred == null ? '' : cred);
  var whats = acWhats_(linha.whatsapp);
  if (cred.indexOf('tk2.') === 0) {
    var p = cred.split('.');
    return p.length === 3 && p[1] === whats && acIguais_(acHmac_(whats, linha.senha), p[2]);
  }
  if (cred.indexOf('tk1.') === 0) return acIguais_('tk1.' + acHmac_(whats, linha.senha), cred);
  return acSenhaSimples_(linha.senha, cred);
}

/* ---------- leitura das pessoas ---------- */
function acCabecalhos_(sh) {
  var n = sh.getLastColumn();
  if (n < 1) return [];
  return sh.getRange(1, 1, 1, n).getValues()[0].map(function (h) { return String(h).trim().toLowerCase(); });
}
function acObj_(headers, valores, numLinha) {
  var o = { __row: numLinha };
  headers.forEach(function (h, i) { if (h) o[h] = valores[i]; });
  return o;
}
/** Só as linhas de UM WhatsApp (lê a coluna de WhatsApp inteira e depois só as linhas certas -- rápido mesmo com muitos clientes). */
function acLinhasDoWhats_(whatsapp) {
  var alvo = acWhats_(whatsapp);
  if (!alvo) return [];
  var sh = acPlanilha_();
  var ultima = sh.getLastRow();
  if (ultima < 2) return [];
  var headers = acCabecalhos_(sh);
  var iw = headers.indexOf('whatsapp');
  if (iw === -1) return [];
  var col = sh.getRange(2, iw + 1, ultima - 1, 1).getValues();
  var out = [];
  for (var i = 0; i < col.length; i++) {
    if (acWhats_(col[i][0]) === alvo) {
      var n = i + 2;
      out.push(acObj_(headers, sh.getRange(n, 1, 1, headers.length).getValues()[0], n));
    }
  }
  return out;
}
function acTodasLinhas_() {
  var sh = acPlanilha_();
  var dados = sh.getDataRange().getValues();
  if (dados.length < 2) return [];
  var headers = dados[0].map(function (h) { return String(h).trim().toLowerCase(); });
  var out = [];
  for (var i = 1; i < dados.length; i++) {
    if (dados[i].join('').trim() === '') continue;
    out.push(acObj_(headers, dados[i], i + 1));
  }
  return out;
}

        perms[k] = b === true;
      });
      mapa[acNorm_(nome)] = { nome: nome, desc: String(dados[i][h.indexOf('descricao')] || ''), perms: perms };
      ordem.push(acNorm_(nome));
    }
  }
  if (!ordem.length) {
    AC_PERFIS_PADRAO.forEach(function (p) { var o = acPerfilPadraoObj_(p); mapa[acNorm_(p.nome)] = o; ordem.push(acNorm_(p.nome)); });
  }
  return { mapa: mapa, ordem: ordem };
}
/** Modelo de fábrica de um perfil (usado se o nome não existir mais na aba Perfis: ninguém fica sem acesso por causa de um renomeio). */
function acPerfilDeFabrica_(nome) {
  var p = AC_PERFIS_PADRAO.filter(function (x) { return acNorm_(x.nome) === acNorm_(nome); })[0];
  return p ? acPerfilPadraoObj_(p) : null;
}
function acPerfilLegado_(nivel) {
  var n = parseInt(nivel, 10) || 1;
  if (n >= 10) return 'Admin total';
  if (n >= 3) return 'Funcionário'; // 4 a 9 (não usados pelo painel) contam como Funcionário
  if (n === 2) return 'Estagiário';
  return 'Aluno'; // quem se cadastrou no portal é Aluno; "Novato" é escolha manual da gestão
}
function acNivelDe_(perms) {
  if (perms.gerir_acessos) return 10;
  if (perms.manual_3) return 3;
  if (perms.manual_2) return 2;
  if (perms.portal || perms.manual_1) return 1;
  return 0;
}
/** Permissões efetivas de UMA linha: caixinha marcada/desmarcada vale; em branco herda do perfil (ou do Nivel antigo). */
function acPermsDaLinha_(linha, perfis) {
  perfis = perfis || acPerfis_();
  var nomePerfil = String(linha.perfil || '').trim() || acPerfilLegado_(linha.nivel);
  var base = (perfis.mapa[acNorm_(nomePerfil)] || perfis.mapa[acNorm_(acPerfilLegado_(linha.nivel))] || acPerfilDeFabrica_(nomePerfil) || acPerfilDeFabrica_(acPerfilLegado_(linha.nivel)) || { perms: {} }).perms;
  var out = {};
  acChaves_().forEach(function (k) {
    var b = acBool_(linha['perm_' + k]);
    out[k] = b === null ? base[k] === true : b;
  });
  return out;
}
function acPermsLista_(perms) { return acChaves_().filter(function (k) { return perms[k]; }); }

/* ---------- verificação de credenciais ---------- */
/**
 * Confere whatsapp + (senha OU token de sessão). Se o mesmo WhatsApp aparecer em
 * mais de uma linha, une as permissões e fica ativo se qualquer linha estiver ativa.
 * Devolve { ok, usuario, sessao } ou { ok:false, erro, inativa?, bloqueado? }.
 */
function acVerificar_(whatsapp, cred) {
  cred = String(cred == null ? '' : cred);
  if (cred.indexOf('tk2.') === 0) {
    var dono = acWhats_(cred.split('.')[1] || '');
    // um token só fala pelo seu dono: se o pedido cita OUTRO WhatsApp, é tentativa de se passar por alguém
    if (acWhats_(whatsapp) && dono && acWhats_(whatsapp) !== dono) return { ok: false, sessaoInvalida: true, erro: 'Sessão inválida. Entre de novo.' };
    whatsapp = dono || whatsapp;
  }
  var alvo = acWhats_(whatsapp);
  if (!alvo || !cred) return { ok: false, erro: 'WhatsApp ou senha incorretos.' };
  var ehToken = acEhToken_(cred);
  if (!ehToken && acBloqueado_(alvo)) {
    return { ok: false, bloqueado: true, erro: 'Muitas tentativas erradas. Aguarde 15 minutos ou fale com a gestão.' };
  }
  var linhas = acLinhasDoWhats_(alvo);
  var conferem = linhas.filter(function (l) { return acCredencialConfere_(l, cred); });
  var bateu = conferem[0];
  if (!bateu) {
    if (!ehToken) acFalha_(alvo);
    return { ok: false, erro: ehToken ? 'Sessão expirada. Entre de novo.' : 'WhatsApp ou senha incorretos.', sessaoInvalida: ehToken };
  }
  acLimpaFalhas_(alvo);
  var perfis = acPerfis_();
  var ativa = false, nome = '', perms = {}, perfil = '', pessoal = {};
  // só contam as linhas que a credencial abriu E que estão ativas (uma linha antiga suspensa não empresta poder à outra)
  var ativas = conferem.filter(function (l) { return String(l.status || 'Ativo').trim().toLowerCase() !== 'inativo'; });
  ativa = ativas.length > 0;
  ativas.forEach(function (l) {
    nome = String(l.nome || '').trim() || nome;
    var p = acPermsDaLinha_(l, perfis);
    acChaves_().forEach(function (k) { if (p[k]) perms[k] = true; });
    perfil = String(l.perfil || '').trim() || perfil || acPerfilLegado_(l.nivel);
    ['datanascimento', 'instagram', 'cep', 'endereco', 'bairro', 'cidade', 'estado', 'altura', 'peso', 'fotoperfilurl', 'fotocorpourl'].forEach(function (c) {
      var v = l[c];
      if (v !== undefined && v !== '' && v !== null) pessoal[c] = Object.prototype.toString.call(v) === '[object Date]' ? v.toISOString() : String(v);
    });
  });
  if (!ativa) return { ok: false, inativa: true, erro: 'Essa conta está desativada. Fale com a gestão.' };
  var usuario = {
    whatsapp: alvo, name: nome || alvo, status: 'ativo', perfil: perfil,
    level: acNivelDe_(perms), permissoes: acPermsLista_(perms),
    dataNascimento: pessoal.datanascimento || '', instagram: pessoal.instagram || '', cep: pessoal.cep || '',
    endereco: pessoal.endereco || '', bairro: pessoal.bairro || '', cidade: pessoal.cidade || '', estado: pessoal.estado || '',
    altura: pessoal.altura || '', peso: pessoal.peso || '',
    fotoPerfilUrl: pessoal.fotoperfilurl || '', fotoCorpoUrl: pessoal.fotocorpourl || ''
  };
  return { ok: true, usuario: usuario, sessao: acToken_(alvo, bateu.senha) };
}
/** Exige login válido E uma permissão. Devolve { ok, usuario } ou { ok:false, erro, semPermissao? }. */
function acExigir_(whatsapp, cred, permissao) {
  var v = acVerificar_(whatsapp, cred);
  if (!v.ok) return v;
  if (permissao && v.usuario.permissoes.indexOf(permissao) === -1) {
    return { ok: false, semPermissao: true, erro: 'Seu acesso não permite esta ação. Fale com a gestão.' };
  }
  return v;
}

/* ---------- escrita ---------- */
function acGarantirColunas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var faltam = [];
  if (headers.indexOf('perfil') === -1) faltam.push({ chave: 'perfil', nome: 'Perfil' });
  acChaves_().forEach(function (k) { if (headers.indexOf('perm_' + k) === -1) faltam.push({ chave: 'perm_' + k, nome: 'perm_' + k }); });
  if (headers.indexOf('nivel') === -1) faltam.unshift({ chave: 'nivel', nome: 'Nivel' });
  if (headers.indexOf('status') === -1) faltam.unshift({ chave: 'status', nome: 'Status' });
  if (faltam.length) {
    var col = headers.length + 1;
    faltam.forEach(function (f) { sh.getRange(1, col).setValue(f.nome); col++; });
  }
  return sh;
}
/** Marca as caixinhas visuais nas colunas de permissão (só estética; o valor continua TRUE/FALSE). */
function acCaixinhas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var ultima = sh.getLastRow();
  if (ultima < 2) return;
  var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
  acChaves_().forEach(function (k) {
    var i = headers.indexOf('perm_' + k);
    // validação de caixinha NÃO altera o que já está marcado (insertCheckboxes() desmarcaria tudo)
    if (i !== -1) { try { sh.getRange(2, i + 1, ultima - 1, 1).setDataValidation(regra); } catch (e) {} }
  });
}
/** Caixinhas visuais só na linha nova (quando alguém se cadastra). */
function acCaixinhaLinha_(sh, headers, numLinha) {
  try {
    var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
    acChaves_().forEach(function (k) {
      var i = headers.indexOf('perm_' + k);
      if (i !== -1) sh.getRange(numLinha, i + 1).setDataValidation(regra);
    });
  } catch (e) {}
}
/** Grava só as células que mudaram (uma linha), sem reinterpretar o resto dela. */
function acRegravarLinha_(sh, headers, numLinha, mudancas) {
  var atuais = sh.getRange(numLinha, 1, 1, headers.length).getValues()[0];
  Object.keys(mudancas).forEach(function (chave) {
    var i = headers.indexOf(chave);
    if (i === -1) return;
    var novo = mudancas[chave];
    if (String(atuais[i]) === String(novo)) return; // já está assim: não mexe
    var celula = sh.getRange(numLinha, i + 1);
    // texto puro nas colunas de texto: preserva zeros à esquerda (senha "0123") e nunca vira fórmula
    if (typeof novo === 'string' && chave !== 'nivel' && chave.indexOf('perm_') !== 0) celula.setNumberFormat('@');
    celula.setValue(novo);
  });
}
function acMudancasPerfil_(perfilNome, perms) {
  var m = { perfil: perfilNome, nivel: acNivelDe_(perms) };
  acChaves_().forEach(function (k) { m['perm_' + k] = perms[k] === true; });
  return m;
}
function acLog_(quemNome, quemWhats, alvo, acao, detalhe) {
  var aba = acAba_('Acessos_Log', ['DataHora', 'Quem', 'QuemWhatsApp', 'Alvo', 'Acao', 'Detalhe']);
  aba.appendRow([new Date(), quemNome || '', quemWhats || '', alvo || '', acao || '', detalhe || '']);
}
/** Cria uma pessoa nova com o perfil dado (todas as caixinhas explícitas). */
function acRegistrar_(nome, whatsapp, senha, perfilNome, extras) {
  var w = acWhats_(whatsapp);
  nome = String(nome || '').trim();
  senha = String(senha || '');
  if (!nome || !w || !senha) return { ok: false, erro: 'Nome, WhatsApp e senha são obrigatórios.' };
  if (senha.length < 4) return { ok: false, erro: 'A senha precisa ter pelo menos 4 caracteres.' };
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    if (acLinhasDoWhats_(w).length) return { ok: false, erro: 'Já existe uma conta com esse WhatsApp.' };
    var sh = acGarantirColunas_();
    var headers = acCabecalhos_(sh);
    var perfis = acPerfis_();
    var perfil = perfis.mapa[acNorm_(perfilNome)];
    if (!perfil) {
      // perfil apagado/renomeado na aba Perfis: usa o modelo de fábrica (nunca "o primeiro da lista", que poderia ser Admin)
      var padrao = AC_PERFIS_PADRAO.filter(function (p) { return acNorm_(p.nome) === acNorm_(perfilNome); })[0];
      if (!padrao) return { ok: false, erro: 'Perfil não encontrado.' };
      perfil = acPerfilPadraoObj_(padrao);
    }
    var m = acMudancasPerfil_(perfil.nome, perfil.perms);
    m.nome = nome; m.whatsapp = w; m.senha = acGuardarSenha_(senha); m.status = 'Ativo';
    extras = extras || {};
    Object.keys(extras).forEach(function (k) { if (extras[k]) m[String(k).toLowerCase()] = extras[k]; });
    var linha = headers.map(function (h) { return m[h] !== undefined ? m[h] : ''; });
    var novaLinha = sh.getLastRow() + 1;
    // texto puro nas colunas de texto (não vira fórmula, não perde zero à esquerda)
    headers.forEach(function (h, i) {
      if (h && h !== 'nivel' && h.indexOf('perm_') !== 0 && linha[i] !== '') sh.getRange(novaLinha, i + 1).setNumberFormat('@');
    });
    sh.getRange(novaLinha, 1, 1, headers.length).setValues([linha]);
    acCaixinhaLinha_(sh, headers, novaLinha);
  } finally {
    lock.releaseLock();
  }
  return { ok: true };
}

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'funcionarios';

    }

  }

function registrarFuncionario_(data) {
  var nome = String(data.nome || '').trim();
  var whatsapp = normalizarWhats_(data.whatsapp);
  var senha = String(data.senha || '');
  var dataNascimento = String(data.dataNascimento || '').trim();
  var instagram = String(data.instagram || '').trim();

  if (!nome || !whatsapp || !senha) {
    return jsonOut_({ ok: false, erro: 'Nome, WhatsApp e senha são obrigatórios.' });
  }
  // Cadastro aberto ao público sempre entra com o perfil mais básico (Aluno, ou
  // Cliente se vier do catálogo). Quem sobe de perfil é o Admin total, no painel.
  var perfilInicial = data.origem === 'catalogo' ? 'Cliente' : 'Aluno';
  var cacheReg = CacheService.getScriptCache();
  var feitosReg = parseInt(cacheReg.get('reg_hora') || '0', 10);
  if (feitosReg >= 120) return jsonOut_({ ok: false, erro: 'Muitos cadastros agora. Tente de novo em alguns minutos.' });
  cacheReg.put('reg_hora', String(feitosReg + 1), 3600);
  var r = acRegistrar_(nome, whatsapp, senha, perfilInicial, {});
  if (!r.ok && perfilInicial === 'Aluno') {
    // Cliente do catálogo que agora se cadastra no portal: com a MESMA senha, sobe de Cliente para Aluno (nunca mais que isso)
    var ex = acVerificar_(whatsapp, senha);
    if (ex.ok && ex.usuario && acNorm_(ex.usuario.perfil) === 'cliente') {
      var up = acessosDefinirPerfil_(whatsapp, 'Aluno', { name: ex.usuario.name, whatsapp: ex.usuario.whatsapp });
      if (up && up.ok) r = { ok: true };
    }
  }
  if (!r.ok) return jsonOut_({ ok: false, erro: r.erro });
  if (dataNascimento) gravarCampoFuncionario_(whatsapp, 'datanascimento', 'DataNascimento', dataNascimento);
  if (instagram) gravarCampoFuncionario_(whatsapp, 'instagram', 'Instagram', instagram);
  // site novo já entra logado; site antigo ignora esses campos extras
  var v = acVerificar_(whatsapp, senha);
  if (v.ok) return jsonOut_({ ok: true, usuario: v.usuario, sessao: v.sessao });
  return jsonOut_({ ok: true });
}

 * Localiza a linha do WhatsApp na planilha Funcionarios e grava UM campo,
 * criando a coluna no fim da planilha na hora se ela ainda não existir --
 * é assim que os campos novos do perfil (data de nascimento, Instagram,
 * endereço, altura, peso, fotos...) vão aparecendo na planilha sozinhos,
 * sem precisar editar a estrutura dela na mão.
 */
function gravarCampoFuncionario_(whatsapp, colunaChave, colunaBonita, valor) {
  var alvo = normalizarWhats_(whatsapp);
  var sheet = getFuncionariosSheet_();
  var dados = sheet.getDataRange().getValues();
  var headers = dados[0].map(function (h) { return String(h).trim().toLowerCase(); });
  var idxWhats = headers.indexOf('whatsapp');
  var idxColuna = headers.indexOf(colunaChave);
  if (idxColuna === -1) {
    idxColuna = headers.length;
    sheet.getRange(1, idxColuna + 1).setValue(colunaBonita);
  }
  for (var i = 1; i < dados.length; i++) {
    if (idxWhats !== -1 && normalizarWhats_(dados[i][idxWhats]) === alvo) {
      sheet.getRange(i + 1, idxColuna + 1).setNumberFormat('@').setValue(String(valor == null ? '' : valor));
      return true;
    }
  }
  return false;
}

/**

 * whatsapp+senha, não precisa ser admin). Só grava as chaves conhecidas em
 * "campos" -- nada de deixar o cliente escrever em coluna arbitrária.
 */
function atualizarPerfilUsuario_(data) {

/**
 * Recebe uma foto (perfil ou corpo inteiro) em base64 -- já redimensionada
 * no navegador antes de enviar, pra não pesar -- salva como arquivo no
 * Drive e grava o link na planilha.
 */
function enviarFotoPerfil_(data) {
  var whatsapp = normalizarWhats_(data.whatsapp);
  var vu = verificarUsuario_(whatsapp, data.senha);
  if (!vu.ok) {
    return jsonOut_({ ok: false, erro: vu.erro || 'WhatsApp ou senha incorretos.' });
  }
  var tipo = data.tipo === 'corpo' ? 'corpo' : 'perfil';
  var base64 = String(data.fotoBase64 || '');
  var m = base64.match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,([\s\S]+)$/);
  if (!m) return jsonOut_({ ok: false, erro: 'Formato de imagem inválido.' });
  var bytes;
  try {
    bytes = Utilities.base64Decode(m[2]);
  } catch (err) {
    return jsonOut_({ ok: false, erro: 'Não foi possível ler a imagem enviada.' });
  }
  if (bytes.length > 6 * 1024 * 1024) {
    return jsonOut_({ ok: false, erro: 'Imagem grande demais (máximo 6MB).' });
  }
  var blob = Utilities.newBlob(bytes, m[1], whatsapp + '-' + tipo);
  var pasta = getPastaFotosPerfil_();
  var arquivo = pasta.createFile(blob);
  arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var url = 'https://lh3.googleusercontent.com/d/' + arquivo.getId();
  if (tipo === 'corpo') {
    gravarCampoFuncionario_(whatsapp, 'fotocorpourl', 'FotoCorpoURL', url);
  } else {
    gravarCampoFuncionario_(whatsapp, 'fotoperfilurl', 'FotoPerfilURL', url);
  }
  return jsonOut_({ ok: true, url: url });
}

/**
 * Extrai uma imagem inline do Google Doc, salva como arquivo separado no
 * Drive (compartilhado como "qualquer pessoa com o link pode ver") e devolve
 * um bloco {type:'img'} pronto pro site -- em vez de guardar a imagem inteira
 * em base64 dentro do JSON (o que estouraria rápido o limite de ~50 mil
 * caracteres por célula da planilha).
 */
function extrairImagemDrive_(imagemInline, pasta) {
  var blob = imagemInline.getBlob();
  var arquivo = pasta.createFile(blob);
  arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    type: 'img',
    src: 'https://lh3.googleusercontent.com/d/' + arquivo.getId(),
    width: imagemInline.getWidth(),
    height: imagemInline.getHeight()
  };
}

/**
 * Lê um Google Doc inteiro e transforma em {parts:[{title, chapters:[{title,
 * blocks}]}]} -- o mesmo formato que o site já usava. Convenções:
 *  - Título 1 (Heading 1)  -> nova PARTE do manual
 *  - Título 2 ou 3         -> novo CAPÍTULO dentro da parte atual
 *  - Subtítulo (Subtitle)  -> vira uma citação em destaque
 *  - Texto começando com "> "  -> também vira citação (alternativa ao Subtítulo)
 *  - Lista com marcadores   -> vira tópicos (mantém o nível de indentação)
 *  - Texto todo em negrito  -> vira parágrafo em destaque
 *  - Imagem dentro do texto -> extraída pro Drive e inserida como bloco de imagem,
 *    já no tamanho (largura/altura) em que foi deixada no Doc
 *  - TUDO que vem antes do primeiro Título 1 (título do documento, subtítulo,
 *    sumário...) é ignorado: é só apresentação do Doc, não vira parte nem
 *    capítulo no site. (Se o Doc não tiver nenhum Título 1, nada é ignorado.)
 */
function analisarDocManual_(docId) {
  var doc = DocumentApp.openById(docId);
  var body = doc.getBody();
  var pastaImagens = null; // só cria/abre a pasta se realmente aparecer uma imagem
  var parts = [];
  var parteAtual = null;
  var capituloAtual = null;

  function garantirParte() {
    if (!parteAtual) {
      parteAtual = { title: 'Geral', chapters: [] };
      parts.push(parteAtual);
    }
    return parteAtual;
  }
  function garantirCapitulo() {
    garantirParte();
    if (!capituloAtual) {
      capituloAtual = { title: 'Introdução', blocks: [] };
      parteAtual.chapters.push(capituloAtual);
    }
    return capituloAtual;
  }

  var total = body.getNumChildren();

  // Tudo antes do primeiro Título 1 (título, subtítulo, sumário) é só
  // apresentação do Doc: não vira parte nem capítulo no site.
  var inicio = 0;
  for (var k = 0; k < total; k++) {
    var elK = body.getChild(k);
    if (elK.getType() === DocumentApp.ElementType.PARAGRAPH &&
        elK.asParagraph().getHeading() === DocumentApp.ParagraphHeading.HEADING1) {
      inicio = k;
      break;
    }
  }

  for (var i = inicio; i < total; i++) {
    var el = body.getChild(i);
    var tipo = el.getType();

    if (tipo === DocumentApp.ElementType.PARAGRAPH) {
      var par = el.asParagraph();
      var heading = par.getHeading();
      var texto = par.getText();

      // imagens inline dentro deste parágrafo
      var imagensDoParagrafo = [];
      var nFilhos = par.getNumChildren();
      for (var j = 0; j < nFilhos; j++) {
        var filho = par.getChild(j);
        if (filho.getType() === DocumentApp.ElementType.INLINE_IMAGE) {
          if (!pastaImagens) pastaImagens = getPastaImagens_();
          imagensDoParagrafo.push(extrairImagemDrive_(filho.asInlineImage(), pastaImagens));
        }
      }

      if (heading === DocumentApp.ParagraphHeading.HEADING1) {
        parteAtual = { title: texto || 'Parte', chapters: [] };
        parts.push(parteAtual);
        capituloAtual = null;
        continue;
      }
      if (heading === DocumentApp.ParagraphHeading.HEADING2 || heading === DocumentApp.ParagraphHeading.HEADING3) {
        garantirParte();
        capituloAtual = { title: texto || 'Capítulo', blocks: [] };
        parteAtual.chapters.push(capituloAtual);
        continue;
      }
      if (heading === DocumentApp.ParagraphHeading.TITLE) {
        continue; // título do documento -- o título do manual já vem da configuração no site
      }

      if (texto.trim() === '' && !imagensDoParagrafo.length) continue; // parágrafo realmente vazio -- nem cria capítulo "Geral" à toa

      var cap = garantirCapitulo();
      imagensDoParagrafo.forEach(function (img) { cap.blocks.push(img); });

      if (texto.trim() === '') continue; // só tinha imagem, já foi adicionada acima

      if (heading === DocumentApp.ParagraphHeading.SUBTITLE) {
        cap.blocks.push({ type: 'quote', text: texto });
        continue;
      }
      var mQuote = texto.match(/^>\s?(.*)$/);
      if (mQuote) {
        cap.blocks.push({ type: 'quote', text: mQuote[1] });
        continue;
      }
      // isBold() sem argumentos: true = parágrafo todo em negrito, false = nada,
      // null = misturado (nesse caso trata como não-negrito, pra não travar a sincronização).
      var negrito = par.editAsText().isBold();
      cap.blocks.push({ type: 'p', text: texto, bold: negrito === true });

    } else if (tipo === DocumentApp.ElementType.LIST_ITEM) {
      var item = el.asListItem();
      var txt2 = item.getText();
      if (txt2.trim() === '') continue;
      garantirCapitulo().blocks.push({ type: 'bul', level: item.getNestingLevel() || 0, text: txt2 });

    } else if (tipo === DocumentApp.ElementType.INLINE_IMAGE) {
      // imagem solta fora de um parágrafo normal (raro, mas cobre o caso)
      if (!pastaImagens) pastaImagens = getPastaImagens_();
      garantirCapitulo().blocks.push(extrairImagemDrive_(el.asInlineImage(), pastaImagens));
    }
  }

  return parts;
}

/**
 * Escreve um conteúdo {parts:[...]} dentro de um Google Doc recém-criado --
 * usado só na migração dos 3 manuais antigos, pra já nascerem preenchidos
 * com o texto que já existia no site.
 */
function escreverDocDeParts_(doc, parts) {
  var body = doc.getBody();
  body.clear();
  (parts || []).forEach(function (part) {
    body.appendParagraph(part.title || 'Parte').setHeading(DocumentApp.ParagraphHeading.HEADING1);
    (part.chapters || []).forEach(function (ch) {
      body.appendParagraph(ch.title || 'Capítulo').setHeading(DocumentApp.ParagraphHeading.HEADING2);
      (ch.blocks || []).forEach(function (b) {
        if (b.type === 'bul') {
          var li = body.appendListItem(b.text || '');
          li.setNestingLevel(b.level || 0);
          li.setGlyphType(DocumentApp.GlyphType.BULLET);
        } else if (b.type === 'quote') {
          body.appendParagraph(b.text || '').setHeading(DocumentApp.ParagraphHeading.SUBTITLE);
        } else if (b.type === 'img') {
          // imagens antigas (se algum dia existiram em base64) não são recriadas
          // automaticamente no Doc -- a partir de agora, imagem se adiciona direto no Doc.
        } else {
          var p = body.appendParagraph(b.text || '');
          p.setHeading(DocumentApp.ParagraphHeading.NORMAL);
          if (b.bold) p.editAsText().setBold(true);
        }
      });
    });
  });
}

/**
 * Cria o Google Doc (vazio ou já preenchido) e registra a linha na aba
 * "Manuais". Devolve o id gerado, o id do Doc e o link de edição.
 */
function criarManualRegistro_(manualId, titulo, blurb, nivelMinimo, parts) {
  var doc = DocumentApp.create(titulo || 'Novo manual');
  var docId = doc.getId();
  if (parts && parts.length) {
    escreverDocDeParts_(doc, parts);
  } else {
    doc.getBody().appendParagraph('Capítulo 1').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    doc.getBody().appendParagraph('Escreva aqui o conteúdo do capítulo...');
  }
  doc.saveAndClose();

  var sheet = getManuaisSheet_();
  var dados = sheet.getDataRange().getValues();
  var maxOrdem = 0;
  for (var i = 1; i < dados.length; i++) { maxOrdem = Math.max(maxOrdem, Number(dados[i][5]) || 0); }

  sheet.appendRow([
    manualId, titulo || '', blurb || '', parseInt(nivelMinimo, 10) || 1,
    docId, maxOrdem + 1, JSON.stringify(parts || []), new Date()
  ]);
  return { manualId: manualId, docId: docId, docUrl: 'https://docs.google.com/document/d/' + docId + '/edit' };
}

function sincronizarManual_(data) {
  var _ac = exigirPerm_(data, 'gerir_manuais');
  if (!_ac.ok) return negarAcesso_(_ac);
  var manualId = String(data.manualId || '');
  var sheet = getManuaisSheet_();
  var dados = sheet.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) {
    if (String(dados[i][0]) === manualId) {
      var docId = dados[i][4];
      if (!docId) return jsonOut_({ ok: false, erro: 'Este manual não tem um Google Doc associado.' });
      var parts;
      try {
        parts = analisarDocManual_(docId);
      } catch (err) {
        return jsonOut_({ ok: false, erro: 'Não foi possível ler o Google Doc: ' + String(err) });
      }
      var partsJson = JSON.stringify(parts);
      if (partsJson.length > 45000) {
        return jsonOut_({ ok: false, erro: 'Este manual ficou grande demais pra salvar (limite da planilha). Fale com o suporte pra dividir em dois manuais.' });
      }
      sheet.getRange(i + 1, 7).setValue(partsJson);
      var agora = new Date();
      sheet.getRange(i + 1, 8).setValue(agora);
      return jsonOut_({ ok: true, parts: parts, atualizadoEm: agora.toISOString() });
    }
  }
  return jsonOut_({ ok: false, erro: 'Manual não encontrado.' });
}

  var dados = sheet.getDataRange().getValues();

    if (String(dados[i][0]) === manualId) {
      if (data.titulo != null) sheet.getRange(i + 1, 2).setValue(String(data.titulo));
      if (data.blurb != null) sheet.getRange(i + 1, 3).setValue(String(data.blurb));
      if (data.nivelMinimo != null) sheet.getRange(i + 1, 4).setValue(parseInt(data.nivelMinimo, 10) || 1);
      if (data.ordem != null) sheet.getRange(i + 1, 6).setValue(Number(data.ordem) || 0);
      return jsonOut_({ ok: true });
    }
  }
  return jsonOut_({ ok: false, erro: 'Manual não encontrado.' });
}

function excluirManual_(data) {

  if (!_ac.ok) return negarAcesso_(_ac);

/**
 * Copia as planilhas de Funcionarios e Resultados para a pasta de backups
 * do Drive e mantém só as 8 cópias mais recentes de cada. Para automatizar:
 * Gatilhos (relógio, menu à esquerda) -> Adicionar gatilho -> fazerBackupAgora
 * -> Baseado em tempo -> Contador semanal.
 */
function fazerBackupAgora() {
  var pasta = getPastaBackups_();
  var carimbo = Utilities.formatDate(new Date(), 'America/Fortaleza', 'yyyy-MM-dd HH-mm');
  var alvos = [
    { nome: 'Funcionarios+Manuais', id: FUNCIONARIOS_SHEET_ID },
    { nome: 'Resultados', id: RESULTADOS_SHEET_ID }
  ];
  alvos.forEach(function (a) {
    DriveApp.getFileById(a.id).makeCopy('Backup ' + a.nome + ' ' + carimbo, pasta);
    var copias = [];
    var it = pasta.getFiles();
    while (it.hasNext()) {
      var f = it.next();
      if (f.getName().indexOf('Backup ' + a.nome + ' ') === 0) copias.push(f);
    }
    copias.sort(function (x, y) { return y.getDateCreated() - x.getDateCreated(); });
    copias.slice(8).forEach(function (f) { f.setTrashed(true); });
  });
  Logger.log('Backup feito em ' + carimbo + '.');
}

function migrarSenhasParaHash() {
  if (!listaFechada_()) {

/** Preenche caixinhas/perfil em branco com o que a pessoa já tinha (perfil ou Nivel antigo). Devolve quantas linhas mexeu. */
function reconciliarLinhas_(maximo) {
  var sh = acGarantirColunas_();
  var headers = acCabecalhos_(sh);
  var perfis = acPerfis_();
  var mexeu = 0;
  var linhas = acTodasLinhas_();
  for (var i = 0; i < linhas.length && mexeu < (maximo || 200); i++) {
    var l = linhas[i];
    var precisa = !String(l.perfil || '').trim();
    if (!precisa) acChaves_().forEach(function (k) { if (acBool_(l['perm_' + k]) === null) precisa = true; });
    if (!precisa) continue;
    var nomePerfil = String(l.perfil || '').trim() || acPerfilLegado_(l.nivel);
    if (String(l.perfil || '').trim() && !perfis.mapa[acNorm_(nomePerfil)]) continue; // perfil renomeado/apagado na aba Perfis: não mexe nessa pessoa
    var perms = acPermsDaLinha_(l, perfis);
    var m = acMudancasPerfil_((perfis.mapa[acNorm_(nomePerfil)] || {}).nome || nomePerfil, perms);
    acRegravarLinha_(sh, headers, l.__row, m);
    mexeu++;
  }
  return mexeu;
}

function acAdminsAtivos_(excluirWhats) {
  var perfis = acPerfis_();
  var vistos = {}, n = 0;
  acTodasLinhas_().forEach(function (l) {
    var w = acWhats_(l.whatsapp);
    if (!w || w === excluirWhats || vistos[w]) return;
    if (String(l.status || 'Ativo').trim().toLowerCase() === 'inativo') return;
    if (acPermsDaLinha_(l, perfis).gerir_acessos) { vistos[w] = true; n++; }
  });
  return n;
}

function acessosListar_(data) {
  var _ac = exigirPerm_(data, 'gerir_acessos');
  if (!_ac.ok) return negarAcesso_(_ac);
  reconciliarLinhas_(200);
  var perfis = acPerfis_();
  var busca = acNorm_(data.busca || '');
  var incluirClientes = data.incluirClientes === true;
  var filtroPerfis = null;
  if (Array.isArray(data.perfis)) { filtroPerfis = {}; data.perfis.forEach(function (n) { filtroPerfis[acNorm_(n)] = true; }); }
  var pessoas = [], totalClientes = 0, vistos = {}, contagem = {};
  acTodasLinhas_().forEach(function (l) {
    var w = acWhats_(l.whatsapp);
    if (!w || vistos[w]) return;
    vistos[w] = true;
    var nomePerfil = String(l.perfil || '').trim() || acPerfilLegado_(l.nivel);
    var ehCliente = acNorm_(nomePerfil) === 'cliente';
    if (ehCliente) totalClientes++;
    contagem[nomePerfil] = (contagem[nomePerfil] || 0) + 1;
    var bate = !busca || acNorm_(l.nome).indexOf(busca) !== -1 || w.indexOf(busca.replace(/\D/g, '') || '§') !== -1;
    // com texto na busca, procura em TODOS os perfis (ninguém "some" por causa de um filtro esquecido)
    var passaFiltro = busca ? true : (filtroPerfis ? !!filtroPerfis[acNorm_(nomePerfil)] : (!ehCliente || incluirClientes));
    if (!passaFiltro || (busca && !bate)) return;
    pessoas.push({
      whatsapp: w, nome: String(l.nome || ''), perfil: nomePerfil,
      status: String(l.status || 'Ativo'), nivel: parseInt(l.nivel, 10) || 0,
      permissoes: acPermsDaLinha_(l, perfis)
    });
  });
  pessoas.sort(function (a, b) { return a.nome.localeCompare(b.nome); });
  var truncado = pessoas.length > 300;
  if (truncado) pessoas = pessoas.slice(0, 300);
  return jsonOut_({
    ok: true, eu: _ac.usuario.whatsapp, pessoas: pessoas, totalClientes: totalClientes, contagem: contagem, truncado: truncado,
    perfis: perfis.ordem.map(function (k) { var p = perfis.mapa[k]; return { nome: p.nome, desc: p.desc, permissoes: p.perms }; }),
    permissoes: AC_PERMISSOES
  });
}

function acessosDefinirPerfilSemTrava_(whatsappAlvo, nomePerfil, quem) {

/**
 * PASSO DA MIGRAÇÃO. Faz backup, cria as colunas Perfil e perm_... na aba de

 * perfil + caixinhas: 1 = Novato, 2 = Estagiário, 3 = Funcionário, 10 = Admin total.
 * Não apaga nada e pode ser rodada de novo sem problema.

  fazerBackupAgora();
  acGarantirColunas_();
  garantirAbaPerfis_();
  var mexeu = reconciliarLinhas_(100000);
  acCaixinhas_();

}


/** Executa uma alteração de acesso uma de cada vez (evita dois admins se removendo ao mesmo tempo). */
function acComTrava_(fn) {
  var lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (e) { return { ok: false, erro: 'O sistema está ocupado agora. Tente de novo em alguns segundos.' }; }
  try { return fn(); } finally { lock.releaseLock(); }
}
function acessosDefinirPerfil_(a, b, c) { return acComTrava_(function () { return acessosDefinirPerfilSemTrava_(a, b, c); }); }
function acessosDefinirPermissao_(a, b, c, d) { return acComTrava_(function () { return acessosDefinirPermissaoSemTrava_(a, b, c, d); }); }
function acessosDefinirStatus_(a, b, c) { return acComTrava_(function () { return acessosDefinirStatusSemTrava_(a, b, c); }); }


/** Rode UMA vez: quem hoje está como "Novato" passa a "Aluno" (mesmas permissões). Novato daqui pra frente é só escolha manual no painel. */
function converterNovatosEmAlunos() {

  linhas.forEach(function (l) {

  });

  Logger.log('Novatos convertidos em Alunos: ' + n);
  return n;
}
