function normalizar_(texto) {
  return String(texto || '').trim().toLowerCase();
}

function escaparHtml_(texto) {
  return String(texto || '')
    .replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

/**
 * Registra um erro inesperado na aba "Erros_Log", sem nunca lançar
 * exceção por conta própria (se o log falhar, só ignora -- logar erro
 * não pode gerar outro erro). Usado pelo try/catch de doGet/doPost.
 */
function logErroServidor_(origem, erro, contexto) {
  try {
    const sheet = getOrCreateSheet_(SHEET_ERROS, ['DataHora', 'Origem', 'Mensagem', 'Stack', 'Contexto']);
    sheet.appendRow([
      new Date(), origem || '', (erro && erro.message) || String(erro),
      (erro && erro.stack) || '', JSON.stringify(contexto || {})
    ]);
  } catch (erroLog) {
    // Silencioso de propósito.
  }
}

function getSS_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function getOrCreateSheet_(nome, headers) {
  const ss = getSS_();
  let sheet = ss.getSheetByName(nome);
  if (!sheet) {
    sheet = ss.insertSheet(nome);
    if (headers && headers.length > 0) {
      sheet.appendRow(headers);
    }
  }
  return sheet;
}

function getListaValores_(nome, valoresPadrao) {
  const sheet = getOrCreateSheet_(nome, ['Nome']);
  const dados = sheet.getDataRange().getValues();
  const valores = [];
  for (let i = 1; i < dados.length; i++) {
    if (dados[i][0]) valores.push(String(dados[i][0]).trim());
  }
  if (valores.length === 0 && valoresPadrao) {
    valoresPadrao.forEach(v => {
      sheet.appendRow([v]);
      valores.push(v);
    });
  }
  return valores;
}

/**
 * Adiciona um valor novo (categoria ou gênero) numa lista, evitando duplicado.
 * Esta função já era chamada pelo doPost (addCategoria/addGenero) mas não existia
 * no arquivo anterior — corrigida aqui para não dar erro de servidor se essas
 * ações forem usadas no futuro.
 */
function adicionarValorLista_(nomeSheet, valoresPadrao, novoValor) {
  const valorLimpo = nomeDeLista_(novoValor);
  if (!valorLimpo) {
    return { ok: false, erro: 'Valor vazio.' };
  }
  const valores = getListaValores_(nomeSheet, valoresPadrao);
  const existente = valores.filter(v => acNorm_(v) === acNorm_(valorLimpo))[0];
  if (existente) {
    return { ok: true, jaExistia: true, nome: existente };
  }
  const sheet = getOrCreateSheet_(nomeSheet, ['Nome']);
  sheet.appendRow([acTexto_(valorLimpo)]);
  return { ok: true, nome: valorLimpo };
}

/** Nome de categoria/gênero pronto para gravar: sem vírgula (ela separa as categorias dentro da peça), sem espaço repetido, até 60 caracteres. */
function nomeDeLista_(valor) {
  return String(valor == null ? '' : valor).replace(/,/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 60).trim();
}

/** Números das linhas (1 = cabeçalho) da aba de categorias cujo nome é `nome` (ignora acento e maiúscula). */
function linhasDaCategoria_(aba, nome) {
  const dados = aba.getDataRange().getValues();
  const alvo = acNorm_(nome);
  const out = [];
  for (let i = 1; i < dados.length; i++) if (acNorm_(dados[i][0]) === alvo) out.push(i + 1);
  return out;
}

/** Peças (linha da planilha de produtos + as categorias que elas têm) que usam a categoria `nome`. A coluna 3 é a Categoria (mesma posição usada no resto do script). */
function pecasDaCategoria_(nome) {
  const prod = getProdutosSheet_();
  const dados = prod.getDataRange().getValues();
  const alvo = acNorm_(nome);
  const achadas = [];
  for (let i = 1; i < dados.length; i++) {
    const cats = String(dados[i][2] == null ? '' : dados[i][2]).split(',').map(function (c) { return c.trim(); }).filter(Boolean);
    if (cats.some(function (c) { return acNorm_(c) === alvo; })) achadas.push({ linha: i + 1, cats: cats });
  }
  return { prod: prod, achadas: achadas };
}

/**
 * Apaga uma categoria (Admin total, só com login). Se peças usam a categoria e a chamada não trouxe confirmar:true, NÃO apaga:
 * devolve emUso = quantas peças. Com confirmar:true tira a categoria dessas peças (elas continuam no catálogo, com as outras categorias) e apaga.
 */
function excluirCategoria_(body) {
  const nome = nomeDeLista_(body.nome);
  if (!nome) return jsonResponse_({ ok: false, erro: 'Informe a categoria.' });
  const aba = getOrCreateSheet_(SHEET_CATEGORIAS, ['Nome']);
  const linhas = linhasDaCategoria_(aba, nome);
  if (!linhas.length) return jsonResponse_({ ok: false, naoExiste: true, erro: 'Essa categoria não existe mais.' });
  const uso = pecasDaCategoria_(nome);
  if (uso.achadas.length && body.confirmar !== true) {
    return jsonResponse_({ ok: false, emUso: uso.achadas.length, erro: 'Essa categoria está em ' + uso.achadas.length + (uso.achadas.length === 1 ? ' peça.' : ' peças.') });
  }
  const alvo = acNorm_(nome);
  uso.achadas.forEach(function (a) {
    uso.prod.getRange(a.linha, 3).setValue(acTexto_(a.cats.filter(function (c) { return acNorm_(c) !== alvo; }).join(', ')));
  });
  for (let k = linhas.length - 1; k >= 0; k--) aba.deleteRow(linhas[k]); // de baixo para cima: o número das linhas de cima não muda
  registrarAudit_(body.funcionario, 'excluir_categoria', '', { nome: nome, pecas: uso.achadas.length });
  return jsonResponse_({ ok: true, nome: nome, pecas: uso.achadas.length });
}

/**
 * Corrige o nome de uma categoria na lista E em todas as peças que a usam (Admin total, só com login). Se o nome novo já existe como OUTRA
 * categoria, junta as duas (as peças ficam com a que já existia). Peça com várias categorias mantém as outras e nunca fica com a mesma duas vezes.
 */
function renomearCategoria_(body) {
  const antigo = nomeDeLista_(body.nome);
  const novo = nomeDeLista_(body.novoNome);
  if (!antigo || !novo) return jsonResponse_({ ok: false, erro: 'Informe o nome atual e o nome novo.' });
  const aba = getOrCreateSheet_(SHEET_CATEGORIAS, ['Nome']);
  const linhas = linhasDaCategoria_(aba, antigo);
  if (!linhas.length) return jsonResponse_({ ok: false, naoExiste: true, erro: 'Essa categoria não existe mais.' });
  const mudouSoGrafia = acNorm_(antigo) === acNorm_(novo); // ex.: "calcas" -> "Calças"
  const outra = mudouSoGrafia ? undefined : getListaValores_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO).filter(function (v) { return acNorm_(v) === acNorm_(novo); })[0];
  const nomeFinal = outra || novo;
  const alvo = acNorm_(antigo);
  const uso = pecasDaCategoria_(antigo);
  uso.achadas.forEach(function (a) {
    const saida = [];
    a.cats.forEach(function (c) {
      const n = acNorm_(c) === alvo ? nomeFinal : c;
      if (!saida.some(function (o) { return acNorm_(o) === acNorm_(n); })) saida.push(n);
    });
    uso.prod.getRange(a.linha, 3).setValue(acTexto_(saida.join(', ')));
  });
  if (outra) {
    for (let k = linhas.length - 1; k >= 0; k--) aba.deleteRow(linhas[k]); // a que já existia fica; a velha sai
  } else {
    aba.getRange(linhas[0], 1).setValue(acTexto_(nomeFinal));
    for (let k = linhas.length - 1; k >= 1; k--) aba.deleteRow(linhas[k]); // sobras repetidas
  }
  registrarAudit_(body.funcionario, 'renomear_categoria', '', { de: antigo, para: nomeFinal, pecas: uso.achadas.length, juntou: !!outra });
  return jsonResponse_({ ok: true, nome: nomeFinal, juntou: !!outra, pecas: uso.achadas.length });
}

/**
 * Move para a lixeira do Drive o arquivo antigo de foto (do Drive) quando ele é
 * substituído por um novo upload -- evita acumular fotos órfãs que nunca mais são
 * usadas. Usa lixeira (setTrashed) em vez de exclusão definitiva: se algo der errado,
 * o arquivo pode ser restaurado manualmente pela Lixeira do Drive por 30 dias. Nunca
 * lança erro -- falha aqui não pode travar a atualização do produto.
 */
function excluirFotoAntigaDrive_(urlAntiga, urlNova) {
  if (!urlAntiga || urlAntiga === urlNova) return;
  if (urlAntiga.indexOf('cloudinary.com') !== -1) return; // mídia do Cloudinary não é arquivo do Drive
  try {
    const m = String(urlAntiga).match(/\/d\/([a-zA-Z0-9_-]+)/);
    if (!m || !m[1]) return;
    DriveApp.getFileById(m[1]).setTrashed(true);
  } catch (erro) {
    logErroServidor_('excluirFotoAntigaDrive_', erro, { urlAntiga: urlAntiga });
  }
}

/** SHA-1 em hexadecimal (mesmo formato que o Cloudinary exige para a assinatura). */
function cldHex_(bytes) {
  return bytes.map(function (b) { const v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}

/** Credenciais do Cloudinary nas Propriedades do Script, ou lança erro claro se faltar alguma. */
function cldCredenciais_() {
  const props = PropertiesService.getScriptProperties();
  const cloudName = props.getProperty('CLOUDINARY_CLOUD_NAME');
  const apiKey = props.getProperty('CLOUDINARY_API_KEY');
  const apiSecret = props.getProperty('CLOUDINARY_API_SECRET');
  if (!cloudName || !apiKey || !apiSecret) {
    throw new Error('Cloudinary não configurado no servidor (faltam CLOUDINARY_CLOUD_NAME/CLOUDINARY_API_KEY/CLOUDINARY_API_SECRET nas Propriedades do Script).');
  }
  return { cloudName: cloudName, apiKey: apiKey, apiSecret: apiSecret };
}

/**
 * v2.4: devolve os parâmetros assinados para o NAVEGADOR subir um vídeo DIRETO pro Cloudinary,
 * sem passar o arquivo pelo Apps Script. Um vídeo pode levar minutos pra enviar numa rede móvel
 * e o Apps Script tem limite de tempo de execução -- não é o lugar certo pra isso, e qualquer
 * soluço no caminho derruba o envio inteiro. O Cloudinary é feito pra aguentar upload grande
 * direto do navegador numa conexão instável. O api_secret nunca sai daqui -- só a assinatura.
 */
function assinarUploadVideo_(body) {
  const cred = cldCredenciais_();
  const timestamp = Math.floor(new Date().getTime() / 1000);
  const paramsAssinados = 'folder=' + CLOUDINARY_PASTA + '&timestamp=' + timestamp;
  const assinatura = cldHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_1, paramsAssinados + cred.apiSecret));
  return jsonResponse_({
    ok: true,
    cloudName: cred.cloudName,
    apiKey: cred.apiKey,
    timestamp: timestamp,
    folder: CLOUDINARY_PASTA,
    signature: assinatura
  });
}

/**
 * Faz upload de 1 arquivo (imagem ou vídeo) para o Cloudinary, assinado (sem precisar
 * de upload preset público). Devolve a URL segura (https) do arquivo salvo.
 */
function cldSubir_(base64Data, mimeType, nomeArquivo, tipoRecurso) {
  const cred = cldCredenciais_();
  const timestamp = Math.floor(new Date().getTime() / 1000);
  // A assinatura cobre todos os parâmetros ENVIADOS exceto api_key, file e a própria signature,
  // em ordem alfabética do nome do parâmetro -- regra do Cloudinary.
  const paramsAssinados = 'folder=' + CLOUDINARY_PASTA + '&timestamp=' + timestamp;
  const assinatura = cldHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_1, paramsAssinados + cred.apiSecret));

  const bytes = Utilities.base64Decode(base64Data);
  const blob = Utilities.newBlob(bytes, mimeType, nomeArquivo);

  const resp = UrlFetchApp.fetch('https://api.cloudinary.com/v1_1/' + cred.cloudName + '/' + tipoRecurso + '/upload', {
    method: 'post',
    muteHttpExceptions: true,
    payload: {
      file: blob,
      api_key: cred.apiKey,
      timestamp: String(timestamp),
      folder: CLOUDINARY_PASTA,
      signature: assinatura
    }
  });
  const codigoResp = resp.getResponseCode();
  const dadosResp = JSON.parse(resp.getContentText());
  if (codigoResp !== 200) {
    const msgErro = dadosResp.error && dadosResp.error.message ? dadosResp.error.message : ('HTTP ' + codigoResp);
    throw new Error('Erro do Cloudinary: ' + msgErro);
  }
  return dadosResp.secure_url;
}

/** Salva 1 foto no Cloudinary e devolve a URL. Usada por produtos novos/editados e pela galeria. */
function salvarFotoCloudinary_(base64Data, mimeType, nomeArquivo) {
  return cldSubir_(base64Data, mimeType || 'image/jpeg', nomeArquivo, 'image');
}

/** Salva 1 vídeo no Cloudinary e devolve a URL (tocável direto num <video>, sem iframe). */
function salvarVideoCloudinary_(base64Data, mimeType, nomeBase) {
  const permitidos = { 'video/mp4': 'mp4', 'video/quicktime': 'mov', 'video/webm': 'webm', 'video/3gpp': '3gp' };
  const mime = String(mimeType || 'video/mp4').toLowerCase();
  if (!permitidos[mime]) throw new Error('Formato de vídeo não aceito. Use MP4, MOV ou WEBM.');
  const bytes = Utilities.base64Decode(base64Data);
  if (bytes.length > MAX_VIDEO_BYTES) throw new Error('Vídeo grande demais (máximo 20 MB). Encurte ou reduza a qualidade e tente de novo.');
  return cldSubir_(base64Data, mime, nomeBase + '.' + permitidos[mime], 'video');
}

/**
 * Apaga do Cloudinary o arquivo antigo quando é substituído por um novo upload --
 * mesmo espírito do excluirFotoAntigaDrive_, mas para mídia que está no Cloudinary.
 * Nunca lança erro: falha aqui não pode travar a atualização do produto.
 */
function cloudinaryExcluir_(urlAntiga) {
  if (!urlAntiga || urlAntiga.indexOf('res.cloudinary.com') === -1) return; // só mexe em mídia que está no Cloudinary
  try {
    const cred = cldCredenciais_();
    const m = String(urlAntiga).match(/\/upload\/(?:v\d+\/)?([^./]+(?:\/[^./]+)*)\.[a-zA-Z0-9]+$/);
    if (!m || !m[1]) return;
    const publicId = m[1];
    const tipoRecurso = /\/video\/upload\//.test(urlAntiga) ? 'video' : 'image';
    const timestamp = Math.floor(new Date().getTime() / 1000);
    const paramsAssinados = 'public_id=' + publicId + '&timestamp=' + timestamp;
    const assinatura = cldHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_1, paramsAssinados + cred.apiSecret));
    UrlFetchApp.fetch('https://api.cloudinary.com/v1_1/' + cred.cloudName + '/' + tipoRecurso + '/destroy', {
      method: 'post',
      muteHttpExceptions: true,
      payload: { public_id: publicId, api_key: cred.apiKey, timestamp: String(timestamp), signature: assinatura }
    });
  } catch (erro) {
    logErroServidor_('cloudinaryExcluir_', erro, { urlAntiga: urlAntiga });
  }
}

/** Apaga a mídia antiga (Drive OU Cloudinary, o que for) quando substituída por uma nova. */
function excluirMidiaAntiga_(urlAntiga, urlNova) {
  if (!urlAntiga || urlAntiga === urlNova) return;
  if (urlAntiga.indexOf('cloudinary.com') !== -1) { cloudinaryExcluir_(urlAntiga); return; }
  excluirFotoAntigaDrive_(urlAntiga, urlNova);
}

const CABECALHOS_PRODUTOS_ = ['ID', 'Nome', 'Categoria', 'Genero', 'Preco', 'Foto_URL', 'Status', 'Data', 'Descricao', 'Estoque', 'Tamanhos', 'Cores', 'Novidade', 'Codigo', 'Foto_Stories_URL', 'Fotos_Galeria'];

/** A lista lê os produtos pelo NOME da coluna. Se algum título da linha 1 estiver em branco (ex.: coluna das fotos extras criada sem título),
 *  os dados até estavam sendo gravados, mas nunca voltavam para a tela. Aqui os títulos que faltam são preenchidos (nunca renomeia um existente). */
function garantirCabecalhos_(sheet) {
  const n = Math.max(sheet.getLastColumn(), CABECALHOS_PRODUTOS_.length);
  let cab = sheet.getRange(1, 1, 1, n).getValues()[0].map(function (h) { return String(h).trim(); });
  let mudou = false;

  // Conserto de um erro de versão anterior: o vídeo (Video_URL) foi criado na coluna P, que é a coluna das fotos extras.
  // Insere uma coluna P nova para as fotos e separa, linha a linha, o que era foto do que era vídeo.
  if (cab.indexOf('Fotos_Galeria') === -1 && cab[15] === 'Video_URL') {
    sheet.insertColumnBefore(16);
    sheet.getRange(1, 16).setValue('Fotos_Galeria');
    const ultima = sheet.getLastRow();
    if (ultima > 1) {
      const vals = sheet.getRange(2, 17, ultima - 1, 1).getValues();
      const saida = vals.map(function (r) {
        const partes = String(r[0] || '').split(',').map(function (x) { return x.trim(); }).filter(Boolean);
        const video = partes.filter(function (u) { return /^https:\/\/drive\.google\.com\/file\/d\/[\w-]+\/preview$/.test(u); })[0] || '';
        const fotos = partes.filter(function (u) { return u !== video && /^https:\/\/(lh3\.googleusercontent\.com|drive\.google\.com)\//.test(u); }).slice(0, MAX_FOTOS_GALERIA);
        return [fotos.join(','), video];
      });
      sheet.getRange(2, 16, ultima - 1, 2).setValues(saida);
    }
    mudou = true;
    cab = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), CABECALHOS_PRODUTOS_.length)).getValues()[0].map(function (h) { return String(h).trim(); });
  }

  // títulos que faltam nas 15 primeiras colunas (nunca renomeia um título existente)
  CABECALHOS_PRODUTOS_.slice(0, 15).forEach(function (nome, i) {
    if (!cab[i]) { sheet.getRange(1, i + 1).setValue(nome); cab[i] = nome; mudou = true; }
  });
  // coluna das fotos extras: usa a que existe; senão, a P se estiver livre; senão, cria no fim
  if (cab.indexOf('Fotos_Galeria') === -1) {
    const pos = !cab[15] ? 16 : (cab.length + 1);
    sheet.getRange(1, pos).setValue('Fotos_Galeria');
    mudou = true;
  }
  if (mudou) invalidarCacheProdutos_();
  return mudou;
}

function obterStats_() {
  const ss = getSS_();
  const sheetEventos = ss.getSheetByName(SHEET_EVENTOS);
  const contagem = {};
  let totalView = 0, totalWhats = 0;
  if (sheetEventos) {
    const dados = sheetEventos.getDataRange().getValues();
    for (let i = 1; i < dados.length; i++) {
      const pid = String(dados[i][1]);
      const codigo = dados[i][2];
      const tipo = dados[i][3];
      if (!contagem[pid]) contagem[pid] = { visualizacao: 0, whatsapp: 0, codigo: codigo };
      if (tipo === 'visualizacao') { contagem[pid].visualizacao++; totalView++; }
      if (tipo === 'whatsapp') { contagem[pid].whatsapp++; totalWhats++; }
    }
  }
  let maisVistoId = null, maiorViews = 0;
  Object.keys(contagem).forEach(pid => {
    if (contagem[pid].visualizacao > maiorViews) { maiorViews = contagem[pid].visualizacao; maisVistoId = pid; }
  });
  return jsonResponse_({
    ok: true,
    totalVisualizacoes: totalView,
    totalWhatsapp: totalWhats,
    produtoMaisVistoId: maisVistoId,
    produtoMaisVistoCodigo: maisVistoId && contagem[maisVistoId] ? contagem[maisVistoId].codigo : '',
    produtoMaisVistoViews: maiorViews
  });
}

function acNorm_(s) {
  return String(s == null ? '' : s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').trim();
}
function acWhats_(w) {
  var d = String(w || '').replace(/\D/g, '');
  if (d.length >= 12 && d.slice(0, 2) === '55') d = d.slice(2);
  if (d.length === 10) d = d.slice(0, 2) + '9' + d.slice(2); // igual ao treinamento: sempre ddd+9+8 dígitos
  return d;
}
/** Texto vindo de fora nunca vira fórmula na planilha: se começa com = + - @ ganha um apóstrofo (o Sheets guarda como texto puro). */
function acTexto_(v) {
  var t = String(v == null ? '' : v);
  return /^[=+\-@\t\r]/.test(t) ? "'" + t : t;
}
function acBool_(v) {
  if (v === true || v === 1) return true;
  if (v === false || v === 0) return false;
  var s = String(v == null ? '' : v).trim().toLowerCase();
  if (s === 'true' || s === 'sim' || s === '1') return true;
  if (s === 'false' || s === 'nao' || s === 'não' || s === '0') return false;
  return null; // em branco / indefinido -> herda
}

/* ---------- senha, hash e token ---------- */
function acBytesHex_(bytes) {
  return bytes.map(function (b) { var v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}
function acSha256Hex_(texto) {
  return acBytesHex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, texto, Utilities.Charset.UTF_8));
}
function acHashSenha_(senha) {
  var sal = Utilities.getUuid().replace(/-/g, '').slice(0, 16);
  return 'h1$' + sal + '$' + acSha256Hex_(sal + ':' + senha);
}
function acGuardarSenha_(senha) { return acSenhasEmHash_() ? acHashSenha_(senha) : String(senha); }
function acIguais_(a, b) {
  a = String(a); b = String(b);
  if (a.length !== b.length) return false;
  var r = 0;
  for (var i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}
function acSenhaSimples_(guardada, informado) {
  guardada = String(guardada == null ? '' : guardada);
  informado = String(informado == null ? '' : informado);
  if (!informado) return false;
  if (guardada.indexOf('h1$') === 0) {
    var p = guardada.split('$');
    if (p.length !== 3) return false;
    return acIguais_(acSha256Hex_(p[1] + ':' + informado), p[2]);
  }
  return acIguais_(guardada, informado);
}
function acHmac_(whats, guardada) {
  return acBytesHex_(Utilities.computeHmacSha256Signature(whats + '|' + String(guardada), acSegredo_()));
}
/** Token de sessão (autodescritivo): tk2.<whatsapp>.<assinatura>. Só vale enquanto a senha guardada não mudar. */
function acToken_(whats, guardada) { return 'tk2.' + whats + '.' + acHmac_(whats, guardada); }
function acEhToken_(cred) { var s = String(cred || ''); return s.indexOf('tk1.') === 0 || s.indexOf('tk2.') === 0; }
function acCredencialConfere_(linha, cred) {
  cred = String(cred == null ? '' : cred);
  var whats = acWhats_(linha.whatsapp);
  if (cred.indexOf('tk2.') === 0) {
    var p = cred.split('.');
    return p.length === 3 && p[1] === whats && acIguais_(acHmac_(whats, linha.senha), p[2]);
  }
  if (cred.indexOf('tk1.') === 0) return acIguais_('tk1.' + acHmac_(whats, linha.senha), cred);
  return acSenhaSimples_(linha.senha, cred);
}

function acCabecalhos_(sh) {
  var n = sh.getLastColumn();
  if (n < 1) return [];
  return sh.getRange(1, 1, 1, n).getValues()[0].map(function (h) { return String(h).trim().toLowerCase(); });
}
function acObj_(headers, valores, numLinha) {
  var o = { __row: numLinha };
  headers.forEach(function (h, i) { if (h) o[h] = valores[i]; });
  return o;
}
/** Só as linhas de UM WhatsApp (lê a coluna de WhatsApp inteira e depois só as linhas certas -- rápido mesmo com muitos clientes). */
function acLinhasDoWhats_(whatsapp) {
  var alvo = acWhats_(whatsapp);
  if (!alvo) return [];
  var sh = acPlanilha_();
  var ultima = sh.getLastRow();
  if (ultima < 2) return [];
  var headers = acCabecalhos_(sh);
  var iw = headers.indexOf('whatsapp');
  if (iw === -1) return [];
  var col = sh.getRange(2, iw + 1, ultima - 1, 1).getValues();
  var out = [];
  for (var i = 0; i < col.length; i++) {
    if (acWhats_(col[i][0]) === alvo) {
      var n = i + 2;
      out.push(acObj_(headers, sh.getRange(n, 1, 1, headers.length).getValues()[0], n));
    }
  }
  return out;
}
function acTodasLinhas_() {
  var sh = acPlanilha_();
  var dados = sh.getDataRange().getValues();
  if (dados.length < 2) return [];
  var headers = dados[0].map(function (h) { return String(h).trim().toLowerCase(); });
  var out = [];
  for (var i = 1; i < dados.length; i++) {
    if (dados[i].join('').trim() === '') continue;
    out.push(acObj_(headers, dados[i], i + 1));
  }
  return out;
}

function acChaves_() { return AC_PERMISSOES.map(function (p) { return p.chave; }); }
function acPerfilPadraoObj_(p) {
  var perms = {};
  acChaves_().forEach(function (k) { perms[k] = p.perms.indexOf(k) !== -1; });
  return { nome: p.nome, desc: p.desc, perms: perms };
}
/** Lê a aba "Perfis" (editável por você); se não existir, usa os perfis de fábrica. */
function acPerfis_() {
  var mapa = {}, ordem = [];
  var aba = acSS_().getSheetByName('Perfis');
  if (aba && aba.getLastRow() >= 2) {
    var dados = aba.getDataRange().getValues();
    var h = dados[0].map(function (x) { return String(x).trim().toLowerCase(); });
    for (var i = 1; i < dados.length; i++) {
      var nome = String(dados[i][h.indexOf('perfil')] || '').trim();
      if (!nome) continue;
      var perms = {};
      acChaves_().forEach(function (k) {
        var idx = h.indexOf('perm_' + k);
        var b = idx === -1 ? null : acBool_(dados[i][idx]);
        perms[k] = b === true;
      });
      mapa[acNorm_(nome)] = { nome: nome, desc: String(dados[i][h.indexOf('descricao')] || ''), perms: perms };
      ordem.push(acNorm_(nome));
    }
  }
  if (!ordem.length) {
    AC_PERFIS_PADRAO.forEach(function (p) { var o = acPerfilPadraoObj_(p); mapa[acNorm_(p.nome)] = o; ordem.push(acNorm_(p.nome)); });
  }
  return { mapa: mapa, ordem: ordem };
}
/** Modelo de fábrica de um perfil (usado se o nome não existir mais na aba Perfis: ninguém fica sem acesso por causa de um renomeio). */
function acPerfilDeFabrica_(nome) {
  var p = AC_PERFIS_PADRAO.filter(function (x) { return acNorm_(x.nome) === acNorm_(nome); })[0];
  return p ? acPerfilPadraoObj_(p) : null;
}
function acPerfilLegado_(nivel) {
  var n = parseInt(nivel, 10) || 1;
  if (n >= 10) return 'Admin total';
  if (n >= 3) return 'Funcionário'; // 4 a 9 (não usados pelo painel) contam como Funcionário
  if (n === 2) return 'Estagiário';
  return 'Aluno'; // quem se cadastrou no portal é Aluno; "Novato" é escolha manual da gestão
}
function acNivelDe_(perms) {
  if (perms.gerir_acessos) return 10;
  if (perms.manual_3) return 3;
  if (perms.manual_2) return 2;
  if (perms.portal || perms.manual_1) return 1;
  return 0;
}
/** Permissões efetivas de UMA linha: caixinha marcada/desmarcada vale; em branco herda do perfil (ou do Nivel antigo). */
function acPermsDaLinha_(linha, perfis) {
  perfis = perfis || acPerfis_();
  var nomePerfil = String(linha.perfil || '').trim() || acPerfilLegado_(linha.nivel);
  var base = (perfis.mapa[acNorm_(nomePerfil)] || perfis.mapa[acNorm_(acPerfilLegado_(linha.nivel))] || acPerfilDeFabrica_(nomePerfil) || acPerfilDeFabrica_(acPerfilLegado_(linha.nivel)) || { perms: {} }).perms;
  var out = {};
  acChaves_().forEach(function (k) {
    var b = acBool_(linha['perm_' + k]);
    out[k] = b === null ? base[k] === true : b;
  });
  return out;
}
function acPermsLista_(perms) { return acChaves_().filter(function (k) { return perms[k]; }); }

/* ---------- verificação de credenciais ---------- */
/**
 * Confere whatsapp + (senha OU token de sessão). Se o mesmo WhatsApp aparecer em
 * mais de uma linha, une as permissões e fica ativo se qualquer linha estiver ativa.
 * Devolve { ok, usuario, sessao } ou { ok:false, erro, inativa?, bloqueado? }.
 */
function acVerificar_(whatsapp, cred) {
  cred = String(cred == null ? '' : cred);
  if (cred.indexOf('tk2.') === 0) {
    var dono = acWhats_(cred.split('.')[1] || '');
    // um token só fala pelo seu dono: se o pedido cita OUTRO WhatsApp, é tentativa de se passar por alguém
    if (acWhats_(whatsapp) && dono && acWhats_(whatsapp) !== dono) return { ok: false, sessaoInvalida: true, erro: 'Sessão inválida. Entre de novo.' };
    whatsapp = dono || whatsapp;
  }
  var alvo = acWhats_(whatsapp);
  if (!alvo || !cred) return { ok: false, erro: 'WhatsApp ou senha incorretos.' };
  var ehToken = acEhToken_(cred);
  if (!ehToken && acBloqueado_(alvo)) {
    return { ok: false, bloqueado: true, erro: 'Muitas tentativas erradas. Aguarde 15 minutos ou fale com a gestão.' };
  }
  var linhas = acLinhasDoWhats_(alvo);
  var conferem = linhas.filter(function (l) { return acCredencialConfere_(l, cred); });
  var bateu = conferem[0];
  if (!bateu) {
    if (!ehToken) acFalha_(alvo);
    return { ok: false, erro: ehToken ? 'Sessão expirada. Entre de novo.' : 'WhatsApp ou senha incorretos.', sessaoInvalida: ehToken };
  }
  acLimpaFalhas_(alvo);
  var perfis = acPerfis_();
  var ativa = false, nome = '', perms = {}, perfil = '', pessoal = {};
  // só contam as linhas que a credencial abriu E que estão ativas (uma linha antiga suspensa não empresta poder à outra)
  var ativas = conferem.filter(function (l) { return String(l.status || 'Ativo').trim().toLowerCase() !== 'inativo'; });
  ativa = ativas.length > 0;
  ativas.forEach(function (l) {
    nome = String(l.nome || '').trim() || nome;
    var p = acPermsDaLinha_(l, perfis);
    acChaves_().forEach(function (k) { if (p[k]) perms[k] = true; });
    perfil = String(l.perfil || '').trim() || perfil || acPerfilLegado_(l.nivel);
    ['datanascimento', 'instagram', 'cep', 'endereco', 'bairro', 'cidade', 'estado', 'altura', 'peso', 'fotoperfilurl', 'fotocorpourl'].forEach(function (c) {
      var v = l[c];
      if (v !== undefined && v !== '' && v !== null) pessoal[c] = Object.prototype.toString.call(v) === '[object Date]' ? v.toISOString() : String(v);
    });
  });
  if (!ativa) return { ok: false, inativa: true, erro: 'Essa conta está desativada. Fale com a gestão.' };
  var usuario = {
    whatsapp: alvo, name: nome || alvo, status: 'ativo', perfil: perfil,
    level: acNivelDe_(perms), permissoes: acPermsLista_(perms),
    dataNascimento: pessoal.datanascimento || '', instagram: pessoal.instagram || '', cep: pessoal.cep || '',
    endereco: pessoal.endereco || '', bairro: pessoal.bairro || '', cidade: pessoal.cidade || '', estado: pessoal.estado || '',
    altura: pessoal.altura || '', peso: pessoal.peso || '',
    fotoPerfilUrl: pessoal.fotoperfilurl || '', fotoCorpoUrl: pessoal.fotocorpourl || ''
  };
  return { ok: true, usuario: usuario, sessao: acToken_(alvo, bateu.senha) };
}
/** Exige login válido E uma permissão. Devolve { ok, usuario } ou { ok:false, erro, semPermissao? }. */
function acExigir_(whatsapp, cred, permissao) {
  var v = acVerificar_(whatsapp, cred);
  if (!v.ok) return v;
  if (permissao && v.usuario.permissoes.indexOf(permissao) === -1) {
    return { ok: false, semPermissao: true, erro: 'Seu acesso não permite esta ação. Fale com a gestão.' };
  }
  return v;
}

/* ---------- escrita ---------- */
function acGarantirColunas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var faltam = [];
  if (headers.indexOf('perfil') === -1) faltam.push({ chave: 'perfil', nome: 'Perfil' });
  acChaves_().forEach(function (k) { if (headers.indexOf('perm_' + k) === -1) faltam.push({ chave: 'perm_' + k, nome: 'perm_' + k }); });
  if (headers.indexOf('nivel') === -1) faltam.unshift({ chave: 'nivel', nome: 'Nivel' });
  if (headers.indexOf('status') === -1) faltam.unshift({ chave: 'status', nome: 'Status' });
  if (faltam.length) {
    var col = headers.length + 1;
    faltam.forEach(function (f) { sh.getRange(1, col).setValue(f.nome); col++; });
  }
  return sh;
}
/** Marca as caixinhas visuais nas colunas de permissão (só estética; o valor continua TRUE/FALSE). */
function acCaixinhas_() {
  var sh = acPlanilha_();
  var headers = acCabecalhos_(sh);
  var ultima = sh.getLastRow();
  if (ultima < 2) return;
  var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
  acChaves_().forEach(function (k) {
    var i = headers.indexOf('perm_' + k);
    // validação de caixinha NÃO altera o que já está marcado (insertCheckboxes() desmarcaria tudo)
    if (i !== -1) { try { sh.getRange(2, i + 1, ultima - 1, 1).setDataValidation(regra); } catch (e) {} }
  });
}
/** Caixinhas visuais só na linha nova (quando alguém se cadastra). */
function acCaixinhaLinha_(sh, headers, numLinha) {
  try {
    var regra = SpreadsheetApp.newDataValidation().requireCheckbox().build();
    acChaves_().forEach(function (k) {
      var i = headers.indexOf('perm_' + k);
      if (i !== -1) sh.getRange(numLinha, i + 1).setDataValidation(regra);
    });
  } catch (e) {}
}
/** Grava só as células que mudaram (uma linha), sem reinterpretar o resto dela. */
function acRegravarLinha_(sh, headers, numLinha, mudancas) {
  var atuais = sh.getRange(numLinha, 1, 1, headers.length).getValues()[0];
  Object.keys(mudancas).forEach(function (chave) {
    var i = headers.indexOf(chave);
    if (i === -1) return;
    var novo = mudancas[chave];
    if (String(atuais[i]) === String(novo)) return; // já está assim: não mexe
    var celula = sh.getRange(numLinha, i + 1);
    // texto puro nas colunas de texto: preserva zeros à esquerda (senha "0123") e nunca vira fórmula
    if (typeof novo === 'string' && chave !== 'nivel' && chave.indexOf('perm_') !== 0) celula.setNumberFormat('@');
    celula.setValue(novo);
  });
}
function acMudancasPerfil_(perfilNome, perms) {
  var m = { perfil: perfilNome, nivel: acNivelDe_(perms) };
  acChaves_().forEach(function (k) { m['perm_' + k] = perms[k] === true; });
  return m;
}

/** Cria uma pessoa nova com o perfil dado (todas as caixinhas explícitas). */
function acRegistrar_(nome, whatsapp, senha, perfilNome, extras) {
  var w = acWhats_(whatsapp);
  nome = String(nome || '').trim();
  senha = String(senha || '');
  if (!nome || !w || !senha) return { ok: false, erro: 'Nome, WhatsApp e senha são obrigatórios.' };
  if (senha.length < 4) return { ok: false, erro: 'A senha precisa ter pelo menos 4 caracteres.' };
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    if (acLinhasDoWhats_(w).length) return { ok: false, erro: 'Já existe uma conta com esse WhatsApp.' };
    var sh = acGarantirColunas_();
    var headers = acCabecalhos_(sh);
    var perfis = acPerfis_();
    var perfil = perfis.mapa[acNorm_(perfilNome)];
    if (!perfil) {
      // perfil apagado/renomeado na aba Perfis: usa o modelo de fábrica (nunca "o primeiro da lista", que poderia ser Admin)
      var padrao = AC_PERFIS_PADRAO.filter(function (p) { return acNorm_(p.nome) === acNorm_(perfilNome); })[0];
      if (!padrao) return { ok: false, erro: 'Perfil não encontrado.' };
      perfil = acPerfilPadraoObj_(padrao);
    }
    var m = acMudancasPerfil_(perfil.nome, perfil.perms);
    m.nome = nome; m.whatsapp = w; m.senha = acGuardarSenha_(senha); m.status = 'Ativo';
    extras = extras || {};
    Object.keys(extras).forEach(function (k) { if (extras[k]) m[String(k).toLowerCase()] = extras[k]; });
    var linha = headers.map(function (h) { return m[h] !== undefined ? m[h] : ''; });
    var novaLinha = sh.getLastRow() + 1;
    // texto puro nas colunas de texto (não vira fórmula, não perde zero à esquerda)
    headers.forEach(function (h, i) {
      if (h && h !== 'nivel' && h.indexOf('perm_') !== 0 && linha[i] !== '') sh.getRange(novaLinha, i + 1).setNumberFormat('@');
    });
    sh.getRange(novaLinha, 1, 1, headers.length).setValues([linha]);
    acCaixinhaLinha_(sh, headers, novaLinha);
  } finally {
    lock.releaseLock();
  }
  return { ok: true };
}

/**
 * Autoriza uma ação protegida. Aceita (1) sessão de login + permissão, ou
 * (2) o PIN antigo, enquanto ele estiver ligado. Devolve { ok, nome? } ou { ok:false, erro, ... }.
 */
function autorizarAcao_(credencial, whatsapp, permissao, bodyPin) {
  const cred = String(credencial || '');
  if (acEhToken_(cred)) {
    const v = acExigir_(whatsapp, cred, permissao);
    if (!v.ok) return v;
    if (v.usuario.permissoes.indexOf('catalogo_admin') === -1) {
      return { ok: false, semPermissao: true, erro: 'Seu acesso não inclui o admin do catálogo. Fale com a gestão.' };
    }
    return { ok: true, nome: v.usuario.name, whatsapp: v.usuario.whatsapp, via: 'login' };
  }
  if (!pinLegadoAtivo_()) {
    return { ok: false, exigeLogin: true, pinInvalido: true, erro: 'Entre com seu WhatsApp e senha para continuar.' };
  }
  const pinCorreto = PropertiesService.getScriptProperties().getProperty('ADMIN_PIN');
  if (!pinCorreto) return { ok: true, via: 'pin' };
  if (acBloqueado_('pin')) return { ok: false, bloqueado: true, pinInvalido: true, erro: 'Muitas tentativas de PIN. Aguarde 15 minutos ou entre com seu login.' };
  if (String(bodyPin == null ? cred : bodyPin) === String(pinCorreto)) return { ok: true, via: 'pin' };
  acFalha_('pin');
  return { ok: false, pinInvalido: true, erro: 'PIN incorreto.' };
}

function registrarCliente_(body) {
  // freio simples contra cadastro em massa (vale para o site todo)
  const cache = CacheService.getScriptCache();
  const feitos = parseInt(cache.get('reg_hora') || '0', 10);
  if (feitos >= 60) return jsonResponse_({ ok: false, erro: 'Muitos cadastros agora. Tente de novo em alguns minutos.' });
  cache.put('reg_hora', String(feitos + 1), 3600);
  const r = acRegistrar_(body.nome, body.whatsapp, body.senha, 'Cliente', {});
  if (!r.ok) return jsonResponse_({ ok: false, erro: r.erro });
  const v = acVerificar_(body.whatsapp, body.senha);
  if (v.ok) return jsonResponse_({ ok: true, usuario: v.usuario, sessao: v.sessao });
  return jsonResponse_({ ok: true });
}

function limparIds_(lista) {
  if (!Array.isArray(lista)) return [];
  const vistos = {}, out = [];
  lista.forEach(function (x) {
    const s = String(x == null ? '' : x).slice(0, 40);
    if (s && !vistos[s] && out.length < 300) { vistos[s] = true; out.push(s); }
  });
  return out;
}

function linhaDoCliente_(aba, whats) {
  const ultima = aba.getLastRow();
  if (ultima < 2) return -1;
  const col = aba.getRange(2, 1, ultima - 1, 1).getValues();
  for (let i = 0; i < col.length; i++) if (acWhats_(col[i][0]) === whats) return i + 2;
  return -1;
}

function clienteDados_(body) {
  const v = acExigir_(body.whatsapp, body.sessao, 'cliente');
  if (!v.ok) return respostaNegada_(v);
  const aba = getOrCreateSheet_(SHEET_CLIENTES, ['WhatsApp', 'Favoritos', 'Sacola', 'AtualizadoEm']);
  const n = linhaDoCliente_(aba, v.usuario.whatsapp);
  let favoritos = [], sacola = [];
  if (n !== -1) {
    const r = aba.getRange(n, 1, 1, 4).getValues()[0];
    try { favoritos = limparIds_(JSON.parse(r[1] || '[]')); } catch (e) {}
    try { sacola = limparIds_(JSON.parse(r[2] || '[]')); } catch (e) {}
  }
  return jsonResponse_({ ok: true, favoritos: favoritos, sacola: sacola });
}

function clienteSalvar_(body) {
  const v = acExigir_(body.whatsapp, body.sessao, 'cliente');
  if (!v.ok) return respostaNegada_(v);
  const favoritos = JSON.stringify(limparIds_(body.favoritos));
  const sacola = JSON.stringify(limparIds_(body.sacola));
  const trava = LockService.getScriptLock();
  try { trava.waitLock(15000); } catch (e) { return jsonResponse_({ ok: false, erro: 'Tente de novo em instantes.' }); }
  try {
    const aba = getOrCreateSheet_(SHEET_CLIENTES, ['WhatsApp', 'Favoritos', 'Sacola', 'AtualizadoEm']);

    if (n === -1) aba.appendRow(linha); else aba.getRange(n, 1, 1, 4).setValues([linha]);
  } finally {
    trava.releaseLock();
  }
  return jsonResponse_({ ok: true });
}

const TAM_PEDACO_CACHE = 30000; // caracteres por pedaço (limite do Google é ~100KB por valor)

function versaoCacheProdutos_() {
  return PropertiesService.getScriptProperties().getProperty('CACHE_V') || '0';
}

/** Chamado depois de qualquer alteração em produtos: descarta todas as listas em cache. */
function invalidarCacheProdutos_() {
  try {
    const props = PropertiesService.getScriptProperties();
    props.setProperty('CACHE_V', String((parseInt(props.getProperty('CACHE_V') || '0', 10) || 0) + 1));
  } catch (erro) { /* cache é só um extra: nunca pode derrubar um cadastro */ }
}

function cacheLerGrande_(chave) {
  try {
    const cache = CacheService.getScriptCache();
    const n = parseInt(cache.get(chave + ':n') || '0', 10);
    if (!n) return null;
    const chaves = [];
    for (let i = 0; i < n; i++) chaves.push(chave + ':' + i);
    const partes = cache.getAll(chaves);
    let texto = '';
    for (let i = 0; i < n; i++) {
      const p = partes[chaves[i]];
      if (p === undefined || p === null) return null;
      texto += p;
    }
    return texto;
  } catch (erro) { return null; }
}

function cacheGravarGrande_(chave, texto, segundos) {
  try {
    const cache = CacheService.getScriptCache();
    const n = Math.ceil(texto.length / TAM_PEDACO_CACHE);
    const lote = {};
    for (let i = 0; i < n; i++) lote[chave + ':' + i] = texto.substr(i * TAM_PEDACO_CACHE, TAM_PEDACO_CACHE);
    lote[chave + ':n'] = String(n);
    cache.putAll(lote, segundos);
  } catch (erro) { /* sem cache, o site só fica um pouco mais lento */ }
}

}

/**
 * Backup: copia a planilha do catálogo para uma pasta do Drive e mantém só as
 * 8 cópias mais recentes (as mais antigas vão pra lixeira, recuperáveis por 30 dias).
 */
function fazerBackupAgora() {
  const nomePasta = 'TENKiTER - Backups (catálogo)';
  const pastas = DriveApp.getFoldersByName(nomePasta);
  const pasta = pastas.hasNext() ? pastas.next() : DriveApp.createFolder(nomePasta);
  const carimbo = Utilities.formatDate(new Date(), 'America/Fortaleza', 'yyyy-MM-dd HH-mm');
  const prefixo = 'Backup Catalogo ';
  DriveApp.getFileById(getSS_().getId()).makeCopy(prefixo + carimbo, pasta);
  const copias = [];
  const it = pasta.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (f.getName().indexOf(prefixo) === 0) copias.push(f);
  }
  copias.sort((a, b) => b.getDateCreated() - a.getDateCreated());
  copias.slice(8).forEach(f => f.setTrashed(true));
  Logger.log('Backup do catálogo feito em ' + carimbo + '.');
}

/** Porta de entrada do GET: só captura qualquer erro inesperado e registra em Erros_Log,
 *  para nunca devolver uma tela de erro em branco para quem está acessando o catálogo. */
function doGet(e) {
  try {
    return doGetInterno_(e);
  } catch (erro) {
    logErroServidor_('doGet', erro, { parametros: e && e.parameter });
    return jsonResponse_({ ok: false, erro: 'Erro interno no servidor: ' + (erro && erro.message || erro) });
  }
}

function doGetInterno_(e) {
  const action = (e.parameter.action || 'list');

  if (action === 'share') return paginaCompartilhamento_(e.parameter.id);
  if (action === 'stats') return statsComCache_();
  if (action === 'versao') return jsonResponse_({
    ok: true, versao: VERSAO_CATALOGO, acessos: true, pinAtivo: pinLegadoAtivo_(),
    pedidos: true, config: true, metricas: true, importacao: true, feed: true, integracoes: true, codigos: true, pushHistorico: true, loteCategoria: true,
    categoriasGestao: true, pushCompleto: true, conexoes: true, conexoesOnesignal: true
  });
  if (action === 'config') return configPublica_();
  if (action === 'pushconfig') return pushConfigPublica_();
  if (action === 'feed') return feedProdutos_(e.parameter.formato);
  if (action === 'sitemap') return sitemapProdutos_();

  if (action === 'historico') {
    const aut = autorizarAcao_(e.parameter.sessao || e.parameter.pin, e.parameter.whatsapp, 'catalogo_auditoria', e.parameter.pin);
    if (!aut.ok) return respostaNegada_(aut);
    return obterHistorico_(e);
  }

  if (action === 'list') {
    const incluirTodos = e.parameter.todos === 'true';
    const chaveCache = 'lista:' + (incluirTodos ? 't' : 'a') + ':' + versaoCacheProdutos_();
    const emCache = cacheLerGrande_(chaveCache);
    if (emCache) return respostaTexto_(emCache);
    const sheet = getProdutosSheet_();
    try { garantirCabecalhos_(sheet); } catch (erroCab) { /* leitura pública nunca falha por causa disso */ }
    const dados = sheet.getDataRange().getValues();
    const headers = dados[0];
    const produtos = [];
    for (let i = 1; i < dados.length; i++) {
      const linha = dados[i];
      const obj = {};
      headers.forEach((h, idx) => obj[h] = linha[idx]);
      if (incluirTodos || normalizar_(obj.Status) === 'ativo') {
        produtos.push(obj);
      }
    }
    const textoLista = JSON.stringify({ ok: true, produtos: produtos });
    cacheGravarGrande_(chaveCache, textoLista, 60);
    return respostaTexto_(textoLista);
  }

  if (action === 'categorias') return jsonResponse_({ ok: true, categorias: getListaValores_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO) });
  if (action === 'generos') return jsonResponse_({ ok: true, generos: getListaValores_(SHEET_GENEROS, GENEROS_PADRAO) });

  return jsonResponse_({ ok: false, erro: 'Ação desconhecida: ' + action });
}

// ===================== POST =====================
/** Porta de entrada do POST: só captura qualquer erro inesperado e registra em Erros_Log,
 *  para nunca devolver uma resposta quebrada para o app depois de uma ação de gravação. */
function doPost(e) {
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (erroParse) {
    return jsonResponse_({ ok: false, erro: 'Requisição inválida.' });
  }
  try {
    return doPostInterno_(body);
  } catch (erro) {
    logErroServidor_('doPost', erro, { action: body && body.action });
    return jsonResponse_({ ok: false, erro: 'Erro interno no servidor: ' + (erro && erro.message || erro) });
  }
}

function doPostInterno_(body) {
  const action = body.action;

  // ---- abertas (não exigem permissão de gestão)
  if (action === 'registrarEvento') return registrarEvento_(body);
  if (action === 'login') return loginPessoa_(body);
  if (action === 'sessao') return sessaoPessoa_(body);
  if (action === 'registrar_cliente') return registrarCliente_(body);
  if (action === 'cliente_dados') return clienteDados_(body);
  if (action === 'cliente_salvar') return clienteSalvar_(body);
  if (action === 'criarPedido') { try { return criarPedido_(body); } finally { pushDespacharFila_(); } }
  if (action === 'consultarPedido') return consultarPedido_(body);
  if (action === 'pushIdentidade') return pushIdentidade_(body);

  // ---- protegidas: login + permissão (ou PIN antigo enquanto ligado)
  const permissao = permissaoDaAcao_(body);
  if (!permissao) return despacharPost_(body); // ação desconhecida
  const aut = autorizarAcao_(body.sessao || body.pin, body.whatsapp, permissao, body.pin);
  if (!aut.ok) return respostaNegada_(aut);
  if (aut.nome) body.funcionario = aut.nome; // auditoria com o nome verdadeiro de quem fez
  if (ACOES_SO_COM_LOGIN_.indexOf(action) !== -1 && aut.via !== 'login') {
    return respostaNegada_({ exigeLogin: true, erro: 'Esta ação é só para Admin total e exige entrar com sua conta (WhatsApp e senha).' });
  }

  if (ACOES_QUE_ALTERAM_PRODUTOS_.indexOf(action) === -1) return despacharPost_(body);
  if (ACOES_COM_TRAVA_.indexOf(action) === -1) {
    try { return despacharPost_(body); } finally { invalidarCacheProdutos_(); }
  }
  const trava = LockService.getScriptLock();
  try {
    trava.waitLock(30000);
  } catch (erroTrava) {
    return jsonResponse_({ ok: false, erro: 'O sistema está salvando outra alteração agora. Tente de novo em alguns segundos.' });
  }
  try {
    return despacharPost_(body);
  } finally {
    invalidarCacheProdutos_();
    trava.releaseLock();
  }
}

function despacharPost_(body) {
  const action = body.action;

  if (action === 'registrarEvento') return registrarEvento_(body);

  if (action === 'create') return criarProduto_(body);
  if (action === 'update') return atualizarProduto_(body);
  if (action === 'delete') return deletarProduto_(body);
  if (action === 'reativar') return reativarProduto_(body);
  if (action === 'excluirPermanente') return excluirPermanentemente_(body);
  if (action === 'bulkUpdate') return acaoEmLote_(body);
  if (action === 'addCategoria') return jsonResponse_(adicionarValorLista_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO, body.nome));
  if (action === 'addGenero') return jsonResponse_(adicionarValorLista_(SHEET_GENEROS, GENEROS_PADRAO, body.nome));
  if (action === 'excluirCategoria') return excluirCategoria_(body);
  if (action === 'renomearCategoria') return renomearCategoria_(body);
  if (action === 'gerarSeo') return gerarSeoComGemini_(body);
  if (action === 'sugerirPorFoto') return sugerirPorFotoComGemini_(body);
  if (action === 'statusConfigIA') return statusConfigIA_();

function registrarEvento_(body) {
  // Limite de taxa simples: ignora eventos idênticos (mesmo produto+tipo) repetidos em
  // menos de CACHE_TTL_RATE_LIMIT_EVENTO segundos — protege contra cliques duplicados/scripts
  // e contra inflar a planilha de Eventos sem precisar de PIN (esse endpoint é público, de propósito).
  const chaveRate = 'evt_' + (body.produtoId || '') + '_' + (body.tipo || '');
  const cache = CacheService.getScriptCache();
  if (cache.get(chaveRate)) {
    return jsonResponse_({ ok: true, ignorado: true });
  }
  try { cache.put(chaveRate, '1', CACHE_TTL_RATE_LIMIT_EVENTO); } catch (erroCache) { /* silencioso */ }

  const sheet = getOrCreateSheet_(SHEET_EVENTOS, ['DataHora', 'ProdutoID', 'Codigo', 'Tipo']);
  sheet.appendRow([new Date(), acTexto_(String(body.produtoId || '').slice(0, 60)), acTexto_(String(body.codigo || '').slice(0, 60)), acTexto_(String(body.tipo || '').slice(0, 30))]);
  if (body.meta) enviarEventoMetaCapi_(body, String(body.tipo || '')); // v3.1: só faz algo se o Pixel + token estiverem configurados e a pessoa aceitou o aviso
  return jsonResponse_({ ok: true });
}

function processarGaleriaFotos_(fotosBase64Array, idProduto) {
  if (!fotosBase64Array || !Array.isArray(fotosBase64Array) || fotosBase64Array.length === 0) return '';
  const urls = [];
  fotosBase64Array.forEach((b64, idx) => {
    if (b64) {
      const url = salvarFotoCloudinary_(b64, 'image/jpeg', 'produto_' + idProduto + '_galeria_' + idx + '.jpg');
      urls.push(url);
    }
  });
  return urls.join(',');
}

function criarProduto_(body) {
  if (!body.nome || !String(body.nome).trim()) {
    return jsonResponse_({ ok: false, erro: 'Informe o nome do produto.' });
  }
  const precoValidado = Number(body.preco);
  if (body.preco === undefined || body.preco === '' || isNaN(precoValidado) || precoValidado <= 0) {
    return jsonResponse_({ ok: false, erro: 'Informe um preço válido.' });
  }

  const sheet = getProdutosSheet_();
  garantirCabecalhos_(sheet);

  // Anti-duplicado: evita cadastrar o mesmo produto duas vezes (ex.: duplo clique, app travando
  // e a pessoa tenta de novo) -- mesmo nome + mesmo preço cadastrados nos últimos 10 segundos.
  const dadosAtuais = sheet.getDataRange().getValues();
  const agoraMs = new Date().getTime();
  for (let j = 1; j < dadosAtuais.length; j++) {
    const linhaAtual = dadosAtuais[j];
    if (normalizar_(linhaAtual[1]) === normalizar_(body.nome) && Number(linhaAtual[4]) === precoValidado) {
      const dataLinhaMs = linhaAtual[7] instanceof Date ? linhaAtual[7].getTime() : new Date(linhaAtual[7]).getTime();
      if (!isNaN(dataLinhaMs) && (agoraMs - dataLinhaMs) < 10000) {
        return jsonResponse_({ ok: false, erro: 'Esse produto parece ter sido cadastrado agora mesmo. Confira a lista antes de cadastrar de novo.' });
      }
    }
  }

  const id = new Date().getTime().toString();
  const codigo = proximoCodigo_();
  let fotoUrl = body.fotoUrlExistente || '';
  let fotoStoriesUrl = body.fotoStoriesUrlExistente || '';

  if (body.fotoBase64) {
    fotoUrl = salvarFotoCloudinary_(body.fotoBase64, body.fotoMime || 'image/jpeg', 'produto_' + id + '.jpg');
  }
  if (body.fotoStoriesBase64) {
    fotoStoriesUrl = salvarFotoCloudinary_(body.fotoStoriesBase64, 'image/jpeg', 'produto_' + id + '_stories.jpg');
  }

  const galeriaUrls = processarGaleriaFotos_(body.fotosGaleriaBase64, id);
  const fotosGaleriaFinal = limitarGaleria_(body.fotosGaleriaExistentes ? (body.fotosGaleriaExistentes + (galeriaUrls ? ',' + galeriaUrls : '')) : galeriaUrls);
  const estoqueVal = body.estoque !== undefined && body.estoque !== '' ? Number(body.estoque) : '';

  // v2.4: o vídeo normalmente já chega como URL -- subiu direto do navegador pro Cloudinary antes
  // dessa chamada (ver assinarUploadVideo_). videoBase64 continua funcionando como caminho antigo.
  let videoUrl = body.videoUrlExistente || '';
  try {
    if (body.videoBase64) videoUrl = salvarVideoCloudinary_(body.videoBase64, body.videoMime, 'produto_' + id + '_video');
  } catch (erroVideo) {
    return jsonResponse_({ ok: false, erro: String(erroVideo.message || erroVideo) });
  }

  sheet.appendRow([
    id,
    acTexto_(body.nome || ''),
    acTexto_(body.categoria || ''),
    acTexto_(body.genero || ''),
    body.preco || '',
    fotoUrl,
    'Ativo',
    new Date(),
    acTexto_(body.descricao || ''),
    estoqueVal,
    acTexto_(body.tamanhos || ''),
    acTexto_(body.cores || ''),
    !!body.novidade,
    codigo,
    fotoStoriesUrl
  ]);
  sheet.getRange(sheet.getLastRow(), colunaGaleria_(sheet)).setValue(fotosGaleriaFinal);

  if (videoUrl) gravarVideo_(sheet, sheet.getLastRow(), videoUrl);

  registrarHistorico_('criar', { id: id, codigo: codigo, nome: body.nome, preco: body.preco, estoque: estoqueVal }, null, body.funcionario);

  // Push automático de "novidade" — nunca deve impedir o cadastro do produto se falhar.
  if (body.novidade && body.notificarPush) {
    try {
      pushEnviar_({
        titulo: 'Chegou novidade! 🆕',
        mensagem: body.nome ? ('Acabou de chegar: ' + body.nome) : 'Uma peça nova chegou na loja. Vem ver!',
        url: SITE_URL + (codigo ? '?c=' + encodeURIComponent(codigo) : '?id=' + id),
        imagem: /^https:\/\//.test(String(fotoUrl || '')) ? String(fotoUrl) : '',
        audiencia: { tipo: 'todos' }, quem: body.funcionario
      });
    } catch (erroPush) {
      // Silencioso de propósito: falha no push não pode derrubar o cadastro.
    }
  }

  return jsonResponse_({ ok: true, id: id, codigo: codigo, fotoUrl: fotoUrl, fotoStoriesUrl: fotoStoriesUrl, fotosGaleria: fotosGaleriaFinal, videoUrl: videoUrl });
}

function atualizarProduto_(body) {
  if (body.preco !== undefined && body.preco !== '') {
    const precoNum = Number(body.preco);
    if (isNaN(precoNum) || precoNum <= 0) {
      return jsonResponse_({ ok: false, erro: 'Preço inválido — informe um valor maior que zero.' });
    }
  }

  const sheet = getProdutosSheet_();
  garantirCabecalhos_(sheet);
  const dados = sheet.getDataRange().getValues();

  for (let i = 1; i < dados.length; i++) {
    if (String(dados[i][0]) === String(body.id)) {
      const linha = i + 1;
      let fotoUrl = dados[i][5];
      let fotoStoriesUrl = dados[i][14];
      const colGal = colunaGaleria_(sheet);
      let galeriaAtual = dados[i][colGal - 1] || '';
      const fotoUrlAntiga = fotoUrl;
      const fotoStoriesUrlAntiga = fotoStoriesUrl;

      if (body.fotoBase64) {
        const sufixo = '_' + new Date().getTime();
        fotoUrl = salvarFotoCloudinary_(body.fotoBase64, body.fotoMime || 'image/jpeg', 'produto_' + body.id + sufixo + '.jpg');
        excluirMidiaAntiga_(fotoUrlAntiga, fotoUrl);
      } else if (body.fotoUrlExistente !== undefined && body.fotoUrlExistente && body.fotoUrlExistente !== fotoUrlAntiga) {
        // Capa trocada por uma foto que já existia (ex.: virou capa uma das fotos da galeria) --
        // só aponta pra ela, sem apagar nada do Cloudinary: ela pode ter só mudado de lugar pra
        // dentro da galeria, não foi removida.
        fotoUrl = body.fotoUrlExistente;
      }
      if (body.fotoStoriesBase64) {
        const sufixo2 = '_' + new Date().getTime();
        fotoStoriesUrl = salvarFotoCloudinary_(body.fotoStoriesBase64, 'image/jpeg', 'produto_' + body.id + sufixo2 + '_stories.jpg');
        excluirMidiaAntiga_(fotoStoriesUrlAntiga, fotoStoriesUrl);
      }

      const novasGaleria = processarGaleriaFotos_(body.fotosGaleriaBase64, body.id);
      let fotosGaleriaFinal = galeriaAtual;
      if (body.fotosGaleriaExistentes !== undefined) {
        fotosGaleriaFinal = body.fotosGaleriaExistentes + (novasGaleria ? ',' + novasGaleria : '');
      } else if (novasGaleria) {
        fotosGaleriaFinal = galeriaAtual ? (galeriaAtual + ',' + novasGaleria) : novasGaleria;
      }

      fotosGaleriaFinal = limitarGaleria_(fotosGaleriaFinal);
      let videoUrlNovo = null; // null = não mexe no vídeo atual
      try {
        if (body.videoBase64) {
          const videoUrlAntigo = dados[i][colunaVideo_(sheet) - 1] || '';
          videoUrlNovo = salvarVideoCloudinary_(body.videoBase64, body.videoMime, 'produto_' + body.id + '_video_' + new Date().getTime());
          excluirMidiaAntiga_(videoUrlAntigo, videoUrlNovo);
        } else if (body.videoUrlExistente !== undefined && body.videoUrlExistente) {
          // v2.4: vídeo novo subiu direto do navegador pro Cloudinary (ver assinarUploadVideo_) --
          // aqui só aponta pra URL que já está lá e apaga a antiga (não existe "galeria" de vídeo,
          // então ao contrário da capa de foto, aqui é sempre uma substituição de verdade).
          const videoUrlAntigoDireto = dados[i][colunaVideo_(sheet) - 1] || '';
          videoUrlNovo = body.videoUrlExistente;
          if (videoUrlNovo !== videoUrlAntigoDireto) excluirMidiaAntiga_(videoUrlAntigoDireto, videoUrlNovo);
        } else if (body.removerVideo === true) {
          const videoUrlAntigo2 = dados[i][colunaVideo_(sheet) - 1] || '';
          videoUrlNovo = '';
          excluirMidiaAntiga_(videoUrlAntigo2, videoUrlNovo);
        }
      } catch (erroVideo) {
        return jsonResponse_({ ok: false, erro: String(erroVideo.message || erroVideo) });
      }
      const anterior = { preco: dados[i][4], estoque: dados[i][9] };
      const estoqueNovo = body.estoque !== undefined && body.estoque !== '' ? Number(body.estoque) : dados[i][9];

      sheet.getRange(linha, 2, 1, 14).setValues([[
        body.nome !== undefined ? acTexto_(body.nome) : dados[i][1],
        body.categoria !== undefined ? acTexto_(body.categoria) : dados[i][2],
        body.genero !== undefined ? acTexto_(body.genero) : dados[i][3],
        body.preco !== undefined ? body.preco : dados[i][4],
        fotoUrl,
        dados[i][6],
        dados[i][7],
        body.descricao !== undefined ? acTexto_(body.descricao) : dados[i][8],
        estoqueNovo,
        body.tamanhos !== undefined ? acTexto_(body.tamanhos) : dados[i][10],
        body.cores !== undefined ? acTexto_(body.cores) : dados[i][11],
        body.novidade !== undefined ? !!body.novidade : dados[i][12],
        dados[i][13],
        fotoStoriesUrl
      ]]);
      sheet.getRange(linha, colGal).setValue(fotosGaleriaFinal);

      registrarHistorico_('editar',
        { id: body.id, codigo: dados[i][13], nome: body.nome !== undefined ? body.nome : dados[i][1], preco: body.preco !== undefined ? body.preco : dados[i][4], estoque: estoqueNovo },
        anterior, body.funcionario);

      if (videoUrlNovo !== null) gravarVideo_(sheet, linha, videoUrlNovo);
      return jsonResponse_({ ok: true, id: body.id, fotoUrl: fotoUrl, fotoStoriesUrl: fotoStoriesUrl, fotosGaleria: fotosGaleriaFinal, videoUrl: videoUrlNovo });
    }
  }
  return jsonResponse_({ ok: false, erro: 'Produto não encontrado.' });
}

function acaoEmLote_(body) {
  const ids = body.ids;
  if (!ids || !Array.isArray(ids) || ids.length === 0) return jsonResponse_({ ok: false, erro: 'Nenhum produto selecionado.' });

  const sheet = getProdutosSheet_();
  const dados = sheet.getDataRange().getValues();
  let afetados = 0;
  let categoriaLote = '';
  if (body.tipoAcao === 'alterarCategoria') {
    categoriaLote = categoriaExistenteOuNova_(body.valor);
    if (!categoriaLote) return jsonResponse_({ ok: false, erro: 'Escolha a categoria.' });
  }

  for (let i = 1; i < dados.length; i++) {
    const pid = String(dados[i][0]);
    if (ids.indexOf(pid) !== -1) {
      const linha = i + 1;
      let precoAtual = Number(dados[i][4]) || 0;

      if (body.tipoAcao === 'ajustarPrecoPorcento') {
        const pct = Number(body.valor);
        if (isNaN(pct)) continue;
        const novoPreco = Math.max(0, precoAtual * (1 + pct / 100));
        sheet.getRange(linha, 5).setValue(novoPreco.toFixed(2));
      } else if (body.tipoAcao === 'marcarNovidade') {
        sheet.getRange(linha, 13).setValue(!!body.valor);
      } else if (body.tipoAcao === 'alterarEstoque') {
        const estoqueVal = Number(body.valor);
        if (isNaN(estoqueVal)) continue;
        sheet.getRange(linha, 10).setValue(estoqueVal);
      } else if (body.tipoAcao === 'alterarCategoria') {
        sheet.getRange(linha, 3).setValue(acTexto_(categoriaLote));
      } else if (body.tipoAcao === 'arquivar') {
        sheet.getRange(linha, 7).setValue('Inativo');
      } else if (body.tipoAcao === 'reativar') {
        sheet.getRange(linha, 7).setValue('Ativo');
      }

      registrarAudit_(body.funcionario, 'bulk_' + body.tipoAcao, pid, { valor: body.valor });
      afetados++;
    }
  }

  return jsonResponse_({ ok: true, afetados: afetados });
}

/**
 * Gera título e descrição otimizados para busca usando a API do Gemini.
 * A chave fica só no servidor (Script Properties), nunca é enviada ao
 * navegador do cliente/atendente — por isso essa geração tem que passar
 * por aqui e não pode ser feita direto no admin.html.
 *
 * CONFIGURAÇÃO NECESSÁRIA (uma vez só):
 * 1. Gere uma chave gratuita em https://aistudio.google.com/apikey
 * 2. No editor do Apps Script: ⚙️ Configurações do projeto → Propriedades
 *    do script → Adicionar propriedade do script:
 *    Nome: GEMINI_API_KEY   Valor: <sua chave>
 * (Opcional) Nome: GEMINI_MODEL   Valor: gemini-2.0-flash (ou outro modelo)
 */
function gerarSeoComGemini_(body) {
  const prompt = 'Você é especialista em SEO para e-commerce de moda no Brasil. ' +
    'Gere um título curto (máximo 60 caracteres) e uma descrição (2 a 3 frases, até 300 caracteres), ' +
    'otimizados para busca (Google e redes sociais), para este produto de roupa:\n' +
    'Nome atual: ' + (body.nome || '') + '\n' +
    'Categoria: ' + (body.categoria || '') + '\n' +
    'Gênero: ' + (body.genero || '') + '\n' +
    (body.descricao ? ('Descrição atual: ' + body.descricao + '\n') : '') +
    'Responda SOMENTE em JSON válido, sem markdown, no formato exato: {"titulo": "...", "descricao": "..."}';
  const r = chamarGemini_({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: 'application/json', temperature: 0.4 } });
  // v3.1: sem chave, ou se a IA falhar, o admin ainda recebe um texto pronto (montado só com o que a peça já tem -- não inventa nada)
  if (!r.ok) { const local = descricaoLocal_(body); local.aviso = r.erro; return jsonResponse_(local); }
  let resultado;
  try { resultado = JSON.parse(r.texto); } catch (erroParse) { const local = descricaoLocal_(body); local.aviso = 'A IA respondeu num formato inesperado.'; return jsonResponse_(local); }
  return jsonResponse_({ ok: true, titulo: resultado.titulo || '', descricao: resultado.descricao || '' });
}

/**
 * v2.6 -- chave e modelo do Gemini ficam nas Propriedades do script, mas agora se configuram
 * pelo próprio admin (⚙️ Configurações Avançadas), sem abrir o editor do Apps Script. A chave
 * NUNCA volta pro navegador (só "configurada ✓" + últimos 4 caracteres) e nunca vai pro GitHub.
 * Modelo padrão: gemini-3.5-flash, com troca automática se falhar (o 2.0 foi desligado e o 3.8 vive sobrecarregado).
 */
const MODELO_IA_PADRAO_ = 'gemini-3.5-flash';
// Se o modelo escolhido falhar (sobrecarga, desligado, limite), tenta estes em ordem -- máx. 3 tentativas.
const MODELOS_IA_RESERVA_ = ['gemini-3.5-flash', 'gemini-3.7-flash', 'gemini-3.6-flash', 'gemini-3.1-flash-lite'];
function configIA_() {
  const props = PropertiesService.getScriptProperties();
  let modelo = String(props.getProperty('GEMINI_MODEL') || '').trim();
  // gemini-3.8-flash só chegou aqui porque a tela da v2.6 pré-preenchia o campo "Modelo" com o
  // padrão da época e salvava junto com a chave; ele estava sobrecarregado (demorava ~70 s para
  // dar erro 503). Tratado como "não escolhi nenhum" -- vale o padrão atual.
  if (modelo === 'gemini-3.8-flash' || modelo === 'gemini-2.0-flash') modelo = '';
  return { chave: String(props.getProperty('GEMINI_API_KEY') || '').trim(), modelo: modelo || MODELO_IA_PADRAO_ };
}
/** Chama o Gemini com troca automática de modelo. Devolve { ok, texto, modelo } ou { ok:false, erro }. */
function chamarGemini_(payload) {
  const cfg = configIA_();
  if (!cfg.chave) return { ok: false, erro: 'Chave do Gemini não configurada. No admin, abra ⚙️ Configurações Avançadas e cole a chave em "Inteligência artificial".' };
  const fila = [cfg.modelo].concat(MODELOS_IA_RESERVA_.filter(function (m) { return m !== cfg.modelo; })).slice(0, 3);
  let ultimoErro = '';
  for (let i = 0; i < fila.length; i++) {
    try {
      const resp = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + fila[i] + ':generateContent?key=' + cfg.chave,
        { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true });
      const codigo = resp.getResponseCode();
      let dados = {};
      try { dados = JSON.parse(resp.getContentText()); } catch (e) { dados = {}; }
      if (codigo === 200) {
        const partes = (dados.candidates && dados.candidates[0] && dados.candidates[0].content && dados.candidates[0].content.parts) || [];
        const texto = partes.map(function (x) { return x.text || ''; }).join('').replace(/```json/gi, '').replace(/```/g, '').trim();
        if (texto) return { ok: true, texto: texto, modelo: fila[i] };
        ultimoErro = 'resposta vazia (' + fila[i] + ')';
        continue;
      }
      ultimoErro = (dados.error && dados.error.message ? dados.error.message : ('HTTP ' + codigo)) + ' (' + fila[i] + ')';
      if (codigo === 400 || codigo === 401 || codigo === 403) break; // chave/pedido inválido: trocar de modelo não resolve
    } catch (e) { ultimoErro = String(e.message || e) + ' (' + fila[i] + ')'; }
  }
  return { ok: false, erro: 'Erro do Gemini: ' + ultimoErro };
}
function statusConfigIA_() {
  const c = configIA_();
  return jsonResponse_({ ok: true, temChave: !!c.chave, fim: c.chave ? c.chave.slice(-4) : '', modelo: c.modelo, modeloPadrao: MODELO_IA_PADRAO_ });

function salvarConfigIA_(body) {
  const props = PropertiesService.getScriptProperties();
  if (body.apagarChave === true) props.deleteProperty('GEMINI_API_KEY');
  const chave = String(body.chave || '').trim();
  if (chave) {
    if (!/^[A-Za-z0-9._\-]{20,200}$/.test(chave)) return jsonResponse_({ ok: false, erro: 'Essa chave parece incompleta ou com caracteres estranhos. Cole de novo, só a chave.' });
    props.setProperty('GEMINI_API_KEY', chave);
  }
  if (typeof body.modelo === 'string') {
    const modelo = body.modelo.trim();
    if (!modelo) props.deleteProperty('GEMINI_MODEL');
    else if (!/^[A-Za-z0-9._\-]{3,80}$/.test(modelo)) return jsonResponse_({ ok: false, erro: 'Nome de modelo inválido.' });
    else props.setProperty('GEMINI_MODEL', modelo);
  }
  return statusConfigIA_();
}

/**
 * v2.5 -- "Preencher pela foto": olha TODAS as fotos da peça (até 8) e sugere nome, gênero,
 * categoria, cores e descrição. NUNCA sugere preço, estoque nem tamanhos (não dá pra saber
 * pela foto). Gênero e categoria só podem ser valores que já existem nas listas da loja
 * (o navegador manda as listas; o que a IA devolver fora delas é descartado aqui).
 * Usa a mesma chave GEMINI_API_KEY / GEMINI_MODEL do gerarSeo -- nada novo pra configurar.
 * body.fotos = [{ b64: '<jpeg base64 puro>' } | { url: 'https://res.cloudinary.com/...' }]
 */
function sugerirPorFotoComGemini_(body) {

  const fotos = Array.isArray(body.fotos) ? body.fotos.slice(0, 8) : [];
  if (!fotos.length) return jsonResponse_({ ok: false, erro: 'Adicione ao menos uma foto antes.' });
  const listaCategorias = (Array.isArray(body.categorias) ? body.categorias : []).map(String).slice(0, 80);
  const listaGeneros = (Array.isArray(body.generos) ? body.generos : []).map(String).slice(0, 20);

  const partesImagem = [];
  fotos.forEach(function (f) {
    try {
      if (f && f.b64) {
        partesImagem.push({ inline_data: { mime_type: 'image/jpeg', data: String(f.b64) } });
      } else if (f && /^https:\/\/res\.cloudinary\.com\//.test(String(f.url || ''))) {
        // reduz no próprio Cloudinary antes de baixar (rápido e barato); só aceita o nosso domínio
        const pequena = String(f.url).replace('/upload/', '/upload/w_768,q_auto,f_jpg/');
        const r = UrlFetchApp.fetch(pequena, { muteHttpExceptions: true });
        if (r.getResponseCode() === 200) {
          partesImagem.push({ inline_data: { mime_type: 'image/jpeg', data: Utilities.base64Encode(r.getContent()) } });
        }
      }
    } catch (e) { /* foto que não deu pra ler é só ignorada; as outras seguem */ }
  });
  if (!partesImagem.length) return jsonResponse_({ ok: false, erro: 'Não consegui ler nenhuma das fotos. Tente de novo.' });

  const prompt = 'Você cadastra roupas no catálogo de uma loja de moda brasileira. As imagens são fotos DA MESMA PEÇA ' +
    '(ângulos diferentes). Olhe todas e responda SOMENTE em JSON válido, sem markdown, no formato exato: ' +
    '{"nome": "...", "generos": [...], "categorias": [...], "cores": [...], "descricao": "..."}\n' +
    'Regras:\n' +
    '- nome: curto e claro, como "Vestido Floral Midi" ou "Camiseta Oversized Preta" (máx. 60 caracteres).\n' +
    '- generos: escolha SOMENTE entre: ' + JSON.stringify(listaGeneros) + ' (vazio [] se nenhum servir).\n' +
    '- categorias: escolha SOMENTE entre: ' + JSON.stringify(listaCategorias) + ' (vazio [] se nenhuma servir).\n' +
    '- cores: cores que você VÊ na peça, em português, ex. ["Preto", "Off-white"].\n' +
    '- descricao: 1 a 2 frases só com o que é visível (modelagem, estampa, detalhes). NÃO invente tecido, marca, ' +
    'composição, tamanho nem preço. Se não dá pra saber pela foto, não mencione.\n' +
    'Nunca inclua preço.';

  const resp = chamarGemini_({
    contents: [{ parts: [{ text: prompt }].concat(partesImagem) }],
    generationConfig: { responseMimeType: 'application/json', temperature: 0.3 }
  });
  if (!resp.ok) return jsonResponse_({ ok: false, erro: resp.erro });
  try {
    let r;
    try { r = JSON.parse(resp.texto); } catch (e) { return jsonResponse_({ ok: false, erro: 'A IA respondeu num formato inesperado. Tente novamente.' }); }

    function soDaLista(valores, permitidos) {
      const mapa = {};
      permitidos.forEach(function (p) { mapa[p.toLowerCase()] = p; });
      const out = [];
      (Array.isArray(valores) ? valores : []).forEach(function (v) {
        const achado = mapa[String(v).toLowerCase().trim()];
        if (achado && out.indexOf(achado) === -1) out.push(achado);
      });
      return out;
    }
    return jsonResponse_({
      ok: true,
      fotosAnalisadas: partesImagem.length,
      modelo: resp.modelo,
      nome: String(r.nome || '').slice(0, 80),
      generos: soDaLista(r.generos, listaGeneros),
      categorias: soDaLista(r.categorias, listaCategorias),
      cores: (Array.isArray(r.cores) ? r.cores : []).map(function (c) { return String(c).trim(); }).filter(Boolean).slice(0, 8),
      descricao: String(r.descricao || '').slice(0, 400)
    });
  } catch (erroIA) {
    return jsonResponse_({ ok: false, erro: 'Falha ao processar a resposta da IA: ' + erroIA.message });
  }
}

/** Chamada à API do OneSignal. Chave nova (os_v2_...) usa "Key", chave antiga usa "Basic"; se o esquema for recusado (401/403), tenta o outro. */
function pushChamar_(metodo, caminho, corpo) {
  const cfg = pushCfg_();
  const esquemas = /^os_v2_/.test(cfg.chave) ? ['Key', 'Basic'] : ['Basic', 'Key'];
  let ultimo = null;
  for (let i = 0; i < esquemas.length; i++) {
    const opt = { method: metodo, contentType: 'application/json', headers: { 'Authorization': esquemas[i] + ' ' + cfg.chave, 'Accept': 'application/json' }, muteHttpExceptions: true };
    if (corpo) opt.payload = JSON.stringify(corpo);
    const r = UrlFetchApp.fetch(PUSH_API_ + caminho, opt);
    const texto = String(r.getContentText() || '');
    ultimo = { codigo: r.getResponseCode(), dados: jsonSeguro_(texto, {}), texto: texto };
    if (ultimo.codigo !== 401 && ultimo.codigo !== 403) return ultimo;
  }
  return ultimo;
}
function pushRequisicao_(corpo) {
  const cfg = pushCfg_();
  return { url: PUSH_API_ + '/notifications', method: 'post', contentType: 'application/json', headers: { 'Authorization': (/^os_v2_/.test(cfg.chave) ? 'Key ' : 'Basic ') + cfg.chave, 'Accept': 'application/json' }, payload: JSON.stringify(corpo), muteHttpExceptions: true };
}
/** Texto de erro legível a partir da resposta do OneSignal. */
function pushErroTexto_(r) {
  const e = r && r.dados && r.dados.errors;
  let t = '';
  if (Array.isArray(e)) t = e.map(function (x) { return typeof x === 'string' ? x : (x && (x.title || x.message)) || JSON.stringify(x); }).join('; ');
  else if (e && typeof e === 'object') t = Object.keys(e).map(function (k) { return k + ': ' + JSON.stringify(e[k]); }).join('; ');
  if (!t) t = 'HTTP ' + (r ? r.codigo : '?');
  if (/not configured for web|no.*web.*platform/i.test(t)) t += ' (a plataforma Web ainda não foi configurada no OneSignal)';
  return t;
}
/** ids de alias que o OneSignal diz que não têm aparelho com aviso ativo. */
function pushInvalidos_(dados) {

  const out = {};

    Object.keys(e.invalid_aliases).forEach(function (rotulo) { (e.invalid_aliases[rotulo] || []).forEach(function (id) { out[String(id)] = true; }); });
  }
  return out;
}

/** Uma entrada por WhatsApp (contas ativas), com grupo e permissões. Quem tem o mesmo número em mais de uma linha junta as permissões. */
function pushPessoas_() {
  const perfis = acPerfis_();
  const mapa = {};
  acTodasLinhas_().forEach(function (l) {
    const w = acWhats_(l.whatsapp);
    if (!w) return;
    if (String(l.status || 'Ativo').trim().toLowerCase() === 'inativo') return;
    const p = acPermsDaLinha_(l, perfis);
    const perfil = String(l.perfil || '').trim() || acPerfilLegado_(l.nivel);
    let x = mapa[w];
    if (!x) x = mapa[w] = { whats: w, nome: String(l.nome || '').trim() || w, perfis: [], perms: {} };
    if (String(l.nome || '').trim() && x.nome === w) x.nome = String(l.nome).trim();
    if (x.perfis.indexOf(perfil) === -1) x.perfis.push(perfil);
    acChaves_().forEach(function (k) { if (p[k]) x.perms[k] = true; });
  });
  return Object.keys(mapa).map(function (w) {
    const x = mapa[w];
    const perfisNorm = x.perfis.map(acNorm_);
    x.perfil = x.perfis[0] || '';
    x.equipe = !!(x.perms.catalogo_admin || x.perms.catalogo_cadastrar || x.perms.gerir_acessos || x.perms.gerir_manuais || x.perms.equipe_relatorio);
    x.portal = !!x.perms.portal;
    x.aluno = perfisNorm.indexOf('aluno') !== -1 || perfisNorm.indexOf('novato') !== -1;
    x.cliente = !x.equipe && !x.portal;
    x.pushId = pushIdDe_(w);
    return x;
  });
}

function pushRamos_(cond, respeitar) {
  if (!respeitar) return cond.slice();
  return cond.concat([{ field: 'tag', key: 'av_novidades', relation: '=', value: '1' }, { operator: 'OR' }]).concat(cond).concat([{ field: 'tag', key: 'av_novidades', relation: 'not_exists' }]);
}
function pushJuntarRamos_(alternativas, respeitar) {
  let out = [];
  alternativas.forEach(function (c, i) { if (i) out.push({ operator: 'OR' }); out = out.concat(pushRamos_(c, respeitar)); });
  return out;
}
function pushItemPessoa_(x) {
  return { nome: x.nome, ids: { external_id: [x.pushId] } };
}
/** Cada pedido tem o PRÓPRIO apelido no OneSignal ("ped_0001" = chave do pedido): um aparelho pode acompanhar vários pedidos ao mesmo tempo
 *  (o OneSignal só aceita UM valor por apelido, então um apelido único como "pedido" só guardaria o último). */
function pushRotuloPedido_(codigo) { return 'ped_' + String(codigo).replace(/\D/g, ''); }
function pushItemPedido_(o) {
  const w = acWhats_(o.WhatsApp);
  const ids = {};
  ids[pushRotuloPedido_(o.Codigo)] = [pushChavePedido_(String(o.Codigo))];
  if (acLinhasDoWhats_(w).length) ids.external_id = [pushIdDe_(w)];
  return { nome: String(o.Nome || ''), ids: ids };
}

/**
 * Devolve { ok, tipo: 'aliases'|'filtros'|'segmentos', descricao, itens?, filtros?, segmentos? } ou { ok:false, erro }.
 * 'aliases' = lista de pessoas com os ids que o OneSignal conhece (mensagem direta); 'filtros' = etiquetas do aparelho; 'segmentos' = segmentos do OneSignal.
 */
function pushAudiencia_(aud, respeitar) {
  aud = aud && typeof aud === 'object' ? aud : { tipo: 'todos' };
  const tipo = String(aud.tipo || 'todos');

  if (tipo === 'todos') {
    return respeitar
      ? { ok: true, tipo: 'filtros', filtros: pushJuntarRamos_([[]], true), descricao: 'Todos que ativaram os avisos (respeitando quem desligou promoções)' }
      : { ok: true, tipo: 'segmentos', segmentos: ['Subscribed Users'], descricao: 'Todos que ativaram os avisos' };
  }
  if (tipo === 'segmento') {
    const permitidos = ['Subscribed Users'].concat(String(PropertiesService.getScriptProperties().getProperty('PUSH_SEGMENTOS') || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean));
    const nome = String(aud.nome || 'Subscribed Users');
    if (permitidos.indexOf(nome) === -1) return { ok: false, erro: 'Esse segmento não está cadastrado (Integrações → segmentos).' };
    return { ok: true, tipo: 'segmentos', segmentos: [nome], descricao: nome === 'Subscribed Users' ? 'Todos que ativaram os avisos' : 'Segmento: ' + nome };
  }
  if (tipo === 'app') {
    return { ok: true, tipo: 'filtros', filtros: pushJuntarRamos_([[{ field: 'tag', key: 'origem', relation: '=', value: 'app' }]], respeitar), descricao: 'Quem instalou o app' };
  }
  if (tipo === 'interesse') {
    const chaves = (Array.isArray(aud.chaves) ? aud.chaves : [aud.chave]).map(pushSlug_).filter(Boolean).slice(0, 10);
    if (!chaves.length) return { ok: false, erro: 'Escolha pelo menos um interesse.' };
    const rotulos = (Array.isArray(aud.rotulos) ? aud.rotulos : chaves).map(function (r) { return String(r).slice(0, 40); });
    return { ok: true, tipo: 'filtros', filtros: pushJuntarRamos_(chaves.map(function (c) { return [{ field: 'tag', key: 'int_' + c, relation: '=', value: '1' }]; }), respeitar), descricao: 'Interessados em: ' + rotulos.join(', ') };
  }

  // ---- por conta (mensagem direta)
  let itens = [], descricao = '';
  if (tipo === 'pedido') {
    const cod = codigoPedidoNormalizado_(aud.codigo);
    const o = cod ? pedidoPorCodigo_(cod) : null;
    if (!o) return { ok: false, erro: 'Pedido não encontrado.' };
    itens = [pushItemPedido_(o)]; descricao = 'Cliente do pedido ' + cod;
  } else if (tipo === 'pedidos') {
    const st = String(aud.status || '');
    if (STATUS_PEDIDO_.indexOf(st) === -1) return { ok: false, erro: 'Etapa de pedido inválida.' };
    const vistos = {};
    linhasComoObjetos_(getOrCreateSheet_(SHEET_PEDIDOS, CABECALHOS_PEDIDOS_)).forEach(function (o) {
      if (String(o.Status || 'novo') !== st) return;
      const it = pushItemPedido_(o);
      const k = (it.ids.external_id || [])[0] || JSON.stringify(it.ids);
      if (!vistos[k]) { vistos[k] = true; itens.push(it); }
    });
    descricao = 'Clientes com pedido em "' + st + '"';
  } else {
    const pessoas = pushPessoas_();
    let sel = [];
    if (tipo === 'grupo') {
      const g = PUSH_GRUPOS_[String(aud.grupo || '')];
      if (!g) return { ok: false, erro: 'Grupo desconhecido.' };
      sel = pessoas.filter(g.teste); descricao = g.rotulo;
    } else if (tipo === 'perfil') {
      const nomes = (Array.isArray(aud.perfis) ? aud.perfis : [aud.perfil]).map(acNorm_).filter(Boolean);
      if (!nomes.length) return { ok: false, erro: 'Escolha pelo menos um perfil.' };
      sel = pessoas.filter(function (x) { return x.perfis.some(function (p) { return nomes.indexOf(acNorm_(p)) !== -1; }); });
      descricao = 'Perfil: ' + (Array.isArray(aud.perfis) ? aud.perfis : [aud.perfil]).join(', ');
    } else if (tipo === 'acesso') {
      const validas = acChaves_();
      const chaves = (Array.isArray(aud.chaves) ? aud.chaves : [aud.chave]).filter(function (k) { return validas.indexOf(k) !== -1; });
      if (!chaves.length) return { ok: false, erro: 'Escolha pelo menos um acesso.' };
      const todas = aud.modo === 'todas';
      sel = pessoas.filter(function (x) { return todas ? chaves.every(function (k) { return x.perms[k]; }) : chaves.some(function (k) { return x.perms[k]; }); });
      const rot = {}; AC_PERMISSOES.forEach(function (p) { rot[p.chave] = p.rotulo; });
      descricao = 'Acesso: ' + chaves.map(function (k) { return rot[k]; }).join(todas ? ' + ' : ' ou ');
    } else if (tipo === 'pessoas') {
      const ids = (Array.isArray(aud.ids) ? aud.ids : []).map(String);
      if (!ids.length) return { ok: false, erro: 'Escolha pelo menos uma pessoa.' };
      sel = pessoas.filter(function (x) { return ids.indexOf(x.pushId) !== -1; });
      descricao = sel.length === 1 ? 'Pessoa: ' + sel[0].nome : 'Pessoas escolhidas (' + sel.length + ')';
    } else {
      return { ok: false, erro: 'Público desconhecido: ' + tipo };
    }
    itens = sel.map(pushItemPessoa_);
  }
  if (!itens.length) return { ok: false, erro: 'Ninguém encontrado nesse público.' };
  return { ok: true, tipo: 'aliases', itens: itens, descricao: descricao };
}

// ---------------------------------------------------------------- enviar

/**
 * Envia (ou agenda) um aviso. op: { titulo, mensagem, url, imagem, audiencia, agendarEm (ISO), ttlHoras, quem, respeitarPreferencias, semHistorico }.
 * Devolve objeto simples (nunca lança): { ok, id, agendado, descricao, pessoas, semAviso:[nomes], individuais, erro }.
 */
function pushEnviar_(op) {
  const cfg = pushCfg_();
  if (!cfg.appId || !cfg.chave) return { ok: false, erro: 'OneSignal não configurado no servidor (faltam ONESIGNAL_APP_ID e/ou ONESIGNAL_REST_API_KEY nas Propriedades do Script). Veja LEIA-ME-AVISOS.md.' };
  const titulo = String(op.titulo == null ? '' : op.titulo).trim();
  const mensagem = String(op.mensagem == null ? '' : op.mensagem).trim();
  if (!titulo || !mensagem) return { ok: false, erro: 'Título e mensagem são obrigatórios para enviar a notificação.' };
  if (titulo.length > 80 || mensagem.length > 300) return { ok: false, erro: 'Título: até 80 letras. Mensagem: até 300.' };
  const url = String(op.url || SITE_URL).trim();
  if (!/^https:\/\//.test(url)) return { ok: false, erro: 'O link do aviso precisa começar com https://' };
  const imagem = String(op.imagem || '').trim();
  if (imagem && !/^https:\/\//.test(imagem)) return { ok: false, erro: 'A imagem do aviso precisa ser um endereço https://' };

  let sendAfter = '';
  if (op.agendarEm) {
    const d = new Date(op.agendarEm);
    if (isNaN(d.getTime())) return { ok: false, erro: 'Data e hora do agendamento inválidas.' };
    const agora = Date.now();
    if (d.getTime() < agora + 60000) return { ok: false, erro: 'O horário agendado precisa estar no futuro.' };
    if (d.getTime() > agora + 30 * 86400000) return { ok: false, erro: 'Agende no máximo para daqui a 30 dias.' };
    sendAfter = d.toISOString();
  }

  const alvo = pushAudiencia_(op.audiencia, op.respeitarPreferencias !== false);
  if (!alvo.ok) return alvo;

  const base = { app_id: cfg.appId, url: url };
  if (imagem) { base.chrome_web_image = imagem; base.big_picture = imagem; }
  const ttl = Number(op.ttlHoras);
  if (ttl > 0 && ttl <= 24 * 28) base.ttl = Math.round(ttl * 3600);
  if (sendAfter) base.send_after = sendAfter;

  const individual = alvo.tipo === 'aliases' && (pushTemNome_(titulo) || pushTemNome_(mensagem));
  if (individual && alvo.itens.length > PUSH_LIMITE_PERSONALIZADO_) {
    return { ok: false, erro: 'Com {nome} o aviso vai um a um e o limite é ' + PUSH_LIMITE_PERSONALIZADO_ + ' pessoas por envio (este público tem ' + alvo.itens.length + '). Tire o {nome} ou escolha um público menor.' };
  }

  const res = { ok: true, id: '', agendado: !!sendAfter, descricao: alvo.descricao, pessoas: alvo.tipo === 'aliases' ? alvo.itens.length : null, semAviso: [], individuais: individual, enviados: 0 };
  let erroApi = '';

  if (individual) {
    const pedidos = alvo.itens.map(function (it) {
      const corpo = Object.assign({}, base, { target_channel: 'push', include_aliases: it.ids });
      const nome = pushPrimeiroNome_(it.nome) || 'cliente';
      corpo.headings = { en: pushTexto_(titulo, nome), pt: pushTexto_(titulo, nome) };
      corpo.contents = { en: pushTexto_(mensagem, nome), pt: pushTexto_(mensagem, nome) };
      return pushRequisicao_(corpo);
    });
    const ids = [];
    for (let i = 0; i < pedidos.length; i += 40) {
      let resp;
      try { resp = UrlFetchApp.fetchAll(pedidos.slice(i, i + 40)); } catch (e) { erroApi = 'Falha ao conectar com o OneSignal: ' + e.message; break; }
      resp.forEach(function (r, j) {
        const dados = jsonSeguro_(r.getContentText(), {});
        const it = alvo.itens[i + j];
        if (r.getResponseCode() === 200 && dados.id) { res.enviados++; ids.push(dados.id); }
        else {
          if (r.getResponseCode() !== 200 && r.getResponseCode() !== 400) erroApi = pushErroTexto_({ codigo: r.getResponseCode(), dados: dados });
          res.semAviso.push(it.nome || 'Sem nome');
        }
      });
    }
    res.id = ids[0] || '';
    res.ids = ids;
    if (!res.enviados) return { ok: false, erro: erroApi ? 'Erro do OneSignal: ' + erroApi : 'Ninguém desse público tem avisos ativados ainda (a pessoa precisa entrar na conta e ativar os avisos no aparelho).', descricao: alvo.descricao, semAviso: res.semAviso };
  } else {
    const corpo = Object.assign({}, base, { headings: { en: pushTexto_(titulo, 'cliente'), pt: pushTexto_(titulo, 'cliente') }, contents: { en: pushTexto_(mensagem, 'cliente'), pt: pushTexto_(mensagem, 'cliente') } });
    if (alvo.tipo === 'aliases') {
      const junto = {};
      alvo.itens.forEach(function (it) { Object.keys(it.ids).forEach(function (k) { junto[k] = (junto[k] || []).concat(it.ids[k]); }); });
      corpo.target_channel = 'push'; corpo.include_aliases = junto;
    } else if (alvo.tipo === 'filtros') corpo.filters = alvo.filtros;
    else corpo.included_segments = alvo.segmentos;
    let r;
    try { r = pushChamar_('post', '/notifications', corpo); } catch (e) { return { ok: false, erro: 'Falha ao conectar com o OneSignal: ' + e.message }; }
    if (r.codigo !== 200) return { ok: false, erro: 'Erro do OneSignal: ' + pushErroTexto_(r) };
    const invalidos = pushInvalidos_(r.dados);
    if (alvo.tipo === 'aliases') {
      alvo.itens.forEach(function (it) {
        const todos = Object.keys(it.ids).reduce(function (a, rot) { return a.concat(it.ids[rot]); }, []);
        if (todos.length && todos.every(function (id) { return invalidos[id]; })) res.semAviso.push(it.nome || 'Sem nome');
      });
      res.enviados = alvo.itens.length - res.semAviso.length;
    } else if (r.dados.recipients !== undefined) res.enviados = Number(r.dados.recipients) || 0;
    res.id = String(r.dados.id || '');
    if (!res.id) return { ok: false, erro: 'Ninguém desse público tem avisos ativados ainda' + (r.dados.errors ? ' (' + pushErroTexto_(r) + ')' : '.'), descricao: alvo.descricao, semAviso: res.semAviso };
    if (alvo.tipo === 'aliases' && res.enviados <= 0) return { ok: false, erro: 'Ninguém desse público tem avisos ativados ainda (a pessoa precisa entrar na conta e ativar os avisos no aparelho).', descricao: alvo.descricao, semAviso: res.semAviso };
  }

  if (!op.semHistorico) {
    pushHistoricoGravar_({ quem: op.quem, titulo: titulo, mensagem: mensagem, url: url, segmento: alvo.tipo === 'segmentos' ? alvo.segmentos[0] : '', destinatarios: res.enviados, resultado: sendAfter ? 'agendado' : 'enviado', id: res.id, audiencia: alvo.descricao, agendado: sendAfter ? dataBR_(new Date(sendAfter)) : '', imagem: imagem });
  }
  return res;
}

/** Compatível com a v3.1/3.2: título, mensagem, link, segmento (ou "Subscribed Users"). Usado pelo aviso automático de novidade e por quem ainda manda o formato antigo. */
/** Formato antigo (só "segmento"): vazio ou "Subscribed Users" = todos (respeitando quem desligou promoções); outro nome = segmento do OneSignal. */
function pushAudienciaDeSegmento_(segmento) {
  const seg = String(segmento || '').trim();
  return !seg || seg === 'Subscribed Users' ? { tipo: 'todos' } : { tipo: 'segmento', nome: seg };
}
function enviarPushOneSignal_(titulo, mensagem, urlDestino, segmento, funcionario) {
  const r = pushEnviar_({ titulo: titulo, mensagem: mensagem, url: urlDestino, audiencia: pushAudienciaDeSegmento_(segmento), quem: funcionario });
  if (!r.ok) return jsonResponse_({ ok: false, erro: r.erro });
  return jsonResponse_({ ok: true, id: r.id, destinatarios: r.enviados || 0 });
}

function enviarPushAcao_(body) {
  const op = {
    titulo: body.titulo, mensagem: body.mensagem, url: body.url || SITE_URL, imagem: body.imagem, agendarEm: body.agendarEm, ttlHoras: body.ttlHoras,
    quem: body.funcionario, respeitarPreferencias: body.respeitarPreferencias !== false,
    audiencia: body.audiencia || pushAudienciaDeSegmento_(body.segmento)
  };
  const r = pushEnviar_(op);
  if (!r.ok) return jsonResponse_({ ok: false, erro: r.erro, semAviso: r.semAviso || [], descricao: r.descricao || '' });
  registrarAudit_(body.funcionario, 'enviar_push', '', { titulo: op.titulo, publico: r.descricao, id: r.id, agendado: r.agendado });
  return jsonResponse_({ ok: true, id: r.id, destinatarios: r.enviados || 0, agendado: r.agendado, descricao: r.descricao, pessoas: r.pessoas, semAviso: r.semAviso, individuais: r.individuais });
}

/** Resolve o público SEM enviar: quantas pessoas (quando dá para saber) e uma amostra de nomes. */
function pushPrevia_(body) {
  const alvo = pushAudiencia_(body.audiencia, body.respeitarPreferencias !== false);
  if (!alvo.ok) return jsonResponse_({ ok: false, erro: alvo.erro });
  if (alvo.tipo !== 'aliases') return jsonResponse_({ ok: true, descricao: alvo.descricao, quantidade: null, porAparelho: true, amostra: [] });
  return jsonResponse_({ ok: true, descricao: alvo.descricao, quantidade: alvo.itens.length, porAparelho: false, amostra: alvo.itens.slice(0, 8).map(function (i) { return i.nome; }), limitePersonalizado: PUSH_LIMITE_PERSONALIZADO_ });
}

function pushHistAba_() {

  const atual = aba.getRange(1, 1, 1, n).getValues()[0].map(function (x) { return String(x).trim(); });
  PUSH_CAB_HIST_.forEach(function (h) {
    if (atual.indexOf(h) === -1) { aba.getRange(1, atual.length + 1).setValue(h); atual.push(h); }
  });
  return { aba: aba, cab: atual };
}
function pushHistoricoGravar_(d) {

  } catch (e) { /* o histórico é um extra: nunca derruba o envio */ }
}

    })
  });
}

/** Entregues / cliques / falhas de UM aviso (o OneSignal calcula e leva alguns minutos para fechar a conta). */
function pushNumeros_(body) {
  const cfg = pushCfg_();
  const id = String(body.id || '').trim();
  if (!cfg.appId || !cfg.chave) return jsonResponse_({ ok: false, erro: 'OneSignal não configurado no servidor.' });
  if (!/^[A-Za-z0-9-]{8,64}$/.test(id)) return jsonResponse_({ ok: false, erro: 'Aviso inválido.' });
  let r;
  try { r = pushChamar_('get', '/notifications/' + id + '?app_id=' + encodeURIComponent(cfg.appId)); } catch (e) { return jsonResponse_({ ok: false, erro: 'Falha ao conectar com o OneSignal: ' + e.message }); }
  if (r.codigo !== 200) return jsonResponse_({ ok: false, erro: 'Erro do OneSignal: ' + pushErroTexto_(r) });
  const d = r.dados || {};
  const n = function (v) { return Number(v) || 0; };
  return jsonResponse_({ ok: true, entregues: n(d.successful), falhas: n(d.failed) + n(d.errored), cliques: n(d.converted), restantes: n(d.remaining), cancelado: !!d.canceled, enviadoEm: d.completed_at || d.queued_at || 0 });
}

function pushCancelar_(body) {
  const cfg = pushCfg_();
  const id = String(body.id || '').trim();
  if (!cfg.appId || !cfg.chave) return jsonResponse_({ ok: false, erro: 'OneSignal não configurado no servidor.' });
  if (!/^[A-Za-z0-9-]{8,64}$/.test(id)) return jsonResponse_({ ok: false, erro: 'Aviso inválido.' });
  let r;
  try { r = pushChamar_('delete', '/notifications/' + id + '?app_id=' + encodeURIComponent(cfg.appId)); } catch (e) { return jsonResponse_({ ok: false, erro: 'Falha ao conectar com o OneSignal: ' + e.message }); }
  if (r.codigo !== 200) return jsonResponse_({ ok: false, erro: 'Não deu para cancelar (talvez já tenha sido enviado): ' + pushErroTexto_(r) });
  try {
    const h = pushHistAba_();
    const iId = h.cab.indexOf('NotifId'), iRes = h.cab.indexOf('Resultado');
    const dados = h.aba.getDataRange().getValues();
    for (let i = dados.length - 1; i >= 1; i--) if (String(dados[i][iId]) === id) { h.aba.getRange(i + 1, iRes + 1).setValue('cancelado'); break; }
  } catch (e) {}
  registrarAudit_(body.funcionario, 'cancelar_push', '', { id: id });
  return jsonResponse_({ ok: true });
}

function pushModeloSalvar_(body) {
  const m = body.modelo || {};
  const rotulo = limparTexto_(m.rotulo, 30), titulo = limparTexto_(m.titulo, 80), mensagem = limparTexto_(m.mensagem, 300);
  if (!rotulo || !titulo || !mensagem) return jsonResponse_({ ok: false, erro: 'Modelo: nome, título e mensagem são obrigatórios.' });
  const url = String(m.url || '').trim(), imagem = String(m.imagem || '').trim();
  if ((url && !/^https:\/\//.test(url)) || (imagem && !/^https:\/\//.test(imagem))) return jsonResponse_({ ok: false, erro: 'Link e imagem precisam começar com https://' });
  const trava = LockService.getScriptLock();
  try { trava.waitLock(15000); } catch (e) { return jsonResponse_({ ok: false, erro: 'O sistema está ocupado. Tente de novo.' }); }
  try {
    const aba = pushModelosAba_();
    const dados = aba.getDataRange().getValues();
    const iId = dados[0].map(function (x) { return String(x).trim(); }).indexOf('ID');
    const linhaVal = function (id) { return [id, acTexto_(rotulo), acTexto_(titulo), acTexto_(mensagem), acTexto_(url), acTexto_(imagem), new Date(), acTexto_(body.funcionario || '')]; };
    const alvo = String(m.id || '');
    for (let i = 1; i < dados.length; i++) {
      if (alvo && String(dados[i][iId]) === alvo) { aba.getRange(i + 1, 1, 1, 8).setValues([linhaVal(alvo)]); return jsonResponse_({ ok: true, id: alvo, modelos: pushModelos_() }); }
    }
    const novo = 'm' + new Date().getTime().toString(36);
    aba.appendRow(linhaVal(novo));
    return jsonResponse_({ ok: true, id: novo, modelos: pushModelos_() });
  } finally { trava.releaseLock(); }
}
function pushModeloExcluir_(body) {
  const alvo = String(body.id || '');
  if (!alvo) return jsonResponse_({ ok: false, erro: 'Modelo inválido.' });
  const trava = LockService.getScriptLock();
  try { trava.waitLock(15000); } catch (e) { return jsonResponse_({ ok: false, erro: 'O sistema está ocupado. Tente de novo.' }); }
  try {
    const aba = pushModelosAba_();
    const dados = aba.getDataRange().getValues();
    for (let i = dados.length - 1; i >= 1; i--) if (String(dados[i][0]) === alvo) { aba.deleteRow(i + 1); return jsonResponse_({ ok: true, modelos: pushModelos_() }); }
    return jsonResponse_({ ok: false, erro: 'Esse modelo não existe mais.' });
  } finally { trava.releaseLock(); }
}

/** O OneSignal responde, sem chave, se a plataforma Web do app já foi configurada (sem isso nenhum aviso chega a navegador/iPhone). */
function pushWebPronto_(appId, forcar) {
  const cache = CacheService.getScriptCache();
  if (!forcar) { const c = cache.get('push_web_' + appId); if (c) return c === '1'; }
  let ok = false;
  try {
    // O OneSignal guarda cada endereço desse em cache por 1 HORA (s-maxage=3600): sem um parâmetro novo, "não configurada" continuaria aparecendo depois de ligar.
    const r = UrlFetchApp.fetch(PUSH_API_ + '/sync/' + encodeURIComponent(appId) + '/web?callback=x&_=' + new Date().getTime(), { muteHttpExceptions: true });
    const t = String(r.getContentText() || '');
    ok = /"success"\s*:\s*false/.test(t) ? false : (r.getResponseCode() === 200 && t.length > 40);
  } catch (e) { return null; }
  try { cache.put('push_web_' + appId, ok ? '1' : '0', 60); } catch (e) {}
  return ok;
}

function pushStatus_() {
  const cfg = pushCfg_();
  const props = PropertiesService.getScriptProperties();
  const pessoas = pushPessoas_();
  const contar = function (f) { return pessoas.filter(f).length; };
  const perfis = {};
  pessoas.forEach(function (x) { x.perfis.forEach(function (p) { perfis[p] = (perfis[p] || 0) + 1; }); });
  const acessos = AC_PERMISSOES.map(function (p) { return { chave: p.chave, rotulo: p.rotulo, grupo: p.grupo, total: contar(function (x) { return x.perms[p.chave]; }) }; });
  return jsonResponse_({
    ok: true,
    appId: cfg.appId, chaveConfigurada: !!cfg.chave, webPronto: cfg.appId ? pushWebPronto_(cfg.appId) : null,
    segmentos: ['Subscribed Users'].concat(String(props.getProperty('PUSH_SEGMENTOS') || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean)),
    contas: { total: pessoas.length, equipe: contar(PUSH_GRUPOS_.equipe.teste), alunos: contar(PUSH_GRUPOS_.alunos.teste), portal: contar(PUSH_GRUPOS_.portal.teste), clientes: contar(PUSH_GRUPOS_.clientes.teste) },
    perfis: Object.keys(perfis).sort().map(function (p) { return { nome: p, total: perfis[p] }; }),
    acessos: acessos,
    interesses: pushInteresses_(),
    autoNovoPedido: props.getProperty('PUSH_NOVO_PEDIDO') !== 'false', autoPedido: props.getProperty('PUSH_PEDIDO_AUTO') !== 'false'
  });
}
function pushSalvarConfig_(body) {
  const props = PropertiesService.getScriptProperties();
  if (body.autoNovoPedido !== undefined) props.setProperty('PUSH_NOVO_PEDIDO', body.autoNovoPedido ? 'true' : 'false');
  if (body.autoPedido !== undefined) props.setProperty('PUSH_PEDIDO_AUTO', body.autoPedido ? 'true' : 'false');
  registrarAudit_(body.funcionario, 'config_push', '', { autoNovoPedido: body.autoNovoPedido, autoPedido: body.autoPedido });
  return pushStatus_();
}

const DESCONTO_AVISTA_ = 0.10;            // IGUAL ao DESCONTO_AVISTA de common.js (o site mostra o mesmo valor à vista)

const STATUS_PEDIDO_ = ['novo', 'em_atendimento', 'aguardando_pagamento', 'separacao', 'pronto', 'concluido', 'cancelado'];

function dataBR_(d) {
  const dt = d instanceof Date ? d : new Date(d);
  if (isNaN(dt.getTime())) return '';
  return Utilities.formatDate(dt, FUSO_, 'dd/MM/yyyy HH:mm');
}
/** Texto de fora: sem caracteres de controle, com limite de tamanho e sem virar fórmula na planilha. */
function limparTexto_(v, max) {
  return acTexto_(String(v == null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, max || 200));
}
function arredondar2_(n) { return Math.round((Number(n) || 0) * 100) / 100; }
function jsonSeguro_(txt, padrao) { try { const v = JSON.parse(txt); return v === null || v === undefined ? padrao : v; } catch (e) { return padrao; } }

/** Linhas de uma aba como objetos {Coluna: valor, __linha: n}. Lê pelo NOME da coluna. */
function linhasComoObjetos_(sheet) {
  const dados = sheet.getDataRange().getValues();
  if (dados.length < 2) return [];
  const h = dados[0].map(function (x) { return String(x).trim(); });
  const out = [];
  for (let i = 1; i < dados.length; i++) {
    const o = {};
    h.forEach(function (k, j) { if (k) o[k] = dados[i][j]; });
    o.__linha = i + 1;
    out.push(o);
  }
  return out;
}

// ---------------------------------------------------------------- PEDIDOS

function codigoPedidoNormalizado_(v) {
  const t = String(v == null ? '' : v).trim().toUpperCase().replace(/\s+/g, '');
  const m = /^PED-?(\d{1,8})$/.exec(t);
  return m ? 'PED-' + String(parseInt(m[1], 10)).padStart(4, '0') : '';
}

function pedidoPorCodigo_(codigo) {
  const aba = getOrCreateSheet_(SHEET_PEDIDOS, CABECALHOS_PEDIDOS_);
  const lista = linhasComoObjetos_(aba);
  for (let i = lista.length - 1; i >= 0; i--) if (String(lista[i].Codigo) === codigo) return lista[i];
  return null;
}

/** O que o CLIENTE pode ver do pedido (sem telefone, sem endereço, sem nota interna). */
function pedidoParaCliente_(o) {
  return {
    codigo: String(o.Codigo), data: dataBR_(o.Data), atualizado: dataBR_(o.Atualizado), status: String(o.Status || 'novo'),
    entrega: String(o.Entrega || 'retirada'), total: Number(o.Total) || 0,
    itens: jsonSeguro_(o.Itens, []).map(function (i) { return { nome: i.nome, codigo: i.codigo || '' }; })
  };
}
/** O que a EQUIPE vê (painel de pedidos). */
function pedidoParaEquipe_(o) {
  const base = pedidoParaCliente_(o);
  base.nome = String(o.Nome || '');
  base.whatsapp = acWhats_(o.WhatsApp);
  base.endereco = String(o.Endereco || '');
  base.obs = String(o.Obs || '');
  base.nota = String(o.NotaInterna || '');
  base.origem = String(o.Origem || '');
  base.historico = jsonSeguro_(o.Historico, []).map(function (h) { return { quando: dataBR_(h.t), status: h.status, por: h.por || '' }; });
  base.itens = jsonSeguro_(o.Itens, []);
  return base;
}

/**
 * Pedido vindo do site (ABERTO ao público, igual ao registrarEvento). O servidor NUNCA confia no preço que o site mandou:
 * recebe só quais peças e os dados do cliente, e recalcula tudo pela planilha. Peça arquivada/esgotada/inexistente é ignorada.
 * Freios: no máximo 6 pedidos/hora por WhatsApp e 150/hora no total; o mesmo cliente com as mesmas peças em 2 min devolve o MESMO pedido
 * (duplo toque ou reenvio não cria dois).
 */
function criarPedido_(body) {
  const nome = limparTexto_(body.nome, 80);
  const whats = acWhats_(body.whatsapp);
  const entrega = body.entrega === 'entrega' ? 'entrega' : 'retirada';
  const endereco = entrega === 'entrega' ? limparTexto_(body.endereco, 300) : '';
  const obs = limparTexto_(body.obs, 500);
  if (nome.length < 2) return jsonResponse_({ ok: false, erro: 'Informe seu nome.' });
  if (whats.length !== 11) return jsonResponse_({ ok: false, erro: 'Informe o WhatsApp com DDD.' });
  if (entrega === 'entrega' && endereco.length < 8) return jsonResponse_({ ok: false, erro: 'Para entrega, informe rua, número e bairro.' });
  const idsBrutos = Array.isArray(body.itens) ? body.itens.map(function (i) { return i && i.id != null ? i.id : ''; }) : [];
  const ids = limparIds_(idsBrutos).slice(0, 30);
  if (!ids.length) return jsonResponse_({ ok: false, erro: 'Escolha ao menos uma peça.' });

  const cache = CacheService.getScriptCache();
  const chaveDup = 'pd_' + acSha256Hex_(whats + '|' + ids.slice().sort().join(',') + '|' + entrega).slice(0, 40);
  const jaFeito = cache.get(chaveDup);
  if (jaFeito) { const r = jsonSeguro_(jaFeito, null); if (r && r.codigo) return jsonResponse_({ ok: true, codigo: r.codigo, total: r.total, repetido: true, pushKey: pushChavePedido_(r.codigo) }); }
  const chaveW = 'pw_' + whats;
  const feitosW = parseInt(cache.get(chaveW) || '0', 10);
  const feitosG = parseInt(cache.get('pg_hora') || '0', 10);
  if (feitosW >= 6) return jsonResponse_({ ok: false, erro: 'Muitos pedidos deste número em pouco tempo. Fale com a loja pelo WhatsApp.' });
  if (feitosG >= 150) return jsonResponse_({ ok: false, erro: 'Muitos pedidos agora. Tente de novo em alguns minutos ou fale com a loja.' });

  const mapa = {};
  linhasComoObjetos_(getProdutosSheet_()).forEach(function (p) { mapa[String(p.ID)] = p; });
  const itens = [];
  let soma = 0, ignorados = 0;
  ids.forEach(function (id) {
    const p = mapa[id];
    const semEstoque = p && p.Estoque !== '' && p.Estoque !== null && p.Estoque !== undefined && Number(p.Estoque) <= 0;
    const preco = p ? Number(p.Preco) : 0;
    if (!p || normalizar_(p.Status) !== 'ativo' || semEstoque || !(preco > 0)) { ignorados++; return; }
    const avista = preco * (1 - DESCONTO_AVISTA_);
    soma += avista;
    itens.push({ id: String(p.ID), codigo: String(p.Codigo || ''), nome: String(p.Nome || ''), preco: arredondar2_(avista) });
  });
  if (!itens.length) return jsonResponse_({ ok: false, erro: 'As peças escolhidas não estão mais disponíveis. Atualize o catálogo e escolha de novo.' });
  const total = arredondar2_(soma);

  const trava = LockService.getScriptLock();
  try { trava.waitLock(20000); } catch (e) { return jsonResponse_({ ok: false, erro: 'O sistema está ocupado. Tente de novo em instantes.' }); }
  let codigo;
  try {
    const props = PropertiesService.getScriptProperties();
    const num = (parseInt(props.getProperty('PROXIMO_PEDIDO_NUM') || '0', 10) || 0) + 1;
    props.setProperty('PROXIMO_PEDIDO_NUM', String(num));
    codigo = 'PED-' + String(num).padStart(4, '0');
    const aba = getOrCreateSheet_(SHEET_PEDIDOS, CABECALHOS_PEDIDOS_);
    const agora = new Date();
    aba.appendRow([codigo, agora, agora, nome, whats, entrega, endereco, obs, JSON.stringify(itens), total, 'novo',
      JSON.stringify([{ t: agora.toISOString(), status: 'novo', por: 'cliente' }]), limparTexto_(body.origem || 'site', 20), '']);
  } finally {
    trava.releaseLock();
  }
  try {
    cache.put(chaveDup, JSON.stringify({ codigo: codigo, total: total }), 120);
    cache.put(chaveW, String(feitosW + 1), 3600);
    cache.put('pg_hora', String(feitosG + 1), 3600);
  } catch (e) { /* os freios são um extra */ }
  pushPedidoNovo_(codigo, total, itens.length, nome); // v3.3: avisa a equipe (enviado em doPostInterno_, depois que a trava foi solta)
  return jsonResponse_({ ok: true, codigo: codigo, total: total, ignorados: ignorados, pushKey: pushChavePedido_(codigo), itens: itens.map(function (i) { return { nome: i.nome, codigo: i.codigo }; }) });
}

/**
 * Acompanhamento pelo cliente: exige o código E os 4 últimos números do WhatsApp usado no pedido. Nunca lista pedidos.
 * Freio: 8 erros por código (e 400 no total) em 15 minutos bloqueiam novas tentativas.
 * Resposta igual para "não existe" e "números não conferem" (não dá para descobrir quais códigos existem).
 */
function consultarPedido_(body) {
  const codigo = codigoPedidoNormalizado_(body.codigo);
  const final4 = String(body.final == null ? '' : body.final).replace(/\D/g, '').slice(-4);
  if (!codigo || final4.length !== 4) return jsonResponse_({ ok: false, naoEncontrado: true });
  const cache = CacheService.getScriptCache();
  const ck = 'pc_' + codigo;
  const falhas = parseInt(cache.get(ck) || '0', 10);
  const globais = parseInt(cache.get('pc_global') || '0', 10);
  if (falhas >= 8 || globais >= 400) return jsonResponse_({ ok: false, bloqueado: true });
  const achado = pedidoPorCodigo_(codigo);
  if (!achado || acWhats_(achado.WhatsApp).slice(-4) !== final4) {
    try { cache.put(ck, String(falhas + 1), 900); cache.put('pc_global', String(globais + 1), 900); } catch (e) {}
    return jsonResponse_({ ok: false, naoEncontrado: true });
  }
  return jsonResponse_({ ok: true, pedido: pedidoParaCliente_(achado), pushKey: pushChavePedido_(codigo) });
}

/** Painel da equipe: lista os pedidos (mais novos primeiro) com a contagem por etapa. */
function listarPedidos_(body) {
  const aba = getOrCreateSheet_(SHEET_PEDIDOS, CABECALHOS_PEDIDOS_);
  const todas = linhasComoObjetos_(aba).reverse();
  const filtro = body.status ? String(body.status) : '';
  const limite = Math.max(1, Math.min(Number(body.limite) || 300, 1000));
  const contagem = {};
  const pedidos = [];
  todas.forEach(function (o) {
    const st = String(o.Status || 'novo');
    contagem[st] = (contagem[st] || 0) + 1;
    if (filtro && st !== filtro) return;
    if (pedidos.length < limite) pedidos.push(pedidoParaEquipe_(o));
  });
  return jsonResponse_({ ok: true, pedidos: pedidos, contagem: contagem, total: todas.length });
}

/** Muda a etapa do pedido e/ou a nota interna. Cada mudança de etapa fica no histórico do pedido (com quem fez). */
function atualizarPedido_(body) {
  const codigo = codigoPedidoNormalizado_(body.codigo);
  const status = body.status ? String(body.status) : '';
  if (!codigo) return jsonResponse_({ ok: false, erro: 'Código do pedido inválido.' });
  if (status && STATUS_PEDIDO_.indexOf(status) === -1) return jsonResponse_({ ok: false, erro: 'Etapa inválida.' });
  const trava = LockService.getScriptLock();
  try { trava.waitLock(20000); } catch (e) { return jsonResponse_({ ok: false, erro: 'O sistema está ocupado. Tente de novo em instantes.' }); }
  try {
    const aba = getOrCreateSheet_(SHEET_PEDIDOS, CABECALHOS_PEDIDOS_);
    const dados = aba.getDataRange().getValues();
    const h = dados[0].map(function (x) { return String(x).trim(); });
    const col = function (nome) { return h.indexOf(nome); };
    for (let i = dados.length - 1; i >= 1; i--) {
      if (String(dados[i][col('Codigo')]) !== codigo) continue;
      const linha = i + 1;
      const agora = new Date();
      if (status && status !== String(dados[i][col('Status')])) {
        aba.getRange(linha, col('Status') + 1).setValue(status);
        const hist = jsonSeguro_(dados[i][col('Historico')], []);
        hist.push({ t: agora.toISOString(), status: status, por: String(body.funcionario || 'equipe').slice(0, 60) });
        aba.getRange(linha, col('Historico') + 1).setValue(JSON.stringify(hist));
        pushPedidoEtapa_(codigo, status, String(dados[i][col('Entrega')] || 'retirada'), String(body.funcionario || 'equipe')); // v3.3: o aviso sai depois que a trava é solta
      }
      if (body.nota !== undefined) aba.getRange(linha, col('NotaInterna') + 1).setValue(limparTexto_(body.nota, 500));
      aba.getRange(linha, col('Atualizado') + 1).setValue(agora);
      registrarAudit_(body.funcionario, 'pedido_' + (status || 'nota'), '', { codigo: codigo, status: status });
      const o = linhasComoObjetos_(aba).filter(function (x) { return x.__linha === linha; })[0];
      return jsonResponse_({ ok: true, pedido: pedidoParaEquipe_(o) });
    }
    return jsonResponse_({ ok: false, erro: 'Pedido não encontrado.' });
  } finally {
    trava.releaseLock();
  }
}

// ---------------------------------------------------------------- CÓDIGOS FALTANTES

/**
 * Dá um código TK-#### às peças que estão sem código (peças antigas, cadastradas antes do código existir) e acerta o contador
 * para nunca repetir um código que já existe. Seguro para rodar mais de uma vez. Não mexe em quem já tem código.
 */
function gerarCodigosFaltantes_(body) {
  const sheet = getProdutosSheet_();
  garantirCabecalhos_(sheet);
  const dados = sheet.getDataRange().getValues();
  const h = dados[0].map(function (x) { return String(x).trim(); });
  const iCod = h.indexOf('Codigo'), iId = h.indexOf('ID'), iNome = h.indexOf('Nome');
  if (iCod === -1 || iId === -1) return jsonResponse_({ ok: false, erro: 'A planilha não tem as colunas ID/Codigo.' });
  const props = PropertiesService.getScriptProperties();
  const contador = parseInt(props.getProperty('PROXIMO_CODIGO_NUM') || '0', 10) || 0;
  let maior = contador;
  const vistos = {}, duplicados = [];
  for (let i = 1; i < dados.length; i++) {
    const c = String(dados[i][iCod] || '').trim().toUpperCase();
    if (!c) continue;
    if (vistos[c]) duplicados.push(c); else vistos[c] = true;
    const m = /^TK-(\d+)$/.exec(c);
    if (m) maior = Math.max(maior, parseInt(m[1], 10));
  }
  const geradas = [];
  for (let i = 1; i < dados.length; i++) {
    if (!String(dados[i][iId] || '').trim()) continue;
    if (String(dados[i][iCod] || '').trim()) continue;
    maior++;
    const cod = 'TK-' + String(maior).padStart(4, '0');
    sheet.getRange(i + 1, iCod + 1).setValue(cod);
    geradas.push({ id: String(dados[i][iId]), nome: String(dados[i][iNome] || ''), codigo: cod });
  }
  if (maior > contador) props.setProperty('PROXIMO_CODIGO_NUM', String(maior));
  if (geradas.length) registrarAudit_(body.funcionario, 'gerar_codigos', '', { quantidade: geradas.length, codigos: geradas.map(function (g) { return g.codigo; }) });
  return jsonResponse_({ ok: true, geradas: geradas, duplicados: duplicados, contadorAjustado: maior > contador });
}

  }
}

// ---------------------------------------------------------------- MÉTRICAS (só dados REAIS da aba Eventos e Pedidos)

function metricas_(body) {
  const dias = Math.max(1, Math.min(Number(body.dias) || 30, 365));
  const chave = 'metricas:' + dias + ':' + versaoCacheProdutos_();
  const emCache = cacheLerGrande_(chave);
  if (emCache) return respostaTexto_(emCache);

  const agora = new Date();
  const inicio = new Date(agora.getTime() - dias * 86400000);
  const prods = {};
  linhasComoObjetos_(getProdutosSheet_()).forEach(function (p) { prods[String(p.ID)] = p; });
  const porDia = {};
  for (let k = dias - 1; k >= 0; k--) porDia[Utilities.formatDate(new Date(agora.getTime() - k * 86400000), FUSO_, 'yyyy-MM-dd')] = { visualizacoes: 0, whatsapp: 0, pedidos: 0 };
  const porPeca = {}, porCat = {};
  let views = 0, whats = 0, eventosNoPeriodo = 0;
  const aba = getSS_().getSheetByName(SHEET_EVENTOS);
  if (aba && aba.getLastRow() > 1) {
    const dados = aba.getDataRange().getValues();
    for (let i = 1; i < dados.length; i++) {
      const quando = dados[i][0] instanceof Date ? dados[i][0] : new Date(dados[i][0]);
      if (isNaN(quando.getTime()) || quando < inicio) continue;
      const tipo = String(dados[i][3] || '');
      if (tipo !== 'visualizacao' && tipo !== 'whatsapp') continue;
      eventosNoPeriodo++;
      const dia = Utilities.formatDate(quando, FUSO_, 'yyyy-MM-dd');
      const pid = String(dados[i][1]);
      const peca = porPeca[pid] || (porPeca[pid] = { visualizacoes: 0, whatsapp: 0, codigo: String(dados[i][2] || '') });
      const p = prods[pid];
      const cats = p ? String(p.Categoria || '').split(',').map(function (x) { return x.trim(); }).filter(Boolean) : [];
      if (tipo === 'visualizacao') { views++; peca.visualizacoes++; if (porDia[dia]) porDia[dia].visualizacoes++; cats.forEach(function (c) { (porCat[c] = porCat[c] || { visualizacoes: 0, whatsapp: 0 }).visualizacoes++; }); }
      else { whats++; peca.whatsapp++; if (porDia[dia]) porDia[dia].whatsapp++; cats.forEach(function (c) { (porCat[c] = porCat[c] || { visualizacoes: 0, whatsapp: 0 }).whatsapp++; }); }
    }
  }
  const taxa = function (w, v) { return v > 0 ? Math.round((w / v) * 1000) / 10 : null; }; // WhatsApp por 100 visualizações

  const topProdutos = Object.keys(porPeca).map(function (pid) {
    const e = porPeca[pid], p = prods[pid];
    return { id: pid, codigo: (p && p.Codigo) || e.codigo, nome: p ? String(p.Nome) : '(peça removida)', visualizacoes: e.visualizacoes, whatsapp: e.whatsapp, taxa: taxa(e.whatsapp, e.visualizacoes) };
  }).sort(function (a, b) { return (b.visualizacoes + b.whatsapp) - (a.visualizacoes + a.whatsapp); }).slice(0, 10);
  const categorias = Object.keys(porCat).map(function (c) {
    return { nome: c, visualizacoes: porCat[c].visualizacoes, whatsapp: porCat[c].whatsapp, taxa: taxa(porCat[c].whatsapp, porCat[c].visualizacoes) };
  }).sort(function (a, b) { return (b.visualizacoes + b.whatsapp) - (a.visualizacoes + a.whatsapp); });

  const ped = { total: 0, valor: 0, porStatus: {}, concluidos: 0, valorConcluidos: 0 };
  const abaPed = getSS_().getSheetByName(SHEET_PEDIDOS);
  if (abaPed && abaPed.getLastRow() > 1) {
    linhasComoObjetos_(abaPed).forEach(function (o) {
      const quando = o.Data instanceof Date ? o.Data : new Date(o.Data);
      if (isNaN(quando.getTime()) || quando < inicio) return;
      const st = String(o.Status || 'novo');
      ped.total++; ped.porStatus[st] = (ped.porStatus[st] || 0) + 1;
      const dia = Utilities.formatDate(quando, FUSO_, 'yyyy-MM-dd');
      if (porDia[dia]) porDia[dia].pedidos++;
      if (st !== 'cancelado') ped.valor += Number(o.Total) || 0;
      if (st === 'concluido') { ped.concluidos++; ped.valorConcluidos += Number(o.Total) || 0; }
    });
    ped.valor = arredondar2_(ped.valor); ped.valorConcluidos = arredondar2_(ped.valorConcluidos);
  }
  const resultado = {
    ok: true, dias: dias, geradoEm: dataBR_(agora), semDados: eventosNoPeriodo === 0 && ped.total === 0,
    totais: { visualizacoes: views, whatsapp: whats, whatsappPor100Visualizacoes: taxa(whats, views) },
    porDia: Object.keys(porDia).sort().map(function (d) { return { dia: d, visualizacoes: porDia[d].visualizacoes, whatsapp: porDia[d].whatsapp, pedidos: porDia[d].pedidos }; }),
    topProdutos: topProdutos, categorias: categorias, pedidos: ped
  };
  const texto = JSON.stringify(resultado);
  cacheGravarGrande_(chave, texto, 120);
  return respostaTexto_(texto);
}

// ---------------------------------------------------------------- IMPORTAR PRODUTOS (CSV): prévia (simular) e aplicação

const CAMPOS_IMPORTACAO_ = ['codigo', 'id', 'nome', 'categoria', 'genero', 'preco', 'estoque', 'tamanhos', 'cores', 'descricao', 'novidade', 'status', 'foto_url'];

/**
 * Recebe linhas já lidas do CSV (chaves em CAMPOS_IMPORTACAO_). Acha a peça pelo código (ou ID); célula vazia = não mexe.
 * Peça que não existe só é CRIADA se vier com nome, preço válido e foto (https). Com simular:true não grava nada e devolve o que ia acontecer
 * (é a prévia que o painel mostra antes de confirmar). Máximo 300 linhas por vez.
 */
function importarProdutos_(body) {
  const entrada = Array.isArray(body.linhas) ? body.linhas.slice(0, 300) : [];
  if (!entrada.length) return jsonResponse_({ ok: false, erro: 'Nenhuma linha para importar.' });
  const simular = body.simular === true;
  const criar = body.criar !== false;
  const sheet = getProdutosSheet_();
  garantirCabecalhos_(sheet);
  const dados = sheet.getDataRange().getValues();
  const h = dados[0].map(function (x) { return String(x).trim(); });
  const col = {};
  ['ID', 'Nome', 'Categoria', 'Genero', 'Preco', 'Foto_URL', 'Status', 'Data', 'Descricao', 'Estoque', 'Tamanhos', 'Cores', 'Novidade', 'Codigo'].forEach(function (n) { col[n] = h.indexOf(n); });
  const porCodigo = {}, porId = {};
  let maiorExistente = 0; // o contador nunca pode ficar abaixo do maior código que já existe (senão repetiria um código)
  for (let i = 1; i < dados.length; i++) {
    const c = String(dados[i][col.Codigo] || '').trim().toUpperCase();
    if (c) {
      porCodigo[c] = i;
      const m = /^TK-(\d+)$/.exec(c);
      if (m) maiorExistente = Math.max(maiorExistente, parseInt(m[1], 10));
    }
    porId[String(dados[i][col.ID])] = i;
  }
  const atualizados = [], criados = [], ignorados = [], historico = [];
  const props = PropertiesService.getScriptProperties();
  let contador = Math.max(parseInt(props.getProperty('PROXIMO_CODIGO_NUM') || '0', 10) || 0, maiorExistente);
  const categoriasNovas = {};
  const mudancas = {}; // índice da linha -> linha nova (valores)
  const tocadas = {};  // peças já vistas neste arquivo

  entrada.forEach(function (l, n) {
    const num = n + 2; // linha do arquivo (1 = cabeçalho)
    const v = {};
    CAMPOS_IMPORTACAO_.forEach(function (k) { v[k] = l && l[k] !== undefined && l[k] !== null ? String(l[k]).trim() : ''; });
    const chaveCod = v.codigo.toUpperCase();
    let i = chaveCod && porCodigo[chaveCod] !== undefined ? porCodigo[chaveCod] : (v.id && porId[v.id] !== undefined ? porId[v.id] : -1);
    // valores validados
    let preco = null, estoque = null, novidade = null;
    if (v.preco !== '') { preco = Number(v.preco.replace(',', '.')); if (isNaN(preco) || preco <= 0) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Preço inválido.' }); return; } }
    if (v.estoque !== '') { estoque = Number(v.estoque.replace(',', '.')); if (isNaN(estoque) || estoque < 0) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Estoque inválido.' }); return; } }
    if (v.novidade !== '') { novidade = acBool_(v.novidade); if (novidade === null) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Novidade: use sim ou não.' }); return; } }
    let status = '';
    if (v.status !== '') { status = normalizar_(v.status) === 'ativo' ? 'Ativo' : (['inativo', 'arquivado'].indexOf(normalizar_(v.status)) !== -1 ? 'Inativo' : null); if (status === null) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Status: use Ativo ou Inativo.' }); return; } }

    if (i !== -1) {
      if (tocadas[i]) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Peça repetida no arquivo (vale só a primeira linha).' }); return; }
      const antes = dados[i].slice();
      const nova = antes.slice();
      const campos = [];
      const set = function (nomeCol, rotulo, valor, atual) { if (String(valor) !== String(atual)) { nova[col[nomeCol]] = valor; campos.push(rotulo + ': ' + atual + ' → ' + valor); } };
      if (v.nome) set('Nome', 'nome', limparTexto_(v.nome, 120), antes[col.Nome]);
      if (v.categoria) { set('Categoria', 'categoria', limparTexto_(v.categoria, 120), antes[col.Categoria]); }
      if (v.genero) set('Genero', 'gênero', limparTexto_(v.genero, 120), antes[col.Genero]);
      if (preco !== null) set('Preco', 'preço', preco, antes[col.Preco]);
      if (estoque !== null) set('Estoque', 'estoque', estoque, antes[col.Estoque]);
      if (v.tamanhos) set('Tamanhos', 'tamanhos', limparTexto_(v.tamanhos, 120), antes[col.Tamanhos]);
      if (v.cores) set('Cores', 'cores', limparTexto_(v.cores, 120), antes[col.Cores]);
      if (v.descricao) set('Descricao', 'descrição', limparTexto_(v.descricao, 1000), antes[col.Descricao]);
      if (novidade !== null) set('Novidade', 'novidade', novidade, acBool_(antes[col.Novidade]) === true);
      if (status) set('Status', 'status', status, antes[col.Status]);
      if (v.foto_url) {
        if (/^https:\/\//.test(v.foto_url)) set('Foto_URL', 'foto', v.foto_url, antes[col.Foto_URL]);
        else { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Foto: precisa começar com https://' }); return; }
      }
      tocadas[i] = true;
      if (!campos.length) { ignorados.push({ linha: num, codigo: String(antes[col.Codigo] || v.codigo), motivo: 'Nada mudou.', semMudanca: true }); return; }
      mudancas[i] = nova; tocadas[i] = true;
      atualizados.push({ linha: num, id: String(antes[col.ID]), codigo: String(antes[col.Codigo] || ''), nome: String(nova[col.Nome]), campos: campos });
      if ((preco !== null && String(antes[col.Preco]) !== String(preco)) || (estoque !== null && String(antes[col.Estoque]) !== String(estoque))) {
        historico.push({ id: String(antes[col.ID]), codigo: String(antes[col.Codigo] || ''), nome: String(nova[col.Nome]), precoAnterior: antes[col.Preco], preco: nova[col.Preco], estoqueAnterior: antes[col.Estoque], estoque: nova[col.Estoque] });
      }
      if (v.categoria) categoriasNovas[limparTexto_(v.categoria, 120)] = true;
      return;
    }
    // peça nova
    if (!criar) { ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Peça não encontrada (criação desligada).' }); return; }
    if (!v.nome || preco === null || !/^https:\/\//.test(v.foto_url)) {
      ignorados.push({ linha: num, codigo: v.codigo, motivo: 'Peça nova precisa de nome, preço e foto (https://).' });
      return;
    }
    contador++;
    const codigoNovo = 'TK-' + String(contador).padStart(4, '0');
    const idNovo = String(new Date().getTime() + n);
    const nova = [];
    for (let k = 0; k < h.length; k++) nova.push('');
    nova[col.ID] = idNovo; nova[col.Nome] = limparTexto_(v.nome, 120); nova[col.Categoria] = limparTexto_(v.categoria, 120); nova[col.Genero] = limparTexto_(v.genero, 120);
    nova[col.Preco] = preco; nova[col.Foto_URL] = v.foto_url; nova[col.Status] = status || 'Ativo'; nova[col.Data] = new Date();
    nova[col.Descricao] = limparTexto_(v.descricao, 1000); nova[col.Estoque] = estoque === null ? '' : estoque;
    nova[col.Tamanhos] = limparTexto_(v.tamanhos, 120); nova[col.Cores] = limparTexto_(v.cores, 120); nova[col.Novidade] = novidade === true; nova[col.Codigo] = codigoNovo;
    criados.push({ linha: num, id: idNovo, codigo: codigoNovo, nome: String(nova[col.Nome]), nova: nova });
    if (v.categoria) categoriasNovas[limparTexto_(v.categoria, 120)] = true;
  });

  if (!simular) {
    Object.keys(mudancas).forEach(function (idx) {
      sheet.getRange(Number(idx) + 1, 1, 1, h.length).setValues([mudancas[idx].slice(0, h.length)]);
    });
    criados.forEach(function (c) { sheet.appendRow(c.nova); });
    if (criados.length) props.setProperty('PROXIMO_CODIGO_NUM', String(contador));
    Object.keys(categoriasNovas).forEach(function (c) { if (c) adicionarValorLista_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO, c); });
    if (historico.length) {
      const abaH = getOrCreateSheet_(SHEET_HISTORICO, ['DataHora', 'Funcionario', 'Acao', 'ProdutoID', 'Codigo', 'Nome', 'PrecoAnterior', 'PrecoNovo', 'EstoqueAnterior', 'EstoqueNovo']);
      const quem = acTexto_(body.funcionario || 'Sistema'), agora = new Date();
      const linhasH = historico.map(function (x) { return [agora, quem, 'importar_csv', x.id, x.codigo, acTexto_(x.nome), x.precoAnterior, x.preco, x.estoqueAnterior, x.estoque]; });
      abaH.getRange(abaH.getLastRow() + 1, 1, linhasH.length, 10).setValues(linhasH);
    }
    registrarAudit_(body.funcionario, 'importar_csv', '', { atualizadas: atualizados.length, criadas: criados.length, ignoradas: ignorados.length });
  }
  return jsonResponse_({
    ok: true, simulacao: simular, atualizados: atualizados, ignorados: ignorados,
    criados: criados.map(function (c) { return { linha: c.linha, id: c.id, codigo: c.codigo, nome: c.nome }; })
  });
}

// ---------------------------------------------------------------- FEED DE CATÁLOGO (Meta/Instagram e Google) e MAPA DO SITE

function csvCampo_(v) {
  const t = String(v == null ? '' : v);
  return /[",\r\n]/.test(t) ? '"' + t.replace(/"/g, '""') + '"' : t;
}
function xmlEsc_(v) {
  return String(v == null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}
function linkPecaSite_(p) {
  const cod = String(p.Codigo || '').trim();
  return cod ? SITE_URL + 'p/' + encodeURIComponent(cod) : SITE_URL + '?id=' + encodeURIComponent(p.ID);
}
/** Uma peça no formato do catálogo da Meta (colunas) — o mesmo vale para o Google Merchant (g:...). Preço = preço CHEIO (o à vista é desconto de pagamento, não promoção). */
function itemFeed_(p) {
  const cod = String(p.Codigo || '').trim();
  const generos = normalizar_(p.Genero);
  const estoqueInf = p.Estoque !== '' && p.Estoque !== null && p.Estoque !== undefined;
  const galeria = String(p.Fotos_Galeria || '').split(',').map(function (x) { return x.trim(); }).filter(Boolean).slice(0, 9);
  return {
    id: cod || String(p.ID),
    title: String(p.Nome || '').slice(0, 150),
    description: String(p.Descricao || p.Nome || '').slice(0, 5000),
    availability: estoqueInf && Number(p.Estoque) <= 0 ? 'out of stock' : 'in stock',
    condition: 'new',
    price: (Number(p.Preco) || 0).toFixed(2) + ' BRL',
    link: linkPecaSite_(p),
    image_link: String(p.Foto_URL || ''),
    additional_image_link: galeria.join(','),
    brand: 'TENKiTER Modas',
    gender: /feminino|menina/.test(generos) ? 'female' : (/masculino|menino/.test(generos) ? 'male' : 'unisex'),
    age_group: /infantil|menina|menino/.test(generos) ? 'kids' : 'adult',
    color: String(p.Cores || '').split(',').map(function (x) { return x.trim(); }).filter(Boolean).slice(0, 3).join('/'),
    product_type: String(p.Categoria || ''),
    quantity_to_sell_on_facebook: estoqueInf ? String(Math.max(0, Number(p.Estoque) || 0)) : ''
  };
}
const COLUNAS_FEED_ = ['id', 'title', 'description', 'availability', 'condition', 'price', 'link', 'image_link', 'additional_image_link', 'brand', 'gender', 'age_group', 'color', 'product_type', 'quantity_to_sell_on_facebook'];

function feedProdutos_(formato) {
  const fmt = formato === 'xml' ? 'xml' : 'csv';
  const chave = 'feed:' + fmt + ':' + versaoCacheProdutos_();
  let texto = cacheLerGrande_(chave);
  if (!texto) {
    const itens = linhasComoObjetos_(getProdutosSheet_()).filter(function (p) { return normalizar_(p.Status) === 'ativo' && String(p.Foto_URL || '').trim() && Number(p.Preco) > 0; }).map(itemFeed_);
    if (fmt === 'csv') {
      texto = [COLUNAS_FEED_.join(',')].concat(itens.map(function (it) { return COLUNAS_FEED_.map(function (c) { return csvCampo_(it[c]); }).join(','); })).join('\r\n') + '\r\n';
    } else {
      texto = '<?xml version="1.0" encoding="UTF-8"?>\n<rss xmlns:g="http://base.google.com/ns/1.0" version="2.0"><channel>' +
        '<title>TENKiTER Modas</title><link>' + xmlEsc_(SITE_URL) + '</link><description>Catálogo da TENKiTER Modas, Crateús-CE</description>' +
        itens.map(function (it) {
          return '<item>' + COLUNAS_FEED_.filter(function (c) { return String(it[c]).length; }).map(function (c) {
            return '<g:' + c + '>' + xmlEsc_(it[c]) + '</g:' + c + '>';
          }).join('') + '</item>';
        }).join('') + '</channel></rss>';
    }
    cacheGravarGrande_(chave, texto, 300);
  }
  return ContentService.createTextOutput(texto).setMimeType(fmt === 'xml' ? ContentService.MimeType.XML : ContentService.MimeType.CSV);
}

/** Mapa do site (sitemap) com a página inicial e cada peça ativa (/p/<código>). Usado pelo Worker da Cloudflare (og-worker) quando publicado. */
function sitemapProdutos_() {
  const itens = linhasComoObjetos_(getProdutosSheet_()).filter(function (p) { return normalizar_(p.Status) === 'ativo' && String(p.Codigo || '').trim(); });
  const urls = [SITE_URL, SITE_URL + 'privacidade.html'].concat(itens.map(linkPecaSite_));
  const xml = '<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">' +
    urls.map(function (u) { return '<url><loc>' + xmlEsc_(u) + '</loc></url>'; }).join('') + '</urlset>';
  return ContentService.createTextOutput(xml).setMimeType(ContentService.MimeType.XML);
}

// ---------------------------------------------------------------- INTEGRAÇÕES (Pixel da Meta, API de Conversões, sinônimos, segmentos de aviso)
// Públicos (ficam em ?action=config): ID do Pixel e sinônimos. SEGREDOS (token da API de Conversões, código de teste): só aqui, nas Propriedades do Script —
// nunca voltam para o navegador, nunca vão para o repositório.

function linhasSinonimos_(texto) {
  return String(texto || '').split(/\r?\n/).map(function (l) { return l.trim(); }).filter(function (l) { return /^[^=]{1,40}=[^=]{1,200}$/.test(l); }).slice(0, 100);
}

function configPublica_() {
  const cache = CacheService.getScriptCache();
  try { const c = cache.get('cfg_pub'); if (c) return respostaTexto_(c); } catch (e) {}
  const props = PropertiesService.getScriptProperties();
  const pixel = String(props.getProperty('META_PIXEL_ID') || '').trim();
  const texto = JSON.stringify({ ok: true, config: { pixelId: /^\d{8,20}$/.test(pixel) ? pixel : '', sinonimos: linhasSinonimos_(props.getProperty('SINONIMOS_BUSCA')) } });
  try { cache.put('cfg_pub', texto, 300); } catch (e) {}
  return respostaTexto_(texto);
}

function statusIntegracoes_() {
  const props = PropertiesService.getScriptProperties();
  return jsonResponse_({
    ok: true,
    pixelId: String(props.getProperty('META_PIXEL_ID') || ''),
    capiConfigurado: !!props.getProperty('META_CAPI_TOKEN'),
    testeCodigo: String(props.getProperty('META_TEST_EVENT_CODE') || ''),
    graphVersao: String(props.getProperty('META_GRAPH_VERSAO') || 'v23.0'),
    sinonimos: String(props.getProperty('SINONIMOS_BUSCA') || ''),
    segmentosPush: String(props.getProperty('PUSH_SEGMENTOS') || '')
  });
}

function salvarIntegracoes_(body) {
  const props = PropertiesService.getScriptProperties();
  if (body.pixelId !== undefined) {
    const id = String(body.pixelId).replace(/\s/g, '');
    if (id && !/^\d{8,20}$/.test(id)) return jsonResponse_({ ok: false, erro: 'O ID do Pixel tem só números (de 8 a 20 dígitos).' });
    if (id) props.setProperty('META_PIXEL_ID', id); else props.deleteProperty('META_PIXEL_ID');
  }
  if (body.limparTokenCapi === true) props.deleteProperty('META_CAPI_TOKEN');
  else if (body.tokenCapi !== undefined && String(body.tokenCapi).trim()) {
    const t = String(body.tokenCapi).trim();
    if (!/^[A-Za-z0-9_\-|.]{20,400}$/.test(t)) return jsonResponse_({ ok: false, erro: 'Token inválido (confira se copiou inteiro, sem espaços).' });
    props.setProperty('META_CAPI_TOKEN', t);
  }
  if (body.testeCodigo !== undefined) {
    const c = String(body.testeCodigo).trim();
    if (c && !/^[A-Za-z0-9_\-]{3,40}$/.test(c)) return jsonResponse_({ ok: false, erro: 'Código de teste inválido.' });
    if (c) props.setProperty('META_TEST_EVENT_CODE', c); else props.deleteProperty('META_TEST_EVENT_CODE');
  }
  if (body.graphVersao !== undefined) {
    const g = String(body.graphVersao).trim();
    if (g && !/^v\d{1,2}\.\d$/.test(g)) return jsonResponse_({ ok: false, erro: 'Versão da API inválida (ex.: v23.0).' });
    if (g) props.setProperty('META_GRAPH_VERSAO', g); else props.deleteProperty('META_GRAPH_VERSAO');
  }
  if (body.sinonimos !== undefined) {
    const linhas = linhasSinonimos_(body.sinonimos);
    if (linhas.length) props.setProperty('SINONIMOS_BUSCA', linhas.join('\n')); else props.deleteProperty('SINONIMOS_BUSCA');
  }
  if (body.segmentosPush !== undefined) {
    const lista = String(body.segmentosPush).split(/[,\n]/).map(function (s) { return s.trim(); }).filter(function (s) { return /^[\w \-áàâãéêíóôõúçÁÀÂÃÉÊÍÓÔÕÚÇ]{2,60}$/.test(s); }).slice(0, 20);
    if (lista.length) props.setProperty('PUSH_SEGMENTOS', lista.join(',')); else props.deleteProperty('PUSH_SEGMENTOS');
  }
  try { CacheService.getScriptCache().remove('cfg_pub'); } catch (e) {}
  registrarAudit_(body.funcionario, 'salvar_integracoes', '', { campos: Object.keys(body).filter(function (k) { return ['action', 'sessao', 'whatsapp', 'pin', 'funcionario', 'tokenCapi'].indexOf(k) === -1; }), tokenAlterado: !!(body.tokenCapi || body.limparTokenCapi) });
  return statusIntegracoes_();
}

/**
 * Repete um evento do site na API de Conversões da Meta (servidor -> Meta). SÓ roda se: o Pixel e o token estão configurados E o navegador mandou
 * "meta" (o site só manda quando a pessoa aceitou o aviso do Pixel). Usa o mesmo event_id do navegador para a Meta não contar duas vezes.
 * Qualquer falha é registrada em Erros_Log e IGNORADA — nunca atrapalha o registro do evento nem o site.
 */
function enviarEventoMetaCapi_(body, tipo) {
  try {
    const props = PropertiesService.getScriptProperties();
    const pixel = String(props.getProperty('META_PIXEL_ID') || '').trim();
    const token = props.getProperty('META_CAPI_TOKEN');
    if (!pixel || !token || !body.meta || typeof body.meta !== 'object') return;
    const nome = tipo === 'whatsapp' ? 'Contact' : (tipo === 'visualizacao' ? 'ViewContent' : '');
    if (!nome) return;
    const evento = {
      event_name: nome, event_time: Math.floor(new Date().getTime() / 1000), action_source: 'website',
      event_source_url: String(body.meta.url || SITE_URL).slice(0, 500),
      user_data: { client_user_agent: String(body.meta.ua || '').slice(0, 400) },
      custom_data: { content_ids: [String(body.codigo || body.produtoId || '')], content_type: 'product' }
    };
    if (body.meta.eventId) evento.event_id = String(body.meta.eventId).slice(0, 100);
    if (body.meta.fbp) evento.user_data.fbp = String(body.meta.fbp).slice(0, 200);
    if (body.meta.fbc) evento.user_data.fbc = String(body.meta.fbc).slice(0, 300);
    const corpo = { data: [evento] };
    const teste = props.getProperty('META_TEST_EVENT_CODE');
    if (teste) corpo.test_event_code = teste;
    const versao = props.getProperty('META_GRAPH_VERSAO') || 'v23.0';
    const resp = UrlFetchApp.fetch('https://graph.facebook.com/' + versao + '/' + pixel + '/events?access_token=' + encodeURIComponent(token), {
      method: 'post', contentType: 'application/json', payload: JSON.stringify(corpo), muteHttpExceptions: true
    });
    if (resp.getResponseCode() >= 300) logErroServidor_('metaCapi', new Error('HTTP ' + resp.getResponseCode() + ' ' + String(resp.getContentText()).slice(0, 300)), { evento: nome });
  } catch (erro) {
    logErroServidor_('metaCapi', erro, {});
  }
}

/** Categoria escolhida no lote: se já existe (sem acento/maiúscula) reaproveita a escrita que já existe; senão cadastra na lista. Devolve '' se vazia. */
function categoriaExistenteOuNova_(valor) {
  const nome = String(valor == null ? '' : valor).replace(/,/g, ' ').trim().slice(0, 60);
  if (!nome) return '';
  const lista = getListaValores_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO);
  const alvo = acNorm_(nome);
  for (let i = 0; i < lista.length; i++) if (acNorm_(lista[i]) === alvo) return lista[i];
  adicionarValorLista_(SHEET_CATEGORIAS, CATEGORIAS_PADRAO, nome);
  return nome;
}

/** Título e descrição de busca montados SÓ com o que a peça já tem (nome, gênero, cores, descrição atual). Não inventa material, caimento nem promoção. */
function descricaoLocal_(body) {
  const nome = String(body.nome || '').trim();
  const cat = String(body.categoria || '').split(',')[0].trim();
  const gen = String(body.genero || '').trim().toLowerCase();
  const cores = String(body.cores || '').split(',').map(function (c) { return c.trim(); }).filter(Boolean).slice(0, 3);
  const atual = String(body.descricao || '').replace(/\s+/g, ' ').trim().slice(0, 140);
  const base = nome || cat || 'Peça';
  const titulo = (base + ' | TENKiTER Modas').slice(0, 60);
  let d = base + (gen ? ' (' + gen + ')' : '') + ' na TENKiTER Modas, em Crateús-CE.';
  if (atual) d += ' ' + atual + (/[.!?]$/.test(atual) ? '' : '.');
  if (cores.length) d += ' Cores: ' + cores.join(', ') + '.';
  d += ' ' + Math.round(DESCONTO_AVISTA_ * 100) + '% de desconto à vista. Peça pelo WhatsApp.';
  return { ok: true, titulo: titulo, descricao: d.slice(0, 300), local: true };
}

/** As chaves ficam nas Propriedades do Script. Esta parte NUNCA devolve a chave, nunca a escreve em log ou auditoria e só é chamada por Admin total com login. */
const RENDER_API_ = 'https://api.render.com/v1';
const CLOUDFLARE_API_ = 'https://api.cloudflare.com/client/v4';
const RENDER_SERVICO_ = 'srv-dasa63fpn0mc73fh8fgg';
const WORKER_NOME_ = 'tenkiter-og';
const WORKER_BASE_ = 'https://tenkiter-og.distkrpconfeccoes.workers.dev';
const REGRAS_RENDER_ = ['/p/*', '/feed.csv', '/feed.xml', '/sitemap.xml']; // cada uma é um Rewrite para WORKER_BASE_ + o mesmo caminho
const ONESIGNAL_APP_ID_PADRAO_ = '535f6b0d-c866-43c2-b241-43bd7ab62fae'; // público (também está em common.js)
const ONESIGNAL_UUID_ = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/;
const CONEXOES_ = {
  render:     { prop: 'RENDER_API_KEY',        rotulo: 'Render',     padrao: /^rnd_[A-Za-z0-9]{16,100}$/ },
  cloudflare: { prop: 'CLOUDFLARE_API_TOKEN',  rotulo: 'Cloudflare', padrao: /^[A-Za-z0-9_\-]{30,160}$/ },
  // Organization API Key do OneSignal (começa com os_v2_org_). Manda em TODOS os apps da organização: só fica no servidor.
  onesignal:  { prop: 'ONESIGNAL_ORG_API_KEY', rotulo: 'OneSignal',  padrao: /^os_v2_org_[A-Za-z0-9]{30,300}$/ }
};

function conexaoChave_(nome) {
  const c = CONEXOES_[nome];
  return c ? String(PropertiesService.getScriptProperties().getProperty(c.prop) || '').trim() : '';
}
/** Chamada autenticada. Devolve { codigo, dados, erro }; nunca lança e nunca devolve a chave. */
function conexaoHttp_(metodo, url, chave, corpo, esquema) {
  const opt = { method: metodo, muteHttpExceptions: true, headers: { Authorization: (esquema || 'Bearer') + ' ' + chave, Accept: 'application/json' } };
  if (corpo !== undefined) { opt.contentType = 'application/json'; opt.payload = JSON.stringify(corpo); }
  try {
    const r = UrlFetchApp.fetch(url, opt);
    let dados = null;
    try { dados = JSON.parse(r.getContentText()); } catch (e) { dados = null; }
    return { codigo: r.getResponseCode(), dados: dados };
  } catch (e) {
    return { codigo: 0, dados: null, erro: 'Sem conexão com o serviço agora. Tente de novo.' };
  }
}
/** Frase curta (sem JSON) para um erro de API. */
function conexaoMsgApi_(r) {
  if (r.erro) return r.erro;
  if (r.codigo === 401 || r.codigo === 403) return 'A chave foi recusada (inválida, vencida ou sem permissão).';
  const m = r.dados && (r.dados.message || (r.dados.errors && r.dados.errors[0] && r.dados.errors[0].message));
  return 'O serviço respondeu com erro ' + r.codigo + (m ? ': ' + String(m).slice(0, 160) : '.');
}
/** App ID do OneSignal usado nas chamadas com a chave da organização (só aceita o formato UUID: ele vai dentro do endereço). */
function onesignalAppId_() {
  const v = String(PropertiesService.getScriptProperties().getProperty('ONESIGNAL_APP_ID') || '').trim().toLowerCase();
  return ONESIGNAL_UUID_.test(v) ? v : ONESIGNAL_APP_ID_PADRAO_;
}
function onesignalOrgId_() { return String(PropertiesService.getScriptProperties().getProperty('ONESIGNAL_ORG_ID') || '').trim(); }
/** Testa a chave de verdade, na API. `extra.orgId` (OneSignal) = o Organization ID que a pessoa informou. */
function conexaoTestar_(nome, chave, extra) {
  if (nome === 'onesignal') {
    // Só lê id/nome/organização do app: a resposta do OneSignal traz também a chave de envio do app, que NUNCA é repassada.
    const r = conexaoHttp_('get', PUSH_API_ + '/apps/' + onesignalAppId_(), chave, undefined, 'Key');
    if (r.codigo === 200 && r.dados && r.dados.id) {
      const orgInf = String((extra && extra.orgId) || onesignalOrgId_() || '');
      if (orgInf && r.dados.organization_id && String(r.dados.organization_id) !== orgInf) return { ok: false, detalhe: 'A chave é válida, mas o Organization ID informado não é o do app do TENKiTER. Copie de novo o código da tela Security.' };
      return { ok: true, detalhe: 'Enxerga o app "' + String(r.dados.name || r.dados.id).slice(0, 60) + '".' };
    }
    if (r.codigo === 404) return { ok: false, detalhe: 'A chave é válida, mas não enxerga o app do TENKiTER (é de outra organização do OneSignal?).' };
    return { ok: false, detalhe: conexaoMsgApi_(r) };
  }
  if (nome === 'render') {
    const r = conexaoHttp_('get', RENDER_API_ + '/services/' + RENDER_SERVICO_, chave);
    if (r.codigo === 200) return { ok: true, detalhe: 'Enxerga o site "' + String((r.dados && r.dados.name) || RENDER_SERVICO_) + '".' };
    if (r.codigo === 404) return { ok: false, detalhe: 'A chave é válida, mas não enxerga o site do TENKiTER (é de outra conta do Render?).' };
    return { ok: false, detalhe: conexaoMsgApi_(r) };
  }
  const v = conexaoHttp_('get', CLOUDFLARE_API_ + '/user/tokens/verify', chave);
  if (v.codigo === 200 && v.dados && v.dados.success && v.dados.result && v.dados.result.status === 'active') return { ok: true, detalhe: 'Chave ativa.' };
  const a = conexaoHttp_('get', CLOUDFLARE_API_ + '/accounts', chave); // token de conta (em vez de usuário) não passa no /user/tokens/verify
  if (a.codigo === 200 && a.dados && a.dados.success && a.dados.result && a.dados.result.length) return { ok: true, detalhe: 'Chave ativa.' };
  return { ok: false, detalhe: v.codigo === 200 ? 'A chave não está ativa.' : conexaoMsgApi_(v) };
}

function statusConexoesObj_() {
  const props = PropertiesService.getScriptProperties();
  const conexoes = {};
  Object.keys(CONEXOES_).forEach(function (n) {
    const k = conexaoChave_(n);
    conexoes[n] = { rotulo: CONEXOES_[n].rotulo, temChave: !!k, fim: k ? k.slice(-4) : '', salvaEm: String(props.getProperty(CONEXOES_[n].prop + '_EM') || '') };
  });
  // OneSignal: o Organization ID e o App ID são públicos (podem ir à tela); a chave de envio do app só aparece como "existe / não existe".
  conexoes.onesignal.orgId = onesignalOrgId_();
  conexoes.onesignal.appId = onesignalAppId_();
  conexoes.onesignal.chaveEnvio = !!pushCfg_().chave;
  return { ok: true, conexoes: conexoes, servico: RENDER_SERVICO_, enderecos: { feedCsv: SITE_URL + 'feed.csv', feedXml: SITE_URL + 'feed.xml', sitemap: SITE_URL + 'sitemap.xml' } };
}
function statusConexoes_() { return jsonResponse_(statusConexoesObj_()); }

function salvarConexao_(body) {
  const nome = String(body.nome || '');
  const c = CONEXOES_[nome];
  if (!c) return jsonResponse_({ ok: false, erro: 'Conexão desconhecida.' });
  const props = PropertiesService.getScriptProperties();
  if (body.apagar === true) {
    props.deleteProperty(c.prop);
    props.deleteProperty(c.prop + '_EM');
    if (nome === 'onesignal') props.deleteProperty('ONESIGNAL_ORG_ID'); // a chave de ENVIO do app (ONESIGNAL_REST_API_KEY) continua: é ela que manda os avisos
    registrarAudit_(body.funcionario, 'apagarConexao', '', { conexao: nome });
    return statusConexoes_();
  }
  const chave = String(body.chave || '').trim();
  if (!c.padrao.test(chave)) return jsonResponse_({ ok: false, erro: 'Essa chave parece incompleta ou com caracteres estranhos. Cole de novo, só a chave, sem espaços.' });
  let extra = {};
  if (nome === 'onesignal') {
    const orgId = String(body.orgId || onesignalOrgId_() || '').trim().toLowerCase();
    if (!ONESIGNAL_UUID_.test(orgId)) return jsonResponse_({ ok: false, erro: 'O Organization ID parece errado: é um código de letras, números e traços (como b036c245-6f4a-4074-b7af-eaaed786933a). Copie de novo na tela Security do OneSignal.' });
    extra.orgId = orgId;
  }
  const t = conexaoTestar_(nome, chave, extra);
  if (!t.ok) return jsonResponse_({ ok: false, erro: 'Não guardei a chave: ' + t.detalhe });
  props.setProperty(c.prop, chave);
  props.setProperty(c.prop + '_EM', new Date().toISOString());
  if (extra.orgId) props.setProperty('ONESIGNAL_ORG_ID', extra.orgId);
  registrarAudit_(body.funcionario, 'salvarConexao', '', { conexao: nome });
  const r = statusConexoesObj_();
  r.detalhe = t.detalhe;
  return jsonResponse_(r);
}

/** Regras (Redirects/Rewrites) do site no Render. */
function renderRegras_(chave) {
  const r = conexaoHttp_('get', RENDER_API_ + '/services/' + RENDER_SERVICO_ + '/routes?limit=100', chave);
  if (r.codigo !== 200 || !Array.isArray(r.dados)) return { ok: false, erro: conexaoMsgApi_(r) };
  return { ok: true, regras: r.dados.map(function (x) { const q = (x && x.route) || x || {}; return { id: q.id, source: q.source, destination: q.destination, type: q.type }; }) };
}
/** Quais regras faltam e quais existem mas apontam para outro lugar (estas NÃO são mexidas). */
function renderAnalisar_(regras) {
  const faltam = [], divergentes = [];
  REGRAS_RENDER_.forEach(function (src) {
    const achou = regras.filter(function (x) { return x.source === src; })[0];
    if (!achou) faltam.push(src);
    else if (achou.type !== 'rewrite' || achou.destination !== WORKER_BASE_ + src) divergentes.push(src);
  });
  return { faltam: faltam, divergentes: divergentes };
}
function renderCriarRegras_(body) {
  const chave = conexaoChave_('render');
  if (!chave) return jsonResponse_({ ok: false, erro: 'Guarde primeiro a chave do Render (em Conexões).' });
  const g = renderRegras_(chave);
  if (!g.ok) return jsonResponse_({ ok: false, erro: g.erro });
  const a = renderAnalisar_(g.regras);
  const criadas = [];
  for (let i = 0; i < a.faltam.length; i++) {
    const src = a.faltam[i];
    const r = conexaoHttp_('post', RENDER_API_ + '/services/' + RENDER_SERVICO_ + '/routes', chave, { type: 'rewrite', source: src, destination: WORKER_BASE_ + src });
    if (r.codigo !== 201 && r.codigo !== 200) {
      registrarAudit_(body.funcionario, 'renderCriarRegras', '', { criadas: criadas, falhouEm: src });
      return jsonResponse_({ ok: false, erro: 'Não consegui criar a regra ' + src + ': ' + conexaoMsgApi_(r), criadas: criadas });
    }
    criadas.push(src);
  }
  registrarAudit_(body.funcionario, 'renderCriarRegras', '', { criadas: criadas, divergentes: a.divergentes });
  return jsonResponse_({ ok: true, criadas: criadas, divergentes: a.divergentes, jaExistiam: REGRAS_RENDER_.length - a.faltam.length - a.divergentes.length });
}
function renderDeploy_(body) {
  const chave = conexaoChave_('render');
  if (!chave) return jsonResponse_({ ok: false, erro: 'Guarde primeiro a chave do Render (em Conexões).' });
  const r = conexaoHttp_('post', RENDER_API_ + '/services/' + RENDER_SERVICO_ + '/deploys', chave, { clearCache: 'do_not_clear' });
  const d = r.dados && (r.dados.deploy || r.dados);
  if ([200, 201, 202].indexOf(r.codigo) === -1 || !d || !d.id) return jsonResponse_({ ok: false, erro: 'Não consegui publicar agora: ' + conexaoMsgApi_(r) });
  registrarAudit_(body.funcionario, 'renderDeploy', '', { deploy: d.id });
  return jsonResponse_({ ok: true, deployId: d.id, status: d.status || 'created' });
}
function renderStatusDeploy_(body) {
  const chave = conexaoChave_('render');
  const id = String(body.deployId || '');
  if (!chave) return jsonResponse_({ ok: false, erro: 'Sem chave do Render guardada.' });
  if (!/^dep-[A-Za-z0-9]{8,40}$/.test(id)) return jsonResponse_({ ok: false, erro: 'Identificador de publicação inválido.' });
  const r = conexaoHttp_('get', RENDER_API_ + '/services/' + RENDER_SERVICO_ + '/deploys/' + id, chave);
  if (r.codigo !== 200 || !r.dados) return jsonResponse_({ ok: false, erro: conexaoMsgApi_(r) });
  return jsonResponse_({ ok: true, status: String(r.dados.status || '') });
}

/**
 * v3.5: liga os avisos pelo painel. Idempotente: só mexe no que ainda não está certo.
 *  1) lê o app (confere que o Organization ID guardado é o dele);
 *  2) liga a plataforma Web (Typical Site) para o domínio da loja, se ainda não estiver;
 *  3) grava ONESIGNAL_APP_ID se faltar;
 *  4) se NÃO existe uma chave de envio que funcione (ONESIGNAL_REST_API_KEY), CRIA uma nova pelo OneSignal e guarda (nunca devolvida);
 *  5) confere se o OneSignal já responde "plataforma Web ligada" (sem o cache de 1 hora).
 */
function onesignalConfigurar_(body) {
  const chave = conexaoChave_('onesignal'), orgId = onesignalOrgId_();
  if (!chave || !orgId) return jsonResponse_({ ok: false, erro: 'Guarde primeiro o Organization ID e a chave do OneSignal (em Conexões).' });
  const props = PropertiesService.getScriptProperties();
  const appId = onesignalAppId_();
  const passos = [];
  const passo = function (id, ok, detalhe) { passos.push({ id: id, ok: ok, detalhe: detalhe }); };
  const parar = function (erro) {
    registrarAudit_(body.funcionario, 'onesignalConfigurar', '', { passos: passos.map(function (x) { return x.id + ':' + x.ok; }), falhouEm: erro.slice(0, 120) });
    return jsonResponse_({ ok: false, erro: erro, passos: passos });
  };

  // 1) ler o app
  const g = conexaoHttp_('get', PUSH_API_ + '/apps/' + appId, chave, undefined, 'Key');
  if (g.codigo !== 200 || !g.dados || !g.dados.id) return parar('Não consegui ler o app no OneSignal: ' + conexaoMsgApi_(g));
  const app = g.dados;
  if (app.organization_id && String(app.organization_id) !== orgId) return parar('O Organization ID guardado não é o do app do TENKiTER. Guarde de novo a chave e o código (tela Security do OneSignal).');
  passo('app', true, 'App "' + String(app.name || appId).slice(0, 60) + '" encontrado.');

  // 2) plataforma Web
  const origem = SITE_URL.replace(/\/+$/, '');
  const jaTem = app.chrome_web_origin === origem && app.safari_site_origin === origem && !!app.site_name && !!app.chrome_web_default_notification_icon;
  if (jaTem) passo('web', true, 'A plataforma Web já estava configurada para ' + origem + '.');
  else {
    const corpo = {
      organization_id: orgId, site_name: String(app.site_name || 'TENKiTER Modas'),
      chrome_web_origin: origem, safari_site_origin: origem, safari_apns_p12: '', safari_apns_p12_password: '',
      chrome_web_default_notification_icon: SITE_URL + 'icon-256.png'
    };
    const p = conexaoHttp_('put', PUSH_API_ + '/apps/' + appId, chave, corpo, 'Key');
    if (p.codigo !== 200) return parar('Não consegui ligar a plataforma Web: ' + conexaoMsgApi_(p));
    passo('web', true, 'Plataforma Web ligada para ' + origem + '.');
  }

  // 3) App ID nas Propriedades do Script
  if (!String(props.getProperty('ONESIGNAL_APP_ID') || '').trim()) { props.setProperty('ONESIGNAL_APP_ID', appId); passo('appid', true, 'App ID gravado no servidor.'); }

  // 4) chave de envio do app
  let envioOk = false, recusada = false;
  if (pushCfg_().chave) {
    const t = pushChamar_('get', '/notifications?app_id=' + encodeURIComponent(appId) + '&limit=1');
    envioOk = t.codigo === 200; recusada = !envioOk;
  }
  if (envioOk) passo('envio', true, 'A chave de envio que já estava no servidor funciona.');
  else {
    const c = conexaoHttp_('post', PUSH_API_ + '/apps/' + appId + '/auth/tokens', chave, { name: 'TENKiTER site (Apps Script)' }, 'Key');
    const nova = c.dados && typeof c.dados.formatted_token === 'string' ? c.dados.formatted_token.trim() : '';
    if ((c.codigo !== 200 && c.codigo !== 201) || nova.length < 20) return parar('A plataforma Web ficou ligada, mas não consegui criar a chave de envio: ' + conexaoMsgApi_(c));
    props.setProperty('ONESIGNAL_REST_API_KEY', nova);
    props.setProperty('ONESIGNAL_REST_API_KEY_EM', new Date().toISOString());
    passo('envio', true, recusada ? 'A chave de envio antiga foi recusada pelo OneSignal; criei uma nova e guardei no servidor.' : 'Criei a chave de envio do app e guardei no servidor.');
  }

  // 5) conferência final (sem o cache do OneSignal)
  const web = pushWebPronto_(appId, true);
  passo('conferencia', web === true ? true : null, web === true ? 'O OneSignal já responde: plataforma Web ligada.' : 'Ligado, mas o OneSignal ainda não confirmou. Toque em "Conferir de novo" daqui a um minuto.');
  registrarAudit_(body.funcionario, 'onesignalConfigurar', '', { passos: passos.map(function (x) { return x.id + ':' + x.ok; }) });
  const r = statusConexoesObj_();
  r.passos = passos; r.webPronto = web;
  return jsonResponse_(r);
}

/** Confere tudo de uma vez. Cada item: { id, rotulo, ok: true|false|null (null = só informação), detalhe, acao } -- `acao` é o botão que resolve (renderCriarRegras | chave:render | chave:cloudflare). */
function testarConexoes_() {
  const itens = [];
  const add = function (id, rotulo, ok, detalhe, acao) { itens.push({ id: id, rotulo: rotulo, ok: ok, detalhe: detalhe, acao: acao || '' }); };
  const props = PropertiesService.getScriptProperties();
  const rk = conexaoChave_('render'), ck = conexaoChave_('cloudflare');

  // 1) o que o público enxerga (sem chave nenhuma)
  let resp = [];
  try {
    resp = UrlFetchApp.fetchAll([SITE_URL + 'feed.csv', SITE_URL + 'feed.xml', SITE_URL + 'sitemap.xml'].map(function (u) { return { url: u, method: 'get', muteHttpExceptions: true, followRedirects: true }; }));
  } catch (e) { resp = []; }
  const lido = function (i) { const r = resp[i]; return r ? { codigo: r.getResponseCode(), texto: String(r.getContentText() || '') } : { codigo: 0, texto: '' }; };
  const csv = lido(0), xml = lido(1), mapa = lido(2);
  const acaoSite = rk ? 'renderCriarRegras' : 'chave:render';
  const nCsv = csv.codigo === 200 && /^id,title,/.test(csv.texto) ? Math.max(0, csv.texto.trim().split('\n').length - 1) : -1;
  add('feed-csv', 'Catálogo para a Meta (feed.csv)', nCsv >= 0, nCsv >= 0 ? nCsv + ' peça(s) no feed do seu site.' : 'O endereço ' + SITE_URL + 'feed.csv não entregou o catálogo (falta a regra no Render).', nCsv >= 0 ? '' : acaoSite);
  const nXml = xml.codigo === 200 && /<rss[\s>]/.test(xml.texto) ? (xml.texto.match(/<item>/g) || []).length : -1;
  add('feed-xml', 'Catálogo para o Google (feed.xml)', nXml >= 0, nXml >= 0 ? nXml + ' peça(s) no feed do seu site.' : 'O endereço ' + SITE_URL + 'feed.xml não entregou o catálogo (falta a regra no Render).', nXml >= 0 ? '' : acaoSite);
  const nMapa = mapa.codigo === 200 && /<urlset/.test(mapa.texto) ? (mapa.texto.match(/<loc>[^<]*\/p\//g) || []).length : -1;
  add('sitemap', 'Mapa do site para o Google (sitemap.xml)', nMapa > 0,
    nMapa > 0 ? 'Lista ' + nMapa + ' peça(s) além das páginas fixas.' : (nMapa === 0 ? 'Está mostrando só as páginas fixas: um arquivo sitemap.xml antigo no repositório do site ganha da regra do Render e precisa ser removido de lá.' : 'O endereço ' + SITE_URL + 'sitemap.xml não respondeu direito.'),
    nMapa > 0 || nMapa === 0 ? '' : acaoSite);

  // 2) avisos (OneSignal)
  const appId = onesignalAppId_();
  const ok_ = conexaoChave_('onesignal');
  const web = pushWebPronto_(appId, true);
  add('onesignal', 'Avisos (OneSignal, plataforma Web)', web === null ? null : web,
    web === null ? 'Não consegui conferir agora.' : (web ? 'Plataforma Web ligada: os avisos podem chegar.' : 'A plataforma Web do OneSignal ainda NÃO foi ligada: sem isso nenhum aviso chega.' + (ok_ ? '' : ' Guarde a chave do OneSignal abaixo e o painel liga sozinho.')),
    web === false && ok_ ? 'onesignalConfigurar' : '');
  const envio = pushCfg_().chave;
  if (!envio) add('onesignal-envio', 'Avisos (chave de envio do app)', false, 'Falta a chave de envio do app (ONESIGNAL_REST_API_KEY): sem ela o site não consegue mandar avisos.' + (ok_ ? '' : ' Guarde a chave do OneSignal abaixo e o painel cria a de envio sozinho.'), ok_ ? 'onesignalConfigurar' : '');
  else {
    const te = pushChamar_('get', '/notifications?app_id=' + encodeURIComponent(appId) + '&limit=1');
    add('onesignal-envio', 'Avisos (chave de envio do app)', te.codigo === 200, te.codigo === 200 ? 'A chave de envio funciona.' : 'O OneSignal recusou a chave de envio guardada.' + (ok_ ? '' : ' Guarde a chave do OneSignal abaixo e o painel cria outra.'), te.codigo === 200 || !ok_ ? '' : 'onesignalConfigurar');
  }
  if (!ok_) add('onesignal-chave', 'OneSignal (ligar os avisos pelo painel)', null, 'Sem chave guardada. Guarde o Organization ID e a chave abaixo para o painel ligar a plataforma Web e criar a chave de envio sozinho.', 'chave:onesignal');
  else {
    const t = conexaoTestar_('onesignal', ok_);
    add('onesignal-chave', 'OneSignal (ligar os avisos pelo painel)', t.ok, t.detalhe, t.ok ? '' : 'chave:onesignal');
  }

  // 3) Render
  if (!rk) add('render-chave', 'Render (publicar e regras)', null, 'Sem chave guardada. Guarde abaixo para o painel conseguir conferir as regras e publicar o site.', 'chave:render');
  else {
    const t = conexaoTestar_('render', rk);
    add('render-chave', 'Render (publicar e regras)', t.ok, t.detalhe, t.ok ? '' : 'chave:render');
    if (t.ok) {
      const g = renderRegras_(rk);
      if (!g.ok) add('render-regras', 'Regras do Render', false, g.erro);
      else {
        const a = renderAnalisar_(g.regras);
        const bom = !a.faltam.length && !a.divergentes.length;
        let det = bom ? 'Todas no lugar: ' + REGRAS_RENDER_.join(', ') + '.' : '';
        if (a.faltam.length) det += 'Faltam: ' + a.faltam.join(', ') + '. ';
        if (a.divergentes.length) det += 'Apontam para outro lugar (não mexo): ' + a.divergentes.join(', ') + '.';
        add('render-regras', 'Regras do Render', bom, det.trim(), a.faltam.length ? 'renderCriarRegras' : '');
      }
      const d = conexaoHttp_('get', RENDER_API_ + '/services/' + RENDER_SERVICO_ + '/deploys?limit=1', rk);
      const ult = d.codigo === 200 && Array.isArray(d.dados) && d.dados[0] ? (d.dados[0].deploy || d.dados[0]) : null;
      if (ult) add('render-deploy', 'Último deploy do site', ult.status === 'live' ? true : null, 'Estado: ' + String(ult.status || '?') + (ult.createdAt ? ' (' + Utilities.formatDate(new Date(ult.createdAt), 'America/Fortaleza', 'dd/MM/yyyy HH:mm') + ')' : '') + '.');
    }
  }

  // 4) Cloudflare
  if (!ck) add('cloudflare-chave', 'Cloudflare (Worker das miniaturas, feed e mapa)', null, 'Sem chave guardada. É opcional: serve só para conferir quando o Worker foi publicado.', 'chave:cloudflare');
  else {
    const t = conexaoTestar_('cloudflare', ck);
    add('cloudflare-chave', 'Cloudflare (Worker das miniaturas, feed e mapa)', t.ok, t.detalhe, t.ok ? '' : 'chave:cloudflare');
    if (t.ok) {
      const c = conexaoHttp_('get', CLOUDFLARE_API_ + '/accounts', ck);
      const conta = c.codigo === 200 && c.dados && c.dados.result && c.dados.result.length === 1 ? c.dados.result[0].id : '';
      if (!conta) add('worker', 'Worker ' + WORKER_NOME_, null, 'A chave enxerga mais de uma conta (ou nenhuma); não consegui achar o Worker.');
      else {
        const w = conexaoHttp_('get', CLOUDFLARE_API_ + '/accounts/' + conta + '/workers/scripts', ck);
        const achou = w.codigo === 200 && w.dados && Array.isArray(w.dados.result) ? w.dados.result.filter(function (x) { return x.id === WORKER_NOME_; })[0] : null;
        if (achou) add('worker', 'Worker ' + WORKER_NOME_, true, 'Publicado em ' + (achou.modified_on ? Utilities.formatDate(new Date(achou.modified_on), 'America/Fortaleza', 'dd/MM/yyyy HH:mm') : 'data desconhecida') + '.');
        else add('worker', 'Worker ' + WORKER_NOME_, w.codigo === 200 ? false : null, w.codigo === 200 ? 'Não encontrei o Worker ' + WORKER_NOME_ + ' nesta conta.' : conexaoMsgApi_(w));
      }
    }
  }
  return jsonResponse_({ ok: true, itens: itens, quando: new Date().toISOString() });
}
